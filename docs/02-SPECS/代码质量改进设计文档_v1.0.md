### 2.1 错误码系统统一 (40小时)

#### 2.1.1 问题分析

**当前错误处理模式**:

```go
// ❌ 错误模式1：直接引用errno
func (s *service) GetBot(ctx context.Context, id int64) (*Bot, error) {
    bot, err := s.repo.Get(ctx, id)
    if err != nil {
        return nil, errno.ErrBotNotFound  // ❌ 绕过errorx包装
    }
    return bot, nil
}

// ❌ 错误模式2：混合使用errorx和errno
func (s *service) CreateBot(ctx context.Context, req *CreateBotRequest) error {
    if err := validate(req); err != nil {
        return errorx.WrapByCode(err, errno.ErrBotInvalidParam)  // ✅ 使用errorx
    }
    // ...
    return errno.ErrBotCreateFailed  // ❌ 直接返回errno
}
```

**问题**:
1. 统计显示：domain层中`errno.`引用375次，`errorx.`使用0次
2. 直接返回`errno.ErrXXX`绕过了`EnhancedError`的增强功能
3. 无法记录错误堆栈、请求上下文等调试信息

#### 2.1.2 正确模式

**标准错误处理模式**:

```go
// ✅ 正确模式1：创建新错误
func (s *service) GetBot(ctx context.Context, id int64) (*Bot, error) {
    bot, err := s.repo.Get(ctx, id)

            errorx.KV("bot_id", id),           // 添加上下文
            errorx.KV("tenant_id", tenantID),   // 添加租户信息

// ✅ 正确模式2：包装底层错误
```

#### 2.1.3 自动化迁移脚本

**文件**: `scripts/migrate_errno_to_errorx.sh`

```bash
#!/bin/bash
# 自动化errno到errorx的迁移脚本

set -e

BACKEND_DIR="backend/domain"
BACKUP_DIR=".backup/errno_migration_$(date +%Y%m%d_%H%M%S)"

# 创建备份目录
mkdir -p "$BACKUP_DIR"

echo "开始errno到errorx迁移..."
echo "备份目录: $BACKUP_DIR"

# 查找所有Go文件
find "$BACKEND_DIR" -name "*.go" -not -name "*_test.go" | while read file; do
    echo "处理文件: $file"

    # 备份原文件
    backup_path="$BACKUP_DIR/${file#$BACKEND_DIR/}"
    mkdir -p "$(dirname "$backup_path")"
    cp "$file" "$backup_path"

    # 模式1: return errno.ErrXXX → return errorx.New(errno.ErrXXX)
    sed -i -E 's/return\s+errno\.([A-Z][a-zA-Z0-9]+)/return errorx.New(errno.\1)/g' "$file"

    # 模式2: return nil, errno.ErrXXX → return nil, errorx.New(errno.ErrXXX)
    sed -i -E 's/return\s+nil,\s+errno\.([A-Z][a-zA-Z0-9]+)/return nil, errorx.New(errno.\1)/g' "$file"

    # 模式3: if xxx { return errno.ErrXXX } → if xxx { return errorx.New(errno.ErrXXX) }
    sed -i -E 's/\{\s*return\s+errno\.([A-Z][a-zA-Z0-9]+)\s*\}/{ return errorx.New(errno.\1) }/g' "$file"

    # 模式4: errorx.WrapByCode(err, errno.ErrXXX) → errorx.Wrap(err, errno.ErrXXX)
    sed -i -E 's/errorx\.WrapByCode\(([^,]+),\s*errno\.([A-Z][a-zA-Z0-9]+)\)/errorx.Wrap(\1, errno.\2)/g' "$file"

    echo "✓ 完成: $file"
done

echo ""
echo "迁移完成！"
echo ""
echo "验证迁移结果："
echo "1. 检查剩余的直接errno引用:"
grep -rn "return.*errno\." "$BACKEND_DIR" --include="*.go" | grep -v "_test.go" | head -20 || echo "  ✓ 无残留"
echo ""
echo "2. 统计errorx使用:"
grep -rn "errorx\." "$BACKEND_DIR" --include="*.go" | wc -l
echo ""
echo "3. 运行测试验证:"
echo "   go test ./backend/domain/..."
echo ""
echo "如需回滚，使用备份: $BACKUP_DIR"
```

**重构方案**:

**文件**: `backend/domain/agent/singleagent/service/display_info_updater.go`

```go
package service

import (
	"reflect"

	"github.com/coze-dev/coze-studio/backend/domain/agent/singleagent/entity"
)

// UpdateDisplayInfo 更新DisplayInfo的通用函数
func UpdateDisplayInfo(target, source *entity.DisplayInfo) {
	if source == nil || target == nil {
		return
	}

	// 1. 更新TabDisplayInfo
	updateTabDisplayInfo(target.TabDisplayInfo, source.TabDisplayInfo)

	// 2. 更新其他DisplayInfo字段
	updateNonNilFields(target, source)
}

// updateTabDisplayInfo 更新TabDisplayInfo
func updateTabDisplayInfo(target, source *entity.TabDisplayInfo) {
	if source == nil || target == nil {
		return
	}

	// 使用反射更新非零字段
	targetValue := reflect.ValueOf(target).Elem()
	sourceValue := reflect.ValueOf(source).Elem()
 targetType := reflect.TypeOf(target).Elem()

	for i := 0; i < targetValue.NumField(); i++ {
		field := targetValue.Field(i)
		sourceField := sourceValue.Field(i)

		// 跳过零值字段
		if sourceField.IsZero() {
			continue
		}

		// 检查是否为指针类型
		if sourceField.Kind() == reflect.Ptr {
			if !sourceField.IsNil() {
				// 创建新的指针副本
				newValue := reflect.New(sourceField.Elem().Type())
				newValue.Elem().Set(sourceField.Elem())
				field.Set(newValue)
			}
		} else {
			field.Set(sourceField)
		}
	}
}

// updateNonNilFields 更新非零字段
func updateNonNilFields(target, source interface{}) {
	targetValue := reflect.ValueOf(target).Elem()
	sourceValue := reflect.ValueOf(source).Elem()

	for i := 0; i < targetValue.NumField(); i++ {
		sourceField := sourceValue.Field(i)
		targetField := targetValue.Field(i)

		// 跳过零值
		if sourceField.IsZero() {
			continue
		}

		// 更新非零字段
		if sourceField.CanSet() {
			targetField.Set(sourceField)
		}
	}
}
```

#### 2.2.2 CRUD泛型封装

**文件**: `backend/domain/common/repository/base_repository.go`

```go
package repository

import (
	"context"

	"github.com/coze-dev/coze-studio/backend/pkg/errorx"
	"github.com/coze-dev/coze-studio/backend/types/errno"
)

// BaseEntity 基础实体接口
type BaseEntity interface {
	GetID() int64
	SetID(int64)
}

// BaseRepository 通用仓储接口
type BaseRepository[T BaseEntity] interface {
	Get(ctx context.Context, id int64) (*T, error)
	Create(ctx context.Context, entity *T) error
	Update(ctx context.Context, entity *T) error
	Delete(ctx context.Context, id int64) error
	List(ctx context.Context, offset, limit int) ([]*T, int64, error)
}

// BaseRepositoryImpl 通用仓储实现
type BaseRepositoryImpl[T BaseEntity] struct {
	dao DAO[T]
}

// NewBaseRepository 创建通用仓储
func NewBaseRepository[T BaseEntity](dao DAO[T]) BaseRepository[T] {
	return &BaseRepositoryImpl[T]{dao: dao}
}

// Get 获取单个实体
func (r *BaseRepositoryImpl[T]) Get(ctx context.Context, id int64) (*T, error) {
	entity, err := r.dao.Get(ctx, id)
	if err != nil {
		return nil, errorx.Wrap(err, errno.ErrDatabase)
	}
	if entity == nil {
		return nil, errorx.New(errno.ErrNotFound, errorx.KV("id", id))
	}
	return entity, nil
}

// Create 创建实体
func (r *BaseRepositoryImpl[T]) Create(ctx context.Context, entity *T) error {
	if entity == nil {
		return errorx.New(errno.ErrInvalidParam, errorx.KV("msg", "entity is nil"))
	}
	return errorx.Wrap(r.dao.Create(ctx, entity), errno.ErrDatabase)
}

// Update 更新实体
func (r *BaseRepositoryImpl[T]) Update(ctx context.Context, entity *T) error {
	if entity == nil {
		return errorx.New(errno.ErrInvalidParam, errorx.KV("msg", "entity is nil"))
	}
	if (*entity).GetID() == 0 {
		return errorx.New(errno.ErrInvalidParam, errorx.KV("msg", "id is required"))
	}
	return errorx.Wrap(r.dao.Update(ctx, entity), errno.ErrDatabase)
}

// Delete 删除实体
func (r *BaseRepositoryImpl[T]) Delete(ctx context.Context, id int64) error {
	if id == 0 {
		return errorx.New(errno.ErrInvalidParam, errorx.KV("msg", "id is required"))
	}
	return errorx.Wrap(r.dao.Delete(ctx, id), errno.ErrDatabase)
}

// List 列表查询
func (r *BaseRepositoryImpl[T]) List(ctx context.Context, offset, limit int) ([]*T, int64, error) {
	if offset < 0 {
		return nil, 0, errorx.New(errno.ErrInvalidParam, errorx.KV("offset", offset))
	}
	if limit <= 0 || limit > 1000 {
		return nil, 0, errorx.New(errno.ErrInvalidParam, errorx.KV("limit", limit))
	}

	entities, total, err := r.dao.List(ctx, offset, limit)
	if err != nil {
		return nil, 0, errorx.Wrap(err, errno.ErrDatabase)
	}

	return entities, total, nil
}

// DAO 数据访问对象接口
type DAO[T BaseEntity] interface {
	Get(ctx context.Context, id int64) (*T, error)
	Create(ctx context.Context, entity *T) error
	Update(ctx context.Context, entity *T) error
	Delete(ctx context.Context, id int64) error
	List(ctx context.Context, offset, limit int) ([]*T, int64, error)
}
```

```go
type Permission struct {}
type Bot struct {}

```

**批量重命名脚本**: `scripts/rename_impl_classes.sh`

```bash
#!/bin/bash
# 批量重命名实现类

find backend/domain -name "*.go" -not -name "*_test.go" | while read file; do
    # 小写开头 → 大写开头+Impl
    sed -i -E 's/type\s+([a-z][a-zA-Z0-9]*)\s+struct/type \1Impl struct/g' "$file"
    sed -i -E 's/&([a-z][a-zA-Z0-9]*)\{/\&\1Impl{/g' "$file"

    # 无后缀 → +Impl后缀
    # 需要手动review，避免误改
done
```

### 3.3 质量门禁

**PR提交前必须通过**:

```bash
#!/bin/bash
# .github/workflows/quality-gate.yml

echo "🚪 代码质量门禁检查"

# 1. 错误码覆盖率
echo "1️⃣ 检查错误码覆盖率..."
errorx_coverage=$(./scripts/calc_errno_coverage.sh)
if [ $errorx_coverage -lt 95 ]; then
    echo "❌ 错误码覆盖率不足: $errorx_coverage% (目标≥95%)"
    exit 1
fi

# 2. 多租户隔离
echo "2️⃣ 检查多租户隔离..."
if ! ./scripts/check_tenant_isolation.sh; then
    echo "❌ 多租户隔离检查失败"
    exit 1
fi

# 3. 测试覆盖率
echo "3️⃣ 检查测试覆盖率..."
coverage=$(go test -cover ./... | grep total | awk '{print $3}' | sed 's/%//')
if [ $(echo "$coverage < 80" | bc) -eq 1 ]; then
    echo "❌ 测试覆盖率不足: $coverage% (目标≥80%)"
    exit 1
fi

# 4. 代码重复
echo "4️⃣ 检查代码重复..."
duplication=$(jsc --format json backend/domain/ | jq '.duplication')
if [ $(echo "$duplication > 5" | bc) -eq 1 ]; then
    echo "⚠️  代码重复率: $duplication% (建议≤5%)"
fi

echo "✅ 质量门禁通过！"
```
