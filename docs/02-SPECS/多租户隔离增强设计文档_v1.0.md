## 1. 项目背景与目标

### 1.1 当前隔离状态

根据深度分析，ZKER项目多租户隔离现状：

| 隔离维度 | 当前状态 | 问题 | 风险等级 |
|----------|----------|------|----------|
| **数据库层** | tenant_id字段存在 | 50+处查询缺少过滤 | 🔴 高 |
| **缓存层** | 部分隔离 | 缓存Key缺少租户前缀 | 🔴 高 |
| **向量索引** | 未隔离 | 跨租户向量检索 | 🟡 中 |
| **全文索引** | 未隔离 | 跨租户全文检索 | 🟡 中 |
| **文件存储** | 部分隔离 | 路径未完全隔离 | 🟡 中 |

**总体隔离得分**: 65/100 → **目标**: ≥98/100

### 1.2 关键风险

#### P0级风险 (必须立即修复)

1. **GORM查询缺少tenant_id过滤** 🔴
   - 统计：50+处 `.Where()` 查询未包含 `tenant_id`
   - 示例代码：
   ```go
   // ❌ 危险：缺少租户过滤
   func (r *AuditLogRepository) FindByUserID(userID int64) ([]*AuditLog, error) {
       return r.query.Where("user_id = ?", userID).Find(&logs).Error  // 跨租户泄露
   }
   ```
   - **影响**: 用户A可以查询到用户B的操作日志

2. **缓存Key未包含租户标识** 🔴
   - 示例代码：
   ```go
   // ❌ 危险：缓存Key无租户隔离
   cacheKey := fmt.Sprintf("bot:%d", botID)
   cache.Set(key, data, ttl)
   ```
   - **影响**: 租户A的缓存数据可能被租户B读取

3. **向量索引未按租户隔离** 🟡
   - 当前：所有租户共享Milvus Collection
   - **影响**: 跨租户知识库向量检索

### 1.3 增强目标

**核心目标**: 建立完整的租户隔离体系，确保零数据泄露。

| 维度 | 当前 | 目标 | 验收标准 |
|------|------|------|----------|
| **数据库查询** | 65%过滤 | 100%过滤 | 所有查询包含tenant_id |
| **缓存隔离** | 30%隔离 | 100%隔离 | 所有Key包含租户前缀 |
| **向量隔离** | 0% | 100% | 租户级Partition |
| **全文隔离** | 0% | 100% | 租户级索引 |
| **文件隔离** | 70% | 100% | 租户级目录 |

func TenantContextMiddleware() app.HandlerFunc {
	return func(ctx context.Context, c *app.RequestContext) {
		// 1. 从JWT token中提取租户信息
		token := string(c.GetHeader("Authorization"))
		if token == "" {
			c.JSON(401, map[string]string{"error": "missing authorization token"})
			c.Abort()
			return
		}

		// 解析JWT（示例，实际应使用jwt库）

		if err != nil {
			c.JSON(401, map[string]string{"error": "invalid token"})
			c.Abort()
			return
		}

		// 2. 构建租户上下文
		tenantCtx := &TenantContext{
			TenantID: tenantID,
			UserID:   userID,
		}

		// 3. 从数据库加载租户权限

		ctx = context.WithValue(ctx, tenantKey{}, tenantCtx)

		// 5. 传递给下一个处理器
		c.Next(ctx)
	}

package repository

import (
	"context"

	"github.com/coze-dev/coze-studio/backend/api/middleware"
	"gorm.io/gorm"
)

// TenantAwareRepository 租户感知仓储
type TenantAwareRepository struct {
	db *gorm.DB
}

// QueryWithTenant 自动添加租户过滤的查询
func (r *TenantAwareRepository) QueryWithTenant(ctx context.Context) *gorm.DB {
	tenantCtx, err := middleware.GetTenantContext(ctx)
	if err != nil {
		// 如果没有租户上下文，使用默认租户（兼容旧代码）
		return r.db
	}

	// 自动添加租户过滤
	return r.db.Where("tenant_id = ?", tenantCtx.TenantID)
}

// QueryWithTenantAndSpace 兼容旧版：同时过滤tenant_id和space_id
func (r *TenantAwareRepository) QueryWithTenantAndSpace(ctx context.Context) *gorm.DB {
	tenantCtx, err := middleware.GetTenantContext(ctx)
	if err != nil {
		return r.db
	}

	// 优先使用tenant_id，兼容space_id
	if tenantCtx.TenantID != "" {
		return r.db.Where("tenant_id = ?", tenantCtx.TenantID)
	}
	if tenantCtx.SpaceID != "" {
		return r.db.Where("space_id = ?", tenantCtx.SpaceID)
	}

	return r.db
}
```

-- 为所有核心表添加行级安全策略

-- 1. 启用RLS
ALTER TABLE bots ENABLE ROW LEVEL SECURITY;
ALTER TABLE conversations ENABLE ROW LEVEL SECURITY;
ALTER TABLE knowledge ENABLE ROW LEVEL SECURITY;
ALTER TABLE workflows ENABLE ROW LEVEL SECURITY;

-- 2. 创建租户隔离策略
-- 策略1: 只能查看本租户数据
CREATE POLICY tenant_isolation_select ON bots
    FOR SELECT
    USING (tenant_id = current_setting('app.current_tenant_id')::text);

CREATE POLICY tenant_isolation_select ON conversations
    FOR SELECT
    USING (tenant_id = current_setting('app.current_tenant_id')::text);

CREATE POLICY tenant_isolation_select ON knowledge
    FOR SELECT
    USING (tenant_id = current_setting('app.current_tenant_id')::text);

-- 策略2: 只能插入本租户数据
CREATE POLICY tenant_isolation_insert ON bots
    FOR INSERT
    WITH CHECK (tenant_id = current_setting('app.current_tenant_id')::text);

-- 策略3: 只能更新本租户数据
CREATE POLICY tenant_isolation_update ON bots
    FOR UPDATE
    USING (tenant_id = current_setting('app.current_tenant_id')::text);

-- 策略4: 只能删除本租户数据
CREATE POLICY tenant_isolation_delete ON bots
    FOR DELETE
    USING (tenant_id = current_setting('app.current_tenant_id')::text);

-- 3. 创建函数：自动设置tenant_id
CREATE OR REPLACE FUNCTION set_tenant_id()
RETURNS TRIGGER AS $$
BEGIN
    -- 如果tenant_id为空，从会话设置中获取
    IF NEW.tenant_id IS NULL OR NEW.tenant_id = '' THEN
        NEW.tenant_id := current_setting('app.current_tenant_id')::text;
    END IF;

    -- 验证tenant_id匹配
    IF NEW.tenant_id != current_setting('app.current_tenant_id')::text THEN
        RAISE EXCEPTION 'Tenant ID mismatch';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 4. 添加触发器
CREATE TRIGGER enforce_tenant_id_before_insert
    BEFORE INSERT ON bots
    FOR EACH ROW
    EXECUTE FUNCTION set_tenant_id();

CREATE TRIGGER enforce_tenant_id_before_update
    BEFORE UPDATE ON bots
    FOR EACH ROW
    EXECUTE FUNCTION set_tenant_id();
```

package cache

import (
	"context"
	"fmt"

	"github.com/coze-dev/coze-studio/backend/api/middleware"
)

// TenantCacheKey 租户缓存Key生成器
type TenantCacheKey struct {
	prefix string // 前缀，如"bot", "user", "knowledge"
}

// Build 生成包含租户ID的缓存Key
func (b *TenantCacheKey) Build(ctx context.Context, parts ...string) string {
	// 1. 获取租户上下文
	tenantCtx, err := middleware.GetTenantContext(ctx)
	if err != nil {
		// 兼容旧代码：没有租户上下文时使用默认
		return b.buildWithoutTenant(parts...)
	}

	// 2. 构建Key: tenant:<tenant_id>:<prefix>:<parts...>
	keyParts := []string{
		"tenant",
		tenantCtx.TenantID,
		b.prefix,
	}
	keyParts = append(keyParts, parts...)

	return fmt.Sprintf("%s:%s:%s:%s", keyParts...)
}

// buildWithoutTenant 不包含租户的Key（兼容旧代码）
func (b *TenantCacheKey) buildWithoutTenant(parts ...string) string {
	keyParts := append([]string{b.prefix}, parts...)
	return fmt.Sprintf("%s:%s", keyParts...)
}

// 预定义Key生成器
var (
	BotKey       = &TenantCacheKey{prefix: "bot"}
	UserKey      = &TenantCacheKey{prefix: "user"}
	KnowledgeKey = &TenantCacheKey{prefix: "knowledge"}
	WorkflowKey  = &TenantCacheKey{prefix: "workflow"}
)

// BuildIndexDocKey 构建索引文档缓存Key
func BuildIndexDocKey(ctx context.Context, knowledgeID, documentID int64) string {
	return KnowledgeKey.Build(ctx,
		fmt.Sprintf("%d", knowledgeID),
		"document",
		fmt.Sprintf("%d", documentID),
	)
	// 结果: tenant:tenant_123:knowledge:456:document:789
}

// BuildBotKey 构建Bot缓存Key
func BuildBotKey(ctx context.Context, botID int64) string {
	return BotKey.Build(ctx, fmt.Sprintf("%d", botID))
	// 结果: tenant:tenant_123:bot:456
}

package cache

import (
	"context"
	"fmt"

	"github.com/coze-dev/coze-studio/backend/api/middleware"
	"github.com/redis/go-redis/v9"
)

// TenantCacheCleaner 租户缓存清理器
type TenantCacheCleaner struct {
	client *redis.Client
}

// NewTenantCacheCleaner 创建租户缓存清理器
func NewTenantCacheCleaner(client *redis.Client) *TenantCacheCleaner {
	return &TenantCacheCleaner{client: client}
}

// ClearTenantCache 清理租户所有缓存
func (c *TenantCacheCleaner) ClearTenantCache(ctx context.Context, tenantID string) error {
	// 使用SCAN命令查找租户所有缓存键
	pattern := fmt.Sprintf("tenant:%s:*", tenantID)

	var cursor uint64
	var count int

	for {
		keys, nextCursor, err := c.client.Scan(ctx, cursor, pattern, 100).Result()
		if err != nil {
			return err
		}

		if len(keys) > 0 {
			// 批量删除
			if err := c.client.Del(ctx, keys...).Err(); err != nil {
				return err
			}
			count += len(keys)
		}

		cursor = nextCursor
		if cursor == 0 {
			break
		}
	}

	return nil
}

// ClearUserCache 清理用户相关缓存
func (c *TenantCacheCleaner) ClearUserCache(ctx context.Context, tenantID string, userID int64) error {
	// 清理用户特定缓存
	patterns := []string{
		fmt.Sprintf("tenant:%s:user:%d", tenantID, userID),
		fmt.Sprintf("tenant:%s:bot:*:user:%d", tenantID, userID),
	}

	for _, pattern := range patterns {
		var cursor uint64
		for {
			keys, nextCursor, err := c.client.Scan(ctx, cursor, pattern, 100).Result()
			if err != nil {
				return err
			}

			if len(keys) > 0 {
				if err := c.client.Del(ctx, keys...).Err(); err != nil {
					return err
				}
			}

			cursor = nextCursor
			if cursor == 0 {
				break
			}
		}
	}

	return nil
}

package milvus

import (
	"context"
	"fmt"

	"github.com/coze-dev/coze-studio/backend/api/middleware"
	"github.com/milvus-io/milvus-sdk-go/v2/client"
	"github.com/milvus-io/milvus-sdk-go/v2/entity"
)

// TenantPartitionManager 租户分区管理器
type TenantPartitionManager struct {
	milvusClient client.Client
}

// CreateTenantPartition 创建租户分区
func (m *TenantPartitionManager) CreateTenantPartition(
	ctx context.Context,
	collectionName string,
) error {
	// 1. 获取租户ID
	tenantCtx, err := middleware.GetTenantContext(ctx)
	if err != nil {
		return err
	}

	// 2. 生成分区名称
	partitionName := m.buildPartitionName(tenantCtx.TenantID)

	// 3. 检查分区是否存在
	has, err := m.milvusClient.HasPartition(
		ctx,
		collectionName,
		partitionName,
	)
	if err != nil {
		return err
	}

	// 4. 创建分区（如果不存在）
	if !has {
		if err := m.milvusClient.CreatePartition(
			ctx,
			collectionName,
			partitionName,
		); err != nil {
			return fmt.Errorf("创建分区失败: %w", err)
		}
	}

	return nil
}

// buildPartitionName 构建分区名称
func (m *TenantPartitionManager) buildPartitionName(tenantID string) string {
	// 格式: tenant_<tenant_id>
	// 注意：Milvus分区名不能包含某些特殊字符，需要转义
	safeTenantID := escapeTenantID(tenantID)
	return fmt.Sprintf("tenant_%s", safeTenantID)
}

// escapeTenantID 转义租户ID（移除特殊字符）
func escapeTenantID(tenantID string) string {
	// 移除或替换特殊字符
	result := tenantID
	result = replaceAll(result, "-", "_")
	result = replaceAll(result, ".", "_")
	result = replaceAll(result, "@", "_")
	return result
}

// InsertToTenantPartition 插入向量到租户分区
func (m *TenantPartitionManager) InsertToTenantPartition(
	ctx context.Context,
	collectionName string,
	columns []entity.Column,
	vectorField string,
	vectors []entity.Vector,
) error {
	// 1. 获取租户ID
	tenantCtx, err := middleware.GetTenantContext(ctx)
	if err != nil {
		return err
	}

	// 2. 生成分区名称
	partitionName := m.buildPartitionName(tenantCtx.TenantID)

	// 3. 插入到指定分区
	_, err = m.milvusClient.Insert(
		ctx,
		collectionName,
		partitionName, // 指定分区
		columns,
	)
	if err != nil {
		return fmt.Errorf("插入向量失败: %w", err)
	}

	// 4. 刷新集合
	if err := m.milvusClient.Flush(ctx, collectionName, false); err != nil {
		return err
	}

	return nil
}

// SearchInTenantPartition 在租户分区内检索
func (m *TenantPartitionManager) SearchInTenantPartition(
	ctx context.Context,
	collectionName string,
	vector []float32,
	topK int,
	sp := entity.SearchParam{
		// 配置检索参数
	}
) ([]*SearchResult, error) {
	// 1. 获取租户ID
	tenantCtx, err := middleware.GetTenantContext(ctx)
	if err != nil {
		return nil, err
	}

	// 2. 生成分区名称
	partitionName := m.buildPartitionName(tenantCtx.TenantID)

	// 3. 创建搜索向量
	searchVector := entity.FloatVector(vector)

	// 4. 只在租户分区内搜索
	searchResult, err := m.milvusClient.Search(
		ctx,
		collectionName,
		[]string{partitionName}, // 只搜索此分区
		[]string{vectorField},
		[][]entity.Vector{searchVector},
		"score",
		topK,
		sp,
	)
	if err != nil {
		return nil, fmt.Errorf("向量检索失败: %w", err)
	}

	// 5. 解析结果
	results := make([]*SearchResult, 0)
	for _, res := range searchResult {
		for i := 0; i < res.ResultCount; i++ {
			id := res.IDs.GetInt64(i)
			score := res.Scores.GetFloat64(i)

			results = append(results, &SearchResult{
				ID:    id,
				Score: score,
			})
		}
	}

	return results, nil
}
