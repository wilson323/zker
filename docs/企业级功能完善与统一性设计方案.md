## 📋 第一部分：统一配置规范设计

### 1.1 当前配置存在的问题

#### 问题1: 配置分散
```go
// 超时配置分散在多处
http_requester.go:     Timeout time.Duration        // HTTP节点超时
sandbox/runner.go:     TimeoutSeconds float64       // 代码节点超时
node_runner.go:        timeoutMS int64              // 通用节点超时
```

#### 问题2: 缺少统一规范
- 没有全局默认值
- 配置命名不一致
- 配置单位不统一(秒/毫秒混用)

#### 问题3: 缺少环境区分
- 开发/测试/生产环境配置混在一起
- 缺少配置热更新机制

### 1.4 配置管理器实现

#### 配置管理器接口
```go
// backend/config/manager.go

package config

import (
    "context"
    "time"
)

// ConfigManager 配置管理器接口
type ConfigManager interface {
    // 全局配置
    GetTimeoutConfig() *TimeoutConfig
    GetRetryConfig() *RetryConfig
    GetRateLimitConfig() *RateLimitConfig
    GetMonitorConfig() *MonitorConfig

    // 模块配置
    GetWorkflowConfig() *WorkflowConfig
    GetKnowledgeConfig() *KnowledgeConfig
    GetPluginConfig() *PluginConfig
    GetAPIConfig() *APIConfig

    // 节点配置
    GetNodeConfig(nodeType string) (NodeConfig, error)

    // 运行时配置
    Get(key string, defaultValue interface{}) interface{}
    Set(key string, value interface{}) error

    // 配置热更新
    Watch(ctx context.Context, key string) (<-chan ConfigChangeEvent, error)

    // 配置验证
    Validate() error
}

// TimeoutConfig 超时配置
type TimeoutConfig struct {
    Workflow       time.Duration `yaml:"workflow" json:"workflow"`
    HTTPNode       time.Duration `yaml:"http_node" json:"http_node"`
    LLMNode        time.Duration `yaml:"llm_node" json:"llm_node"`
    CodeNode       time.Duration `yaml:"code_node" json:"code_node"`
    PluginNode     time.Duration `yaml:"plugin_node" json:"plugin_node"`
    DatabaseQuery  time.Duration `yaml:"database_query" json:"database_query"`
}

// RetryConfig 重试配置
type RetryConfig struct {
    MaxAttempts    int      `yaml:"max_attempts" json:"max_attempts"`
    BackoffPolicy  string   `yaml:"backoff_policy" json:"backoff_policy"` // fixed/linear/exponential
    InitialInterval time.Duration `yaml:"initial_interval" json:"initial_interval"`
    MaxInterval    time.Duration `yaml:"max_interval" json:"max_interval"`
    Multiplier     float64  `yaml:"multiplier" json:"multiplier"`
}

// RateLimitConfig 限流配置
type RateLimitConfig struct {
    Enabled     bool    `yaml:"enabled" json:"enabled"`
    Algorithm   string  `yaml:"algorithm" json:"algorithm"`
    GlobalQPS   int     `yaml:"global_qps" json:"global_qps"`
    PerUserQPS  int     `yaml:"per_user_qps" json:"per_user_qps"`
    PerWorkflowQPS int `yaml:"per_workflow_qps" json:"per_workflow_qps"`
}

// MonitorConfig 监控配置
type MonitorConfig struct {
    Enabled     bool              `yaml:"enabled" json:"enabled"`
    Prometheus  PrometheusConfig  `yaml:"prometheus" json:"prometheus"`
    Tracing     TracingConfig     `yaml:"tracing" json:"tracing"`
}

// WorkflowConfig 工作流配置
type WorkflowConfig struct {
    Engine EngineConfig `yaml:"engine" json:"engine"`
    Nodes  map[string]NodeConfig `yaml:"nodes" json:"nodes"`
}

// EngineConfig 引擎配置
type EngineConfig struct {
    MaxParallelNodes    int  `yaml:"max_parallel_nodes" json:"max_parallel_nodes"`
    MaxNodeExecutions   int  `yaml:"max_node_executions" json:"max_node_executions"`
    CheckpointEnabled   bool `yaml:"checkpoint_enabled" json:"checkpoint_enabled"`
    CheckpointStorage   string `yaml:"checkpoint_storage" json:"checkpoint_storage"`
}

// NodeConfig 节点配置接口
type NodeConfig interface {
    GetTimeout() time.Duration
    GetRetryConfig() *RetryConfig
}

// HTTPNodeConfig HTTP节点配置
type HTTPNodeConfig struct {
    Timeout         time.Duration `yaml:"timeout" json:"timeout"`
    Retry           *RetryConfig  `yaml:"retry" json:"retry"`
    RetryableErrors []int         `yaml:"retryable_errors" json:"retryable_errors"`
}

// ConfigChangeEvent 配置变更事件
type ConfigChangeEvent struct {
    Key     string
    OldValue interface{}
    NewValue interface{}
    Timestamp time.Time
}
```

#### 配置管理器实现
```go
// backend/config/manager_impl.go

package config

import (
    "context"
    "fmt"
    "sync"
    "time"

    "github.com/coze-dev/zker/backend/pkg/logs"
    "gopkg.in/yaml.v3"
)

type configManager struct {
    timeout    *TimeoutConfig
    retry      *RetryConfig
    rateLimit  *RateLimitConfig
    monitor    *MonitorConfig
    workflow   *WorkflowConfig
    knowledge  *KnowledgeConfig
    plugin     *PluginConfig
    api        *APIConfig

    // 运行时配置覆盖
    overrides map[string]interface{}
    mu        sync.RWMutex

    // 配置变更监听器
    watchers map[string][]chan ConfigChangeEvent
    logger   logs.Logger
}

// NewConfigManager 创建配置管理器
func NewConfigManager(configPath string, logger logs.Logger) (ConfigManager, error) {
    mgr := &configManager{
        overrides: make(map[string]interface{}),
        watchers:  make(map[string][]chan ConfigChangeEvent),
        logger:    logger,
    }

    // 加载默认配置
    if err := mgr.loadConfig(configPath); err != nil {
        return nil, fmt.Errorf("load config failed: %w", err)
    }

    // 验证配置
    if err := mgr.Validate(); err != nil {
        return nil, fmt.Errorf("config validation failed: %w", err)
    }

    return mgr, nil
}

// loadConfig 加载配置文件
func (m *configManager) loadConfig(configPath string) error {
    // 读取YAML文件
    data, err := os.ReadFile(configPath)
    if err != nil {
        return err
    }

    // 解析配置
    var config struct {
        Global    map[string]interface{} `yaml:"global"`
        Workflow  *WorkflowConfig         `yaml:"workflow"`
        Knowledge *KnowledgeConfig        `yaml:"knowledge"`
        Plugin    *PluginConfig           `yaml:"plugin"`
        API       *APIConfig              `yaml:"api"`
    }

    if err := yaml.Unmarshal(data, &config); err != nil {
        return err
    }

    // 解析全局配置
    m.parseGlobalConfig(config.Global)

    // 解析模块配置
    m.workflow = config.Workflow
    m.knowledge = config.Knowledge
    m.plugin = config.Plugin
    m.api = config.API

    return nil
}

// parseGlobalConfig 解析全局配置
func (m *configManager) parseGlobalConfig(global map[string]interface{}) {
    // 解析超时配置
    if timeout, ok := global["timeout"].(map[string]interface{}); ok {
        m.timeout = &TimeoutConfig{
            Workflow:      parseDuration(timeout["workflow"], 120*time.Second),
            HTTPNode:      parseDuration(timeout["http_node"], 600*time.Second),
            LLMNode:       parseDuration(timeout["llm_node"], 300*time.Second),
            CodeNode:      parseDuration(timeout["code_node"], 60*time.Second),
            PluginNode:    parseDuration(timeout["plugin_node"], 600*time.Second),
            DatabaseQuery: parseDuration(timeout["database_query"], 30*time.Second),
        }
    }

    // 解析重试配置
    if retry, ok := global["retry"].(map[string]interface{}); ok {
        m.retry = &RetryConfig{
            MaxAttempts:     parseInt(retry["max_attempts"], 3),
            BackoffPolicy:   parseString(retry["backoff_policy"], "exponential"),
            InitialInterval: parseDuration(retry["initial_interval"], 1*time.Second),
            MaxInterval:     parseDuration(retry["max_interval"], 60*time.Second),
            Multiplier:      parseFloat(retry["multiplier"], 2.0),
        }
    }

    // ... 解析其他配置
}

// GetTimeoutConfig 获取超时配置
func (m *configManager) GetTimeoutConfig() *TimeoutConfig {
    m.mu.RLock()
    defer m.mu.RUnlock()
    return m.timeout
}

// GetRetryConfig 获取重试配置
func (m *configManager) GetRetryConfig() *RetryConfig {
    m.mu.RLock()
    defer m.mu.RUnlock()
    return m.retry
}

// GetNodeConfig 获取节点配置
func (m *configManager) GetNodeConfig(nodeType string) (NodeConfig, error) {
    m.mu.RLock()
    defer m.mu.RUnlock()

    if m.workflow == nil || m.workflow.Nodes == nil {
        return nil, fmt.Errorf("workflow config not found")
    }

    config, ok := m.workflow.Nodes[nodeType]
    if !ok {
        return nil, fmt.Errorf("node type %s config not found", nodeType)
    }

    return config, nil
}

// Watch 监听配置变更
func (m *configManager) Watch(ctx context.Context, key string) (<-chan ConfigChangeEvent, error) {
    ch := make(chan ConfigChangeEvent, 10)

    m.mu.Lock()
    m.watchers[key] = append(m.watchers[key], ch)
    m.mu.Unlock()

    go func() {
        <-ctx.Done()
        m.mu.Lock()
        // 移除监听器
        // ...
        m.mu.Unlock()
        close(ch)
    }()

    return ch, nil
}

// Validate 验证配置
func (m *configManager) Validate() error {
    // 验证超时配置
    if m.timeout != nil {
        if m.timeout.Workflow < 10*time.Second {
            return fmt.Errorf("workflow timeout too short: %v", m.timeout.Workflow)
        }
        if m.timeout.Workflow > 10*time.Minute {
            return fmt.Errorf("workflow timeout too long: %v", m.timeout.Workflow)
        }
    }

    // 验证重试配置
    if m.retry != nil {
        if m.retry.MaxAttempts < 0 || m.retry.MaxAttempts > 10 {
            return fmt.Errorf("invalid max_attempts: %d", m.retry.MaxAttempts)
        }
        if m.retry.Multiplier < 1.0 || m.retry.Multiplier > 10.0 {
            return fmt.Errorf("invalid multiplier: %f", m.retry.Multiplier)
        }
    }

    // ... 验证其他配置

    return nil
}
```

```go
// backend/domain/workflow/internal/nodes/httprequester/http_requester.go

package httprequester

import (
    "context"
    "github.com/coze-dev/zker/backend/config"
)

func (c *Config) Build(ctx context.Context, ns *schema.NodeSchema, opts ...schema.BuildOption) (any, error) {
    // 从配置管理器获取节点配置
    configMgr := config.FromContext(ctx)
    if configMgr == nil {
        return nil, fmt.Errorf("config manager not found in context")
    }

    nodeConfig, err := configMgr.GetNodeConfig("http")
    if err != nil {
        return nil, err
    }

    httpConfig, ok := nodeConfig.(*config.HTTPNodeConfig)
    if !ok {
        return nil, fmt.Errorf("invalid http node config type")
    }

    // 使用配置创建HTTP请求器
    hg := &HTTPRequester{
        urlConfig:       c.URLConfig,
        method:          c.Method,
        retryTimes:      httpConfig.Retry.MaxAttempts, // 从配置读取
        authConfig:      c.AuthConfig,
        bodyConfig:      c.BodyConfig,
        md5FieldMapping: c.MD5FieldMapping,
    }

    // 创建HTTP客户端，设置超时
    client := &http.Client{
        Timeout: httpConfig.Timeout, // 从配置读取
    }

    hg.client = client
    return hg, nil
}
```

### 2.2 详细功能设计

#### 2.2.1 审计日志系统

##### 功能需求
1. 记录所有用户操作
2. 记录所有敏感数据访问
3. 支持审计日志查询和导出
4. 满足合规要求(GDPR、SOC2、等保三级)

##### 数据库设计
```sql
-- 审计日志表
CREATE TABLE audit_log (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    -- 操作者信息
    user_id BIGINT NOT NULL COMMENT '用户ID',
    user_name VARCHAR(100) COMMENT '用户名',
    user_type VARCHAR(20) COMMENT '用户类型: user/admin/system',

    -- 操作信息
    action VARCHAR(50) NOT NULL COMMENT '操作类型',
    action_detail VARCHAR(200) COMMENT '操作详情',
```

```go
// TokenBucketLimiter 令牌桶限流器
type TokenBucketLimiter struct {
    redis      *redis.Client
    keyPrefix  string
    capacity   int           // 桶容量
    refillRate int           // 填充速率(tokens/秒)
    interval   time.Duration // 填充间隔
}

// NewTokenBucketLimiter 创建令牌桶限流器
func NewTokenBucketLimiter(redis *redis.Client, capacity, refillRate int) *TokenBucketLimiter {
    return &TokenBucketLimiter{
        redis:      redis,
        keyPrefix:  "token_bucket:",
        capacity:   capacity,
        refillRate: refillRate,
        interval:   time.Second,
    }
}

// Allow 检查是否允许请求
func (t *TokenBucketLimiter) Allow(ctx context.Context, key string, limit int, window time.Duration) (bool, error) {
    // 使用Lua脚本保证原子性
    script := `
        local key = KEYS[1]
        local now = tonumber(ARGV[1])
        local requested = tonumber(ARGV[2])
        local capacity = tonumber(ARGV[3])
        local interval = tonumber(ARGV[4])

        -- 获取当前令牌数和上次填充时间
        local tokens = redis.call('HGET', key, 'tokens')
        local last_refill = redis.call('HGET', key, 'last_refill')

        if tokens == false then
            tokens = capacity
            last_refill = now
        else
            tokens = tonumber(tokens)
            last_refill = tonumber(last_refill)
        end

        -- 计算应该填充的令牌数
        local elapsed = now - last_refill
        local refill = math.floor(elapsed / interval * ` + fmt.Sprint(t.refillRate) + `)

        -- 更新令牌数(不超过容量)
        tokens = math.min(capacity, tokens + refill)

        -- 检查是否有足够令牌
        if tokens >= requested then
            tokens = tokens - requested
            redis.call('HMSET', key, 'tokens', tokens, 'last_refill', now)
            redis.call('EXPIRE', key, ` + fmt.Sprint(window.Seconds()) + `)
            return 1
        else
            return 0
        end
    `

    result, err := t.redis.Eval(ctx, script, []string{t.keyPrefix + key},
        time.Now().Unix(), 1, t.capacity, t.interval.Seconds())

    if err != nil {
        return false, err
    }

    return result.(int64) == 1, nil
}

// SlidingWindowLimiter 滑动窗口限流器
type SlidingWindowLimiter struct {
    redis *redis.Client
}

// NewSlidingWindowLimiter 创建滑动窗口限流器
func NewSlidingWindowLimiter(redis *redis.Client) *SlidingWindowLimiter {
    return &SlidingWindowLimiter{redis: redis}
}

// Allow 检查是否允许请求
func (s *SlidingWindowLimiter) Allow(ctx context.Context, key string, limit int, window time.Duration) (bool, error) {
    now := time.Now().UnixMicro() // 微秒时间戳
    windowStart := now - window.Microseconds()

    pipe := s.redis.Pipeline()

    // 移除窗口外的记录
    pipe.ZRemRangeByScore(ctx, key, "0", fmt.Sprint(windowStart))

    // 统计当前窗口内的请求数
    countCmd := pipe.ZCard(ctx, key)

    // 添加当前请求
    pipe.ZAdd(ctx, key, redis.Z{Score: float64(now), Member: now})

    // 设置过期时间
    pipe.Expire(ctx, key, window)

    _, err := pipe.Exec(ctx)
    if err != nil {
        return false, err
    }

    count := countCmd.Val()
    return count < int64(limit), nil
}
```

```go
    PageSize int        `json:"page_size"` // 每页大小
    Total   int64       `json:"total"`      // 总记录数
    RequestID string    `json:"request_id,omitempty"`
    Timestamp int64     `json:"timestamp"`
}

// Success 成功响应
func Success(data interface{}) *Response {
    return &Response{
        Code:      0,
        Message:   "success",
        Data:      data,
        Timestamp: time.Now().Unix(),
    }
}

// Error 错误响应
func Error(err error) *Response {
    var code int32 = 10000
    var msg string = "internal server error"

    if e, ok := err.(errno.ErrorCode); ok {
        code = e.Code()
        msg = e.Msg()
    } else {
        msg = err.Error()
    }

    return &Response{
        Code:      code,
        Message:   msg,
        Timestamp: time.Now().Unix(),
    }
}

// ErrorWithDetails 错误响应(带详情)
func ErrorWithDetails(err error, details map[string]interface{}) *Response {
    resp := Error(err)
    resp.Data = details
    return resp
}

// PageSuccess 分页成功响应
func PageSuccess(data interface{}, page, pageSize int, total int64) *PageResponse {
    return &PageResponse{
        Code:     0,
        Message:  "success",
        Data:     data,
        Page:     page,
        PageSize: pageSize,
        Total:    total,
        Timestamp: time.Now().Unix(),
    }
}
```

```go
// backend/pkg/logs/structured_logger.go

package logs

import (
    "context"
    "encoding/json"
    "time"
)

// StructuredLogger 结构化日志
type StructuredLogger struct {
    level  Level
    output io.Writer
    fields map[string]interface{}
}

// LogEntry 日志条目
type LogEntry struct {
    Time      time.Time              `json:"time"`
    Level     string                 `json:"level"`
    Message   string                 `json:"message"`
    Fields    map[string]interface{} `json:"fields,omitempty"`
    RequestID string                 `json:"request_id,omitempty"`
    TraceID   string                 `json:"trace_id,omitempty"`
    UserID    int64                  `json:"user_id,omitempty"`
    Duration  int64                  `json:"duration_ms,omitempty"` // 耗时(毫秒)
}

// WithFields 添加字段
func (l *StructuredLogger) WithFields(fields map[string]interface{}) *StructuredLogger {
    newFields := make(map[string]interface{})
    for k, v := range l.fields {
        newFields[k] = v
    }
    for k, v := range fields {
        newFields[k] = v
    }
    return &StructuredLogger{
        level:  l.level,
        output: l.output,
        fields: newFields,
    }
}

// WithError 添加错误
func (l *StructuredLogger) WithError(err error) *StructuredLogger {
    return l.WithFields(map[string]interface{}{
        "error": err.Error(),
        "error_type": fmt.Sprintf("%T", err),
    })
}

// WithRequest 添加请求信息
func (l *StructuredLogger) WithRequest(r *http.Request) *StructuredLogger {
    return l.WithFields(map[string]interface{}{
        "request_id": GetRequestID(r.Context()),
        "method":     r.Method,
        "path":       r.URL.Path,
        "ip":         GetClientIP(r),
    })
}

// Infof Info级别日志
func (l *StructuredLogger) Infof(format string, args ...interface{}) {
    l.log(LevelInfo, format, args...)
}

// Errorf Error级别日志
func (l *StructuredLogger) Errorf(format string, args ...interface{}) {
    l.log(LevelError, format, args...)
}

// log 内部日志方法
func (l *StructuredLogger) log(level Level, format string, args ...interface{}) {
    if level < l.level {
        return
    }

    entry := &LogEntry{
        Time:    time.Now(),
        Level:   level.toString(),
        Message: fmt.Sprintf(format, args...),
        Fields:  l.fields,
    }

    // 从context获取信息
    if requestID := GetRequestID(context.Background()); requestID != "" {
        entry.RequestID = requestID
    }

    // JSON序列化
    data, _ := json.Marshal(entry)
    l.output.Write(data)
    l.output.Write([]byte("\n"))
}
```
