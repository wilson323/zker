## 1. 规范概述

### 1.1 规范目标

本规范旨在确保ZKER项目错误处理：
- ✅ **统一性**: 所有错误使用统一的错误码和格式
- ✅ **可追溯**: 每个错误都有唯一的标识和日志
- ✅ **易排查**: 错误信息清晰，包含足够的上下文
- ✅ **国际化**: 支持中英文错误提示

### 1.2 核心原则

1. **错误分层** - 区分系统错误、业务错误、用户错误
2. **错误码化** - 每个错误都有唯一的错误码
3. **上下文完整** - 错误信息包含请求ID、时间戳、堆栈信息
4. **安全优先** - 不暴露敏感信息（如SQL语句、密码）

### 1.3 错误分类

```go
// 错误级别

const (
    ErrorLevelCritical ErrorLevel = "critical"  // 系统级错误，需要立即处理

    ErrCategoryThirdParty  ErrorCategory = "third_party"
```

## 2. 错误码设计

### 2.1 错误码格式

**格式**: `{模块}{类型}{序号}`

- **模块** (2位数字):
  - `10` = 用户模块
  - `11` = Bot模块
  - `12` = 对话模块
  - `13` = 知识库模块
  - `14` = 插件模块
  - `20` = 租户模块
  - `21` = 组织模块
  - `22` = 权限模块
  - `30` = 工作流模块
  - `99` = 通用模块

- **类型** (1位数字):
  - `1` = 参数错误
  - `2` = 权限错误
  - `3` = 业务逻辑错误
  - `4` = 系统错误
  - `5` = 第三方服务错误

- **序号** (3位数字): 001-999

**示例**: `101001` = 用户模块 + 参数错误 + 第001个错误

### 3.2 错误处理中间件

```go
package middleware

import (
    "context"
    "net/http"
    "runtime/debug"

    "zker/pkg/errors"
    "zker/pkg/response"
)

// 错误处理中间件
func ErrorHandlerMiddleware() app.HandlerFunc {
    return func(ctx context.Context, c *app.RequestContext) {
        // 恢复panic
        defer func() {
            if r := recover(); r != nil {
                // 记录panic堆栈
                logger.Errorf("panic recovered: %v\n%s", r, debug.Stack())

                // 返回500错误
                c.SetStatusCode(http.StatusInternalServerError)
                c.JSON(errors.ErrInternal.WithRequestID(GetRequestID(c)))
            }
        }()

        c.Next(ctx)

        // 检查是否有错误
        if err := c.Errors.Last(); err != nil {
            handleError(c, err.Err)
        }
    }
}

func handleError(c *app.RequestContext, err error) {
    requestID := GetRequestID(c)

    // 类型断言，判断是否为BusinessError
    bizErr, ok := err.(*errors.BusinessError)
    if !ok {
        // 不是业务错误，包装为内部错误
        bizErr = errors.ErrInternal.
            WithRequestID(requestID).
            WithCause(err).
            WithStackTrace()
    } else {
        // 是业务错误，附加request_id
        bizErr.WithRequestID(requestID)
    }

    // 记录错误日志
    logError(bizErr)

    // 设置HTTP状态码
    httpStatus := getHTTPStatus(bizErr.Code)
    c.SetStatusCode(httpStatus)

    // 返回错误响应
    c.JSON(response.Error{
        Code:      bizErr.Code,
        Message:   bizErr.Message,
        ErrorCode: bizErr.ErrorCode,
        Detail:    bizErr.Detail,
        RequestID: bizErr.RequestID,
        Timestamp: bizErr.Timestamp,
    })
}

func getHTTPStatus(errorCode int) int {
    switch errorCode / 10000 {
    case 10: // 参数错误
        return http.StatusBadRequest
    case 11: // 权限错误
        return http.StatusForbidden
    case 12: // 业务错误
        if errorCode == 993001 {
            return http.StatusNotFound
        }
        if errorCode == 993002 || errorCode == 993003 {
            return http.StatusConflict
        }
        return http.StatusBadRequest
    case 13: // 系统错误
        return http.StatusInternalServerError
    case 14: // 第三方错误
        return http.StatusBadGateway
    default:
        return http.StatusInternalServerError
    }
}

func logError(err *errors.BusinessError) {
    switch err.Level {
    case errors.ErrorLevelCritical:
        logger.Errorf("[CRITICAL] %s: %s\n堆栈: %s", err.ErrorCode, err.Message, err.StackTrace)
    case errors.ErrorLevelError:
        logger.Errorf("[ERROR] %s: %s", err.ErrorCode, err.Message)
    case errors.ErrorLevelWarning:
        logger.Warnf("[WARNING] %s: %s", err.ErrorCode, err.Message)
    case errors.ErrorLevelInfo:
        logger.Infof("[INFO] %s: %s", err.ErrorCode, err.Message)
    }
}
```

```go
        return nil, errors.ErrDatabase.WithCause(err)
```

```typescript
  RESOURCE_NOT_FOUND = 'RESOURCE_NOT_FOUND',
  RESOURCE_CONFLICT = 'RESOURCE_CONFLICT',
  QUOTA_EXCEEDED = 'QUOTA_EXCEEDED',

  // 系统错误
  INTERNAL_ERROR = 'INTERNAL_ERROR',
  DATABASE_ERROR = 'DATABASE_ERROR',

  // 第三方错误
  LLM_API_ERROR = 'LLM_API_ERROR',
}

/**
 * 业务错误类
 */
export class BusinessException extends Error {
  public readonly code: number;
  public readonly errorCode: string;
  public readonly detail?: string;
  public readonly requestId?: string;

  constructor(
    code: number,
    errorCode: string,
    message: string,
    detail?: string,
    requestId?: string
  ) {
    super(message);
    this.name = 'BusinessException';
    this.code = code;
    this.errorCode = errorCode;
    this.detail = detail;
    this.requestId = requestId;
  }

  static is(error: any): error is BusinessException {
    return error instanceof BusinessException;
  }
}

/**
 * 创建业务错误
 */
export function createError(
  errorCode: ErrorCode,
  message?: string,
  detail?: string
): BusinessException {
  const errorMap: Record<ErrorCode, { code: number; defaultMessage: string }> = {
    [ErrorCode.INVALID_PARAM]: { code: 991001, defaultMessage: '参数无效' },
    [ErrorCode.MISSING_PARAM]: { code: 991002, defaultMessage: '缺少必需参数' },
    [ErrorCode.UNAUTHORIZED]: { code: 992001, defaultMessage: '未授权' },
    [ErrorCode.TOKEN_EXPIRED]: { code: 992002, defaultMessage: 'Token已过期' },
    [ErrorCode.FORBIDDEN]: { code: 992003, defaultMessage: '禁止访问' },
    [ErrorCode.PERMISSION_DENIED]: { code: 992004, defaultMessage: '权限不足' },
    [ErrorCode.RESOURCE_NOT_FOUND]: { code: 993001, defaultMessage: '资源不存在' },
    [ErrorCode.QUOTA_EXCEEDED]: { code: 993004, defaultMessage: '配额已用尽' },
    [ErrorCode.INTERNAL_ERROR]: { code: 994001, defaultMessage: '内部服务器错误' },
    [ErrorCode.LLM_API_ERROR]: { code: 995002, defaultMessage: 'LLM服务错误' },
  };

  const errorInfo = errorMap[errorCode];
  return new BusinessException(
    errorInfo.code,
    errorCode,
    message || errorInfo.defaultMessage,
    detail
  );
}
```

### 4.2 错误处理Hook

```typescript
// src/hooks/useErrorHandler.ts

import { notification } from '@demi-design/semi-ui';
import { useNavigate } from 'react-router-dom';
import { createError, ErrorCode, BusinessException } from '@/common/errors';

/**
 * 错误处理Hook
 */
export function useErrorHandler() {
  const navigate = useNavigate();

  /**
   * 处理API错误
   */
  const handleError = (error: any) => {
    // 业务错误
    if (error instanceof BusinessException) {
      handleBusinessError(error);
      return;
    }

    // API响应错误
    if (error.response?.data) {
      const apiError = error.response.data;
      if (apiError.error_code) {
        handleBusinessError({
          code: apiError.code,
          errorCode: apiError.error_code,
          message: apiError.message,
          detail: apiError.detail,
          requestId: apiError.request_id,
        } as BusinessException);
        return;
      }
    }

    // 网络错误
    if (error.code === 'ERR_NETWORK') {
      notification.error({
        title: '网络错误',
        content: '请检查网络连接后重试',
      });
      return;
    }

    // 未知错误
    notification.error({
      title: '未知错误',
      content: error.message || '发生了未知错误，请稍后重试',
    });
  };

  /**
   * 处理业务错误
   */
  const handleBusinessError = (error: BusinessException) => {
    const { errorCode, message, detail } = error;

    // 根据错误码处理
    switch (errorCode) {
      case ErrorCode.UNAUTHORIZED:
      case ErrorCode.TOKEN_EXPIRED:
        // Token无效或过期，跳转到登录页
        notification.warning({
          title: '登录已过期',
          content: '请重新登录',
        });
        navigate('/login', { replace: true });
        break;

      case ErrorCode.FORBIDDEN:
      case ErrorCode.PERMISSION_DENIED:
        // 权限不足
        notification.error({
          title: '权限不足',
          content: detail || message,
        });
        break;

      caseErrorCode.RESOURCE_NOT_FOUND:
        // 资源不存在
        notification.warning({
          title: '资源不存在',
          content: detail || message,
        });
        break;

      case ErrorCode.QUOTA_EXCEEDED:
        // 配额超限
        notification.error({
          title: '配额已用尽',
          content: detail || '请联系管理员升级配额',
        });
        break;

      case ErrorCode.LLM_API_ERROR:
        // LLM服务错误
        notification.error({
          title: 'AI服务异常',
          content: detail || 'AI服务暂时不可用，请稍后重试',
        });
        break;

      default:
        // 其他错误
        notification.error({
          title: '操作失败',
          content: detail || message,
        });
    }

    // 记录错误到日志（生产环境）
    if (process.env.NODE_ENV === 'production') {
      logError(error);
    }
  };

  /**
   * 记录错误日志
   */
  const logError = (error: BusinessException) => {
    // 发送到错误监控服务（如Sentry）
    // Sentry.captureException(error);
  };

  return { handleError };
}

/**
 * 异步操作包装器
 */
export function useAsyncError() {
  const { handleError } = useErrorHandler();

  const withErrorHandling = async <T>(
    asyncFn: () => Promise<T>
  ): Promise<T | null> => {
    try {
      return await asyncFn();
    } catch (error) {
      handleError(error);
      return null;
    }
  };

  return { withErrorHandling };
}
```
