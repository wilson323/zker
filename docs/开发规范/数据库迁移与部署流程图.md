## 一、数据库迁移脚本

### 1.1 团队管理表结构（MySQL 8.4+）

```sql
-- =====================================================
-- 团队管理模块数据库迁移脚本
-- 版本: v1.0.0
-- 日期: 2025-12-30
-- 说明: 创建团队管理相关表（teams、team_members、team_quotas）
-- =====================================================

-- =====================================================
-- 1. 创建teams表（团队主表）
-- =====================================================
CREATE TABLE IF NOT EXISTS `teams` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '团队ID',
    `name` VARCHAR(255) NOT NULL COMMENT '团队名称',
    `description` TEXT COMMENT '团队描述',
    `owner_id` BIGINT NOT NULL COMMENT '所有者用户ID',
    `avatar_url` VARCHAR(512) COMMENT '团队头像URL',
    `member_count` INT NOT NULL DEFAULT 1 COMMENT '成员数量',
    `resource_count` INT NOT NULL DEFAULT 0 COMMENT '资源数量（Bot数量）',
    `quota_config` JSON COMMENT '配额配置（JSON格式）',
    `status` VARCHAR(20) NOT NULL DEFAULT 'active' COMMENT '团队状态: active/inactive/deleted',
    `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    `deleted_at` TIMESTAMP NULL COMMENT '删除时间（软删除）',
    PRIMARY KEY (`id`),
    INDEX `idx_owner` (`owner_id`),
    INDEX `idx_deleted` (`deleted_at`),
    INDEX `idx_status` (`status`),
    INDEX `idx_created_at` (`created_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='团队表';

-- =====================================================
-- 2. 创建team_members表（团队成员表）
-- =====================================================
CREATE TABLE IF NOT EXISTS `team_members` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '成员记录ID',
    `team_id` BIGINT NOT NULL COMMENT '团队ID',
    `user_id` BIGINT NOT NULL COMMENT '用户ID',
    `role` VARCHAR(20) NOT NULL DEFAULT 'member' COMMENT '角色: owner/admin/member',
    `permissions` JSON COMMENT '权限配置（JSON格式）',
    `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '加入时间',
    `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_team_user` (`team_id`, `user_id`),
    INDEX `idx_team` (`team_id`),
    INDEX `idx_user` (`user_id`),
    INDEX `idx_role` (`role`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='团队成员表';

-- =====================================================
-- 3. 创建team_quotas表（团队配额表）
-- =====================================================
CREATE TABLE IF NOT EXISTS `team_quotas` (
    `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '配额ID',
    `team_id` BIGINT NOT NULL COMMENT '团队ID',
    `max_members` INT NOT NULL DEFAULT 10 COMMENT '最大成员数',
    `max_bots` INT NOT NULL DEFAULT 5 COMMENT '最大Bot数量',
    `max_knowledge_base` INT NOT NULL DEFAULT 3 COMMENT '最大知识库数量',
    `storage_quota_gb` INT NOT NULL DEFAULT 10 COMMENT '存储配额（GB）',
    `monthly_api_call` INT NOT NULL DEFAULT 10000 COMMENT '月度API调用次数',
    `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_team` (`team_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='团队配额表';

-- =====================================================
-- 4. 外键约束（可选，根据需要启用）
-- =====================================================
-- ALTER TABLE `team_members`
--     ADD CONSTRAINT `fk_team_members_teams`
--     FOREIGN KEY (`team_id`) REFERENCES `teams` (`id`) ON DELETE CASCADE;

-- ALTER TABLE `team_quotas`
--     ADD CONSTRAINT `fk_team_quotas_teams`
--     FOREIGN KEY (`team_id`) REFERENCES `teams` (`id`) ON DELETE CASCADE;

-- =====================================================
-- 5. 初始化数据（示例）
-- =====================================================
-- 插入默认团队配额模板（可选）
INSERT INTO `team_quotas` (`team_id`, `max_members`, `max_bots`, `max_knowledge_base`, `storage_quota_gb`, `monthly_api_call`)
VALUES
    (0, 10, 5, 3, 10, 10000)  -- 默认配额模板
ON DUPLICATE KEY UPDATE `max_members` = 10;

-- =====================================================
-- 6. 迁移完成标记
-- =====================================================
-- 记录迁移版本
INSERT INTO `schema_migrations` (`version`, `applied_at`)
VALUES ('20251230_create_teams', NOW())
ON DUPLICATE KEY UPDATE `applied_at` = NOW();

-- =====================================================
-- 7. 验证脚本
-- =====================================================
-- 检查表是否创建成功
SELECT
    TABLE_NAME,
    TABLE_ROWS,
    CREATE_TIME
FROM information_schema.TABLES
WHERE TABLE_SCHEMA = DATABASE()
    AND TABLE_NAME IN ('teams', 'team_members', 'team_quotas');

-- 检查索引是否创建成功
SELECT
    TABLE_NAME,
    INDEX_NAME,
    COLUMN_NAME,
    NON_UNIQUE
FROM information_schema.STATISTICS
WHERE TABLE_SCHEMA = DATABASE()
    AND TABLE_NAME IN ('teams', 'team_members', 'team_quotas')
ORDER BY TABLE_NAME, INDEX_NAME, SEQ_IN_INDEX;
```

### 1.3 性能优化建议

```sql
-- =====================================================
-- 性能优化脚本
-- =====================================================

-- 1. 为大型团队添加分区（可选）
-- ALTER TABLE `team_members`

    `team_id` BIGINT NOT NULL COMMENT '团队ID',
    `stat_date` DATE NOT NULL COMMENT '统计日期',
    `member_count` INT NOT NULL DEFAULT 0 COMMENT '成员数量',
    `bot_count` INT NOT NULL DEFAULT 0 COMMENT 'Bot数量',
    `api_call_count` INT NOT NULL DEFAULT 0 COMMENT 'API调用次数',
    `storage_used_gb` DECIMAL(10, 2) NOT NULL DEFAULT 0 COMMENT '已用存储（GB）',
    `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (`team_id`, `stat_date`),
    INDEX `idx_stat_date` (`stat_date`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='团队统计表';

-- 4. 创建存储过程（可选）
DELIMITER $$

CREATE PROCEDURE `update_team_member_count`(IN p_team_id BIGINT)
BEGIN
    UPDATE `teams`
    SET `member_count` = (
        SELECT COUNT(*)
        FROM `team_members`
        WHERE `team_id` = p_team_id
    )
    WHERE `id` = p_team_id;
END$$

DELIMITER ;

-- 5. 创建触发器（可选）
DELIMITER $$

CREATE TRIGGER `trg_team_members_after_insert`
AFTER INSERT ON `team_members`
FOR EACH ROW
BEGIN
    CALL `update_team_member_count`(NEW.`team_id`);
END$$

CREATE TRIGGER `trg_team_members_after_delete`
AFTER DELETE ON `team_members`
FOR EACH ROW
BEGIN
    CALL `update_team_member_count`(OLD.`team_id`);
END$$

DELIMITER ;
```

## 二、数据库迁移流程UML

### 2.1 数据库迁移完整流程

```mermaid
sequenceDiagram
    participant Dev as 开发者
    participant Git as Git仓库
    participant CI as CI服务器
    participant DBA as DBA
    participant StagingDB as 测试数据库
    participant ProdDB as 生产数据库
    participant Backup as 备份系统
    participant Monitor as 监控系统

    Note over Dev,Monitor: 准备阶段

    Dev->>Dev: 编写数据库迁移脚本
    Dev->>Dev: 本地测试迁移脚本
    Dev->>Git: 提交迁移脚本（PR）
    Note over Git: 路径: migrations/20251230_create_teams.sql

    Git->>CI: 触发CI流程
    CI->>CI: SQL语法检查
    CI->>CI: 安全规则检查
    CI->>CI: 性能影响分析

    alt 检查失败
        CI-->>Dev: 返回检查失败
        Dev->>Dev: 修复脚本
        Dev->>Git: 重新提交
    end

    Note over Dev,Monitor: 代码审查阶段

    Git->>DBA: 请求DBA审查
    DBA->>Git: 审查迁移脚本
    Note over DBA: 审查要点:<br/>- SQL语法<br/>- 索引设计<br/>- 性能影响<br/>- 回滚方案

    alt 审查不通过
        DBA-->>Dev: 返回审查意见
        Dev->>Dev: 根据意见修改
        Dev->>Git: 更新PR
    else 审查通过
        DBA->>Git: Approve PR
    end

    Note over Dev,Monitor: 测试环境迁移

    Git->>CI: 合并PR
    CI->>StagingDB: 执行迁移脚本
    StagingDB->>Backup: 迁移前自动备份
    Backup-->>StagingDB: 备份完成

    StagingDB->>StagingDB: 执行迁移脚本
    StagingDB->>StagingDB: 验证表结构
    StagingDB->>StagingDB: 插入测试数据
    StagingDB->>StagingDB: 运行功能测试

    alt 测试失败
        StagingDB->>Backup: 恢复备份
        Backup-->>StagingDB: 恢复完成
        StagingDB-->>Dev: 测试失败
        Dev->>Dev: 修复脚本
        Dev->>Git: 重新提交
    else 测试通过
        StagingDB-->>CI: 迁移成功
    end

    Note over Dev,Monitor: 生产环境迁移

    CI->>DBA: 请求生产迁移批准
    DBA->>DBA: 确认迁移窗口
    Note over DBA: 选择业务低峰期<br/>如：凌晨2:00-4:00

    DBA->>ProdDB: 开始生产迁移
    ProdDB->>Backup: 全量备份
    Backup-->>ProdDB: 备份完成
    Note over Backup: 备份位置: /backup/mysql/YYYYMMDD/

    ProdDB->>ProdDB: 检查磁盘空间
    ProdDB->>ProdDB: 检查表锁定情况
    ProdDB->>ProdDB: 执行迁移脚本
    Note over ProdDB: 使用pt-online-schema-change<br/>避免长时间锁表

    ProdDB->>ProdDB: 验证表结构
    ProdDB->>ProdDB: 验证索引创建
    ProdDB->>ProdDB: 验证外键约束

    alt 迁移失败
        ProdDB->>Backup: 立即恢复备份
        Backup-->>ProdDB: 恢复完成
        ProdDB-->>DBA: 迁移失败
        DBA->>Dev: 通知开发
        Dev->>Dev: 分析日志
    else 迁移成功
        ProdDB->>Monitor: 记录迁移成功
        Monitor->>Monitor: 发送通知
        ProdDB-->>DBA: 迁移完成
    end

    Note over Dev,Monitor: 迁移后验证

    Monitor->>ProdDB: 监控数据库性能
    Monitor->>Monitor: 检查慢查询日志
    Monitor->>Monitor: 检查错误日志

    alt 性能下降
        Monitor->>DBA: 发送告警
        DBA->>DBA: 分析性能瓶颈
        DBA->>ProdDB: 优化索引/SQL
    else 性能正常
        Monitor->>Monitor: 发送迁移成功通知
    end

    Note over Dev,Monitor: 清理工作

    ProdDB->>Backup: 清理过期备份
    Backup->>Backup: 保留7天备份
    ProdDB->>ProdDB: 更新schema_migrations表
```

### 2.2 在线DDL变更流程（无锁变更）

```mermaid
sequenceDiagram
    participant Dev as 开发者
    participant Tool as pt-osc工具
    participant OldTable as 原表
    participant NewTable as 新表
    participant Triggers as 触发器
    participant App as 应用

    Note over Dev,App: 在线DDL变更（pt-online-schema-change）

    Dev->>Tool: 执行pt-osc命令
    Note over Dev,Tool: pt-online-schema-change<br/>--alter "ADD COLUMN new_col INT"<br/>--execute

    Tool->>OldTable: 检查表结构
    Tool->>NewTable: 创建空的新表
    Note over NewTable: _teams_new

    Tool->>NewTable: 执行DDL变更
    Note over NewTable: 新表上执行ALTER<br/>不影响原表

    Tool->>Triggers: 创建3个触发器
    Note over Triggers: - INSERT触发器<br/>- UPDATE触发器<br/>- DELETE触发器

    Note over Tool,Triggers: 将原表的增删改操作<br/>同步到新表

    Tool->>NewTable: 分批复制历史数据
    Note over Tool,NewTable: chunk_size=1000<br/>避免长时间锁表

    loop 每次复制1000条
        Tool->>OldTable: SELECT * FROM old_table LIMIT 1000 OFFSET ?
        Tool->>NewTable: INSERT INTO new_table VALUES (...)
    end

    Note over Tool,NewTable: 数据复制过程中<br/>触发器保持同步

    Tool->>NewTable: 检查数据一致性
    Tool->>NewTable: 验证行数一致

    alt 数据不一致
        Tool->>NewTable: 重新同步差异
    end

    Tool->>App: 锁表（极短时间）
    Note over Tool,App: LOCK TABLES old_table WRITE, new_table WRITE

    Tool->>NewTable: 最终同步
    Tool->>OldTable: RENAME old_table TO old_table_backup
    Tool->>NewTable: RENAME new_table TO old_table

    Tool->>App: UNLOCK TABLES
    Note over Tool,App: 锁表时间<1秒

    Tool->>Triggers: 删除触发器
    Tool->>OldTable: 删除旧表（异步）

    Tool->>Dev: 变更完成
    Note over Tool,Dev: 整个过程无停机<br/>对应用透明
```

### 3.1 完整部署流程（蓝绿部署）

```mermaid
sequenceDiagram

    participant User as 用户

    Note over Git: tag: v1.2.3

    Git->>CI: 触发CI流程
    CI->>CI: 拉取代码
    CI->>CI: 运行单元测试
    CI->>CI: 构建Docker镜像

    alt 测试失败
        CI-->>Dev: 返回失败
        CI->>CI: 清理镜像
    end

    Note over Registry: coze-studio:v1.2.3

    NewPod->>NewPod: 拉取镜像
    NewPod->>NewPod: 启动容器

        K8s-->>Dev: 部署失败
        NewPod->>NewPod: 删除新Pod
    end

    K8s->>K8s: 更新Service选择器

    Note over K8s,LB: 流量切换阶段

    LB->>OldPod: 逐步减少流量
    LB->>NewPod: 逐步增加流量
    Note over LB: 每次切换10%<br/>观察10分钟

    loop 渐进式流量切换
        LB->>NewPod: 发送10%流量
        Monitor->>NewPod: 监控错误率
        Monitor->>NewPod: 监控响应时间
        Monitor->>NewPod: 监控CPU/内存

        alt 错误率上升
            LB->>OldPod: 立即回切流量
            LB->>K8s: 标记新版本为失败
            K8s-->>Dev: 回滚通知
        end
    end

    LB->>NewPod: 100%流量切换到新版本
    LB->>OldPod: 停止向旧版本发送流量

    Note over K8s,User: 验证阶段

    Monitor->>NewPod: 持续监控30分钟
    User->>NewPod: 访问应用
    NewPod->>User: 返回响应

    alt 性能正常
        Monitor->>Monitor: 发送部署成功通知
        K8s->>OldPod: 删除旧Pod（绿色）
    else 性能异常
        LB->>OldPod: 立即回切流量
        K8s->>K8s: 滚动回滚
        K8s-->>Dev: 回滚完成
    end

    Note over Dev,User: 部署完成

    Monitor->>Monitor: 记录部署指标
    Note over Monitor: - 部署时间<br/>- 回滚次数<br/>- 错误率<br/>- 响应时间
```

```mermaid
    LB->>Stable: 分发75%流量

    Monitor->>Monitor: 观察30分钟

    alt 错误率异常
        Monitor->>LB: 回切到5%或0%
        Ops->>Ops: 分析问题
    end

    Note over LB,User: 第3阶段：50%流量

    Ops->>LB: 更新流量比例
    Note over LB: 50% -> canary<br/>50% -> stable

    LB->>Canary: 分发50%流量
    LB->>Stable: 分发50%流量

    Monitor->>Monitor: 观察1小时

    alt 错误率异常
        Monitor->>LB: 回切到25%
        Ops->>Ops: 分析问题
    end

    Note over LB,User: 第4阶段：100%切换

    Ops->>LB: 更新流量比例
    Note over LB: 100% -> canary<br/>0% -> stable

    LB->>Canary: 分发100%流量
    LB->>Stable: 停止流量

    Monitor->>Monitor: 观察2小时

    alt 稳定运行

        K8s->>Stable: 删除旧版本Pod
        K8s->>K8s: 将canary提升为stable
    else 出现问题
        Monitor->>LB: 立即回切到stable
        K8s->>Canary: 删除金丝雀Pod
    end
```

## 七、监控告警配置

### 7.1 Prometheus告警规则

```yaml
# prometheus-rules.yaml
groups:
  - name: coze-studio-alerts
    interval: 30s
    rules:
      # Pod崩溃重启告警
      - alert: PodCrashLooping
        expr: rate(kube_pod_container_status_restarts_total[15m]) > 0
        for: 5m
        labels:
          severity: critical
        annotations:
          summary: "Pod {{ $labels.pod }} 崩溃重启"
          description: "命名空间 {{ $labels.namespace }} 中的 Pod {{ $labels.pod }} 在过去15分钟内重启了 {{ $value }} 次"

      # 错误率告警
      - alert: HighErrorRate
        expr: rate(http_requests_total{status=~"5.."}[5m]) > 0.01
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "API错误率过高"
          description: "{{ $labels.endpoint }} 的错误率为 {{ $value | humanizePercentage }}"

      # 响应时间告警
      - alert: HighResponseTime
        expr: histogram_quantile(0.95, rate(http_request_duration_seconds_bucket[5m])) > 0.5
        for: 10m
        labels:
          severity: warning
        annotations:
          summary: "API响应时间过长"
          description: "{{ $labels.endpoint }} 的P95响应时间为 {{ $value }}s"

      # CPU使用率告警
      - alert: HighCPUUsage
        expr: rate(container_cpu_usage_seconds_total[5m]) > 0.8
        for: 10m
        labels:
          severity: warning
        annotations:
          summary: "CPU使用率过高"
          description: "容器 {{ $labels.container }} 的CPU使用率为 {{ $value | humanizePercentage }}"

      # 内存使用率告警
      - alert: HighMemoryUsage
        expr: container_memory_usage_bytes / container_spec_memory_limit_bytes > 0.9
        for: 5m
        labels:
          severity: critical
        annotations:
          summary: "内存使用率过高"
          description: "容器 {{ $labels.container }} 的内存使用率为 {{ $value | humanizePercentage }}"
```
