```go
// backend/rate_limiter/token_bucket.go
package rate_limiter

import (
    "context"
    "time"
    "github.com/go-redis/redis/v8"
)

type TokenBucketLimiter struct {
    redis      *redis.Client
    capacity   int64     // 桶容量
    rate       int64     // 令牌生成速率（每秒）
    window     time.Duration
}

func NewTokenBucketLimiter(redis *redis.Client, capacity, rate int64) *TokenBucketLimiter {
    return &TokenBucketLimiter{
        redis:    redis,
        capacity: capacity,
        rate:     rate,
        window:   time.Second,
    }
}

func (l *TokenBucketLimiter) Allow(ctx context.Context, key string) (bool, error) {
    // 使用Redis实现令牌桶
    now := time.Now().Unix()

    // Lua脚本保证原子性
    script := `
        local key = KEYS[1]
        local now = tonumber(ARGV[1])
        local capacity = tonumber(ARGV[2])
        local rate = tonumber(ARGV[3])

        local bucket = redis.call('HMGET', key, 'tokens', 'last_refill')
        local tokens = tonumber(bucket[1]) or capacity
        local last_refill = tonumber(bucket[2]) or now

        -- 计算需要补充的令牌数
        local elapsed = now - last_refill
        tokens = math.min(capacity, tokens + elapsed * rate)

        -- 检查是否有足够令牌
        if tokens >= 1 then
            tokens = tokens - 1
            redis.call('HMSET', key, 'tokens', tokens, 'last_refill', now)
            redis.call('EXPIRE', key, 3600)
            return 1
        else
            return 0
        end
    `

    result, err := l.redis.Eval(ctx, script, []string{key}, now, l.capacity, l.rate).Result()
    if err != nil {
        return false, err
    }

    return result.(int64) == 1, nil
}
```

```go
func (s *UserService) GetUser(ctx context.Context, userID int64) (*entity.User, error) {
    // 先查缓存
    cacheKey := fmt.Sprintf("user:%d", userID)
    cached, err := s.cache.Get(ctx, cacheKey)
    if err == nil && cached != nil {
        var user entity.User
        if json.Unmarshal([]byte(cached), &user) == nil {
            return &user, nil
        }
    }

    }

    data, _ := json.Marshal(user)
    s.cache.Set(ctx, cacheKey, string(data), 5*time.Minute)

    return user, nil
}
```

### 7.10 限流保护流程

```mermaid
sequenceDiagram
    participant Client as 客户端
    participant Gateway as API网关
    participant RateLimiter as 限流中间件
    participant Redis as Redis集群
    participant Audit as 审计服务
    participant Handler as 业务处理器

    Client->>Gateway: 发送API请求
    Gateway->>RateLimiter: 限流检查

    RateLimiter->>RateLimiter: 提取限流key
    Note over RateLimiter: key格式: ratelimit:{resource}:{user_id}:{window}

    RateLimiter->>RateLimiter: 确定限流策略
    Note over RateLimiter: 策略: Token Bucket / Sliding Window / Fixed Window

    alt 令牌桶算法
        RateLimiter->>Redis: 执行Lua脚本（原子操作）
        Note over Redis: 1. 获取当前tokens<br/>2. 计算需要补充的tokens<br/>3. 检查是否有足够tokens<br/>4. 扣减tokens<br/>5. 更新last_refill时间

        Redis->>Redis: 执行令牌桶算法
        Redis-->>RateLimiter: 返回结果（1=允许, 0=拒绝）

        alt tokens >= 1
            RateLimiter->>Handler: 放行请求
            Handler->>Handler: 处理业务逻辑
            Handler->>Audit: 记录正常业务日志

            Handler-->>Gateway: 返回业务响应
            Gateway-->>Client: 200 OK
        else tokens < 1
            RateLimiter->>Audit: 记录限流触发日志
            Audit->>Audit: 持久化日志

            RateLimiter->>RateLimiter: 设置响应头
            Note over RateLimiter: X-RateLimit-Limit: 100<br/>X-RateLimit-Remaining: 0<br/>X-RateLimit-Reset: 1234567890

            RateLimiter-->>Gateway: 429 Too Many Requests
            Gateway-->>Client: 返回限流错误
        end

    else 滑动窗口算法
        RateLimiter->>Redis: 有序集合存储请求时间戳
        Note over Redis: ZADD ratelimit:key <timestamp> <member_id>

        RateLimiter->>Redis: 移除窗口外的时间戳
        Note over Redis: ZREMRANGEBYSCORE ratelimit:key <current_timestamp - window_size>

        RateLimiter->>Redis: 统计窗口内请求数
        Note over Redis: ZCARD ratelimit:key

        Redis-->>RateLimiter: 返回请求数量

        alt 请求数 <= 阈值
            RateLimiter->>Handler: 放行请求
            Handler->>Handler: 处理业务逻辑
            Handler-->>Gateway: 200 OK
            Gateway-->>Client: 返回业务响应
        else 请求数 > 阈值
            RateLimiter->>Audit: 记录限流触发日志
            RateLimiter-->>Gateway: 429 Too Many Requests
            Gateway-->>Client: 返回限流错误
        end
    end

    Note over Client,Audit: 分布式限流集群部署

    Client->>Gateway: 发送请求到负载均衡器
    Gateway->>Gateway: 转发到实例1
    Gateway->>RateLimiter1: 实例1限流检查
    RateLimiter1->>Redis: 共享Redis集群

    alt 实例1限流通过
        RateLimiter1->>Handler1: 处理请求
        Handler1-->>Gateway: 200 OK
    else 实例1限流触发
        RateLimiter1-->>Gateway: 429 Too Many Requests
        Gateway->>Gateway: 转发到实例2
        Gateway->>RateLimiter2: 实例2限流检查
        RateLimiter2->>Redis: 共享Redis集群

        alt 实例2限流通过
            RateLimiter2->>Handler2: 处理请求
            Handler2-->>Gateway: 200 OK
        else 实例2也限流
            RateLimiter2-->>Gateway: 429 Too Many Requests
            Gateway-->>Client: 429 Too Many Requests
        end
    end
```

### 7.12 Webhook事件推送流程

```mermaid
sequenceDiagram
    participant Trigger as 触发源（Bot/工作流）
    participant WebhookService as Webhook服务
    participant Queue as 消息队列
    participant Consumer as 消费者
    participant Target as 目标系统
    participant Retry as 重试服务
    participant Audit as 审计服务

    Note over Trigger,Target: 事件触发与推送流程

    Trigger->>WebhookService: 触发事件
    Note over Trigger: - Bot对话完成<br/>- 工作流执行完成<br/>- 用户注册

    WebhookService->>WebhookService: 构建事件payload
    Note over WebhookService: {<br/>  event_type: "conversation.completed",<br/>  event_id: "evt_xxx",<br/>  timestamp: 1234567890,<br/>  data: { ... }<br/>}

    WebhookService->>WebhookService: 查询订阅此事件的Webhook配置
    WebhookService->>DB: SELECT * FROM webhooks WHERE event_type = :event_type AND status = 'active'
    DB-->>WebhookService: 返回Webhook列表

    WebhookService->>Queue: 发送Webhook任务到队列
    Note over Queue: topic: webhook_events
    Queue->>Queue: 持久化任务消息

    WebhookService-->>Trigger: 202 Accepted (异步处理)

    Queue->>Consumer: 消费Webhook任务
    Consumer->>Consumer: 解析任务payload
    Consumer->>Consumer: 提取Webhook配置

    loop 每个Webhook配置
        Consumer->>Consumer: 准备请求
        Consumer->>Consumer: 签名payload（HMAC-SHA256）
        Note over Consumer: signature = HMAC-SHA256(secret_key, payload)

        Consumer->>Target: POST webhook_url
        Note over Target: Content-Type: application/json<br/>X-Webhook-Event: conversation.completed<br/>X-Webhook-Signature: xxx<br/>X-Webhook-Timestamp: 1234567890

        alt HTTP 200 OK
            Target-->>Consumer: 返回成功响应
            Consumer->>Audit: 记录推送成功日志
            Consumer->>DB: UPDATE webhook_delivery_logs SET status='success'

        else HTTP 4xx/5xx 错误
            Target-->>Consumer: 返回错误响应
            Consumer->>Audit: 记录推送失败日志
            Consumer->>DB: UPDATE webhook_delivery_logs SET status='failed', error_msg=xxx

            alt 检查是否需要重试
                Consumer->>Retry: 发送到重试队列
                Note over Retry: 重试策略: 指数退避<br/>- 1分钟后重试<br/>- 5分钟后重试<br/>- 30分钟后重试<br/>- 最多重试3次

                Retry->>Retry: 计算下次重试时间
                Retry->>Queue: 延迟重新入队
            end
        end

        alt 响应超时（30秒）
            Consumer->>Consumer: 标记为超时
            Consumer->>Retry: 发送到重试队列
        end
    end

    Consumer->>Audit: 记录批量推送完成日志
    Consumer->>DB: 更新Webhook统计信息
    Note over DB: - 成功推送次数<br/>- 失败推送次数<br/>- 平均响应时间
```
