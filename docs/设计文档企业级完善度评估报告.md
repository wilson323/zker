```go
package middleware

import (
    "context"

// TenantIdentificationMiddleware 租户识别中间件
func TenantIdentificationMiddleware() app.HandlerFunc {
    return func(c context.Context, ctx *app.RequestContext) {
        // 优先级1: 子域名识别
        if tenantID := identifyFromSubdomain(ctx); tenantID != "" {
            setTenantContext(c, ctx, tenantID)
            ctx.Next(c)
            return
        }

        // 优先级2: 路径识别
        if tenantID := identifyFromPath(ctx); tenantID != "" {
            setTenantContext(c, ctx, tenantID)
            ctx.Next(c)
            return
        }

        // 优先级3: Header识别
        if tenantID := identifyFromHeader(ctx); tenantID != "" {
            setTenantContext(c, ctx, tenantID)
            ctx.Next(c)
            return
        }

        // 无法识别租户
        ctx.SetStatusCode(401)
        ctx.JSON(map[string]interface{}{
            "code":       40100,
            "message":    "Tenant not identified",
            "error_code": "TENANT_NOT_IDENTIFIED",
        })
    }
}

func identifyFromSubdomain(ctx *app.RequestContext) string {
    host := string(ctx.GetHeader("Host"))
    // 提取子域名: tenant-a.saas.coze.com -> tenant-a
    parts := strings.Split(host, ".")
    if len(parts) >= 3 {
        return parts[0]
    }
    return ""
}

func setTenantContext(c context.Context, ctx *app.RequestContext, tenantID string) {
    ctx.Set("tenant_id", tenantID)
    // 注入到请求上下文
    ctx.SetRequestWithContext(context.WithValue(c, "tenant_id", tenantID))
}
```

```go
type ConfigManager interface {
    // 加载配置
    Load(configPath string) error

    // 获取配置
    Get(key string) interface{}
    GetString(key string) string
    GetInt(key string) int
    GetBool(key string) bool

    // 监听配置变化
    Watch(key string, callback func(oldValue, newValue interface{})) error

    // 热更新配置
    Reload() error
}

// 使用etcd实现配置中心
type EtcdConfigManager struct {
    client *clientv3.Client
    cache  sync.Map
}

func (m *EtcdConfigManager) Load(configPath string) error {
    // 从etcd加载配置
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()

    resp, err := m.client.Get(ctx, configPath, clientv3.WithPrefix())
    if err != nil {
        return err
    }

    for _, kv := range resp.Kvs {
        m.cache.Store(string(kv.Key), string(kv.Value))
    }

    // 监听配置变化
    go m.watchConfigChanges(configPath)

    return nil
}

func (m *EtcdConfigManager) watchConfigChanges(prefix string) {
    watchChan := m.client.Watch(context.Background(), prefix, clientv3.WithPrefix())
    for wresp := range watchChan {
        for _, ev := range wresp.Events {
            oldVal, _ := m.cache.Load(string(ev.Kv.Key))
            m.cache.Store(string(ev.Kv.Key), string(ev.Kv.Value))
            // 触发回调
            fmt.Printf("Config changed: %s\n%s -> %s\n", ev.Kv.Key, oldVal, ev.Kv.Value)
        }
    }
}
```
