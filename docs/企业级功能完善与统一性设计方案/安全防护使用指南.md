## 后端使用指南

### 1. XSS防护

#### 1.1 在API Handler中使用

```go
package handler

import (
    "github.com/cloudwego/hertz/pkg/app"
    "github.com/coze-dev/coze-studio/backend/pkg/security"
)

// GetBot 获取Bot信息
func GetBot(ctx context.Context, c *app.RequestContext) {
    botID := c.Param("bot_id")

    // 获取Bot
    bot, err := service.GetBot(ctx, botID)
    if err != nil {
        c.JSON(500, map[string]string{"error": err.Error()})
        return
    }

    // ✅ 转义用户输入的字段
    bot.Name = security.EscapeHTML(bot.Name)
    bot.Description = security.EscapeHTML(bot.Description)
    bot.Instruction = security.EscapeHTML(bot.Instruction)

    c.JSON(200, bot)
}

// CreateBot 创建Bot
func CreateBot(ctx context.Context, c *app.RequestContext) {
    var req CreateBotRequest
    c.BindAndValidate(&req)

    // ✅ 验证用户输入
    if valid, reason := security.ValidateInput(req.Name); !valid {
        c.JSON(400, map[string]string{
            "error": "Invalid input: " + reason,
            "code":  "ERR_INVALID_INPUT",
        })
        return
    }

    if valid, reason := security.ValidateInput(req.Description); !valid {
        c.JSON(400, map[string]string{
            "error": "Invalid description: " + reason,
            "code":  "ERR_INVALID_INPUT",
        })
        return
    }

    // 继续处理...
    bot, err := service.CreateBot(ctx, &req)
    if err != nil {
        c.JSON(500, map[string]string{"error": err.Error()})
        return
    }

    c.JSON(200, bot)
}

// UpdateBot 更新Bot
func UpdateBot(ctx context.Context, c *app.RequestContext) {
    botID := c.Param("bot_id")

    var req UpdateBotRequest
    c.BindAndValidate(&req)

    // ✅ 验证URL输入
    if req.IconURL != "" && !security.IsValidURL(req.IconURL) {
        c.JSON(400, map[string]string{
            "error": "Invalid icon URL",
            "code":  "ERR_INVALID_URL",
        })
        return
    }

    // ✅ 截断超长输入
    req.Description = security.TruncateText(req.Description, 10000)

    // 继续处理...
    err := service.UpdateBot(ctx, botID, &req)
    if err != nil {
        c.JSON(500, map[string]string{"error": err.Error()})
        return
    }

    c.JSON(200, map[string]string{"message": "Bot updated successfully"})
}
```

#### 1.2 清理HTML内容(富文本场景)

```go
// SaveKnowledge 保存知识库文档(允许富文本,但需要清理)
func SaveKnowledge(ctx context.Context, c *app.RequestContext) {
    var req SaveKnowledgeRequest
    c.BindAndValidate(&req)

    // ✅ 清理HTML内容(保留安全标签,移除危险内容)
    cleanContent := security.SanitizeHTML(req.Content)

    // 继续处理...
    doc, err := service.SaveKnowledge(ctx, req.KnowledgeID, cleanContent)
    if err != nil {
        c.JSON(500, map[string]string{"error": err.Error()})
        return
    }

    c.JSON(200, doc)
}
```

### 2. CSRF防护

#### 2.1 在路由中使用

```go
package router

import (
    "github.com/cloudwego/hertz/pkg/app/server"
    "github.com/coze-dev/coze-studio/backend/api/handler/coze"
    "github.com/coze-dev/coze-studio/backend/api/middleware"
)

func RegisterRoutes(r *app.Server) {
    // ✅ 1. 全局应用安全头中间件
    r.Use(middleware.SecurityHeadersMiddleware())

    // ✅ 2. 公开接口(不需要CSRF保护)
    r.GET("/api/csrf_token", middleware.GetCSRFTokenHandler)
    r.GET("/api/health", handler.HealthCheck)

    // API组
    api := r.Group("/api")

    // ✅ 3. 需要CSRF保护的写操作
    api.POST("/bots",
        middleware.CSRFMiddleware(),  // ✅ 添加CSRF保护
        handler.CreateBot)

    api.PUT("/bots/:bot_id",
        middleware.CSRFMiddleware(),  // ✅ 添加CSRF保护
        handler.UpdateBot)

    api.DELETE("/bots/:bot_id",
        middleware.CSRFMiddleware(),  // ✅ 添加CSRF保护
        handler.DeleteBot)

    api.POST("/knowledge",
        middleware.CSRFMiddleware(),  // ✅ 添加CSRF保护
        handler.CreateKnowledge)

    // ✅ 4. 读操作不需要CSRF保护(GET自动跳过)
    api.GET("/bots", handler.ListBots)
    api.GET("/bots/:bot_id", handler.GetBot)
}
```

#### 2.2 提供CSRF Token

```go
// GetCSRFToken 提供CSRF Token给前端
func GetCSRFToken(ctx context.Context, c *app.RequestContext) {
    token, err := middleware.GenerateCSRFToken()
    if err != nil {
        c.JSON(500, map[string]string{
            "error": "Failed to generate CSRF token",
            "code":  "ERR_CSRF_GENERATE_FAILED",
        })
        return
    }

    // ✅ 设置Token到Cookie
    middleware.SetCSRFToken(c, token)

    // ✅ 返回Token给前端
    c.JSON(200, map[string]string{
        "token": token,
    })
}
```

```tsx
import { sanitizeHTML, escapeHTML, isInputSafe } from '@/common/biz-components/security';

// ✅ 场景1: 渲染用户输入的HTML内容
function BotDescription({ description }: { description: string }) {
  // 清理HTML内容

  return (
    <div
      className="bot-description"
      dangerouslySetInnerHTML={{ __html: cleanHtml }}

// ✅ 场景2: 渲染纯文本(推荐)
function BotName({ name }: { name: string }) {
  // 转义HTML特殊字符
  const safeName = escapeHTML(name);

  return <div className="bot-name">{safeName}</div>;
}

// ✅ 场景3: 表单输入验证
function CreateBotForm() {
  const [name, setName] = useState('');
  const [error, setError] = useState('');

  const handleSubmit = () => {
    // 验证输入安全性
    if (!isInputSafe(name)) {
      setError('Bot名称包含不安全内容');
      return;
    }

    // 继续提交...
  };

  return (
    <form onSubmit={handleSubmit}>
      <input
        value={name}
        onChange={(e) => setName(e.target.value)}

  );
}
```

### 2. CSRF防护

#### 2.1 使用fetchWithCSRF封装

```tsx
import { fetchWithCSRF } from '@/common/biz-components/security';

// ✅ 创建Bot
async function createBot(botData: Bot) {
  try {
    const response = await fetchWithCSRF('/api/bots', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(botData),
    });

    if (!response.ok) {
      throw new Error('Failed to create bot');
    }

    const data = await response.json();
    return data;
  } catch (error) {
    console.error('Error:', error);
    throw error;
  }
}

// 使用
createBot({
  name: 'My Bot',
  description: 'A helpful assistant',
})
  .then(bot => console.log('Bot created:', bot))
  .catch(err => console.error('Failed:', err));
```

#### 2.2 使用React Hook

```tsx
import { useCSRFProtection } from '@/common/biz-components/security';

function CreateBotForm() {
  const [name, setName] = useState('');
  const { fetch: fetchWithCSRF, loading, error } = useCSRFProtection();

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();

    try {
      // 自动携带CSRF Token
      const response = await fetchWithCSRF('/api/bots', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({ name }),
      });

      if (!response.ok) {
        throw new Error('Failed to create bot');
      }

      const bot = await response.json();
      console.log('Bot created:', bot);

      // 成功后刷新Token(可选)
      // await getToken();
    } catch (err) {
      console.error('Error:', err);
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      <input
        value={name}
        onChange={(e) => setName(e.target.value)}
        placeholder="Bot名称"
      />
      <button type="submit" disabled={loading}>
        {loading ? '创建中...' : '创建Bot'}
      </button>
      {error && <div className="error">{error.message}</div>}
    </form>
  );
}
```

### 3. 完整示例: Bot管理页面

```tsx
import { useState, useEffect } from 'react';
import { useXSSProtection, useCSRFProtection } from '@/common/biz-components/security';

interface Bot {
  bot_id: string;
  name: string;
  description: string;
}

function BotManagement() {
  const [bots, setBots] = useState<Bot[]>([]);
  const [newBotName, setNewBotName] = useState('');
  const [newBotDesc, setNewBotDesc] = useState('');

  // XSS防护
  const { sanitize, validate } = useXSSProtection();

  // CSRF防护
  const { fetch: fetchWithCSRF, loading, error } = useCSRFProtection();

  // 获取Bot列表(GET请求,不需要CSRF)
  useEffect(() => {
    fetch('/api/bots')
      .then(res => res.json())
      .then(data => setBots(data.bots))
      .catch(err => console.error('Failed to fetch bots:', err));
  }, []);

  // 创建Bot(POST请求,需要CSRF)
  const handleCreateBot = async () => {
    // ✅ 验证输入安全性
    if (!validate(newBotName)) {
      alert('Bot名称包含不安全内容');
      return;
    }

    if (!validate(newBotDesc)) {
      alert('Bot描述包含不安全内容');
      return;
    }

    try {
      // ✅ 自动携带CSRF Token
      const response = await fetchWithCSRF('/api/bots', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          name: newBotName,
          description: newBotDesc,
        }),
      });

      if (!response.ok) {
        throw new Error('Failed to create bot');
      }

      const newBot = await response.json();

      // 添加到列表
      setBots([...bots, newBot]);

      // 清空表单
      setNewBotName('');
      setNewBotDesc('');
    } catch (err) {
      console.error('Error:', err);
      alert('创建Bot失败: ' + (err as Error).message);
    }
  };

  // 删除Bot(DELETE请求,需要CSRF)
  const handleDeleteBot = async (botId: string) => {
    if (!confirm('确定要删除这个Bot吗?')) {
      return;
    }

    try {
      // ✅ 自动携带CSRF Token
      const response = await fetchWithCSRF(`/api/bots/${botId}`, {
        method: 'DELETE',
      });

      if (!response.ok) {
        throw new Error('Failed to delete bot');
      }

      // 从列表中移除
      setBots(bots.filter(bot => bot.bot_id !== botId));
    } catch (err) {
      console.error('Error:', err);
      alert('删除Bot失败: ' + (err as Error).message);
    }
  };

  return (
    <div className="bot-management">
      <h1>Bot管理</h1>

      {/* 创建Bot表单 */}
      <div className="create-bot-form">
        <h2>创建新Bot</h2>
        <input
          value={newBotName}
          onChange={(e) => setNewBotName(e.target.value)}
          placeholder="Bot名称"
        />
        <textarea
          value={newBotDesc}
          onChange={(e) => setNewBotDesc(e.target.value)}
          placeholder="Bot描述"
        />
        <button onClick={handleCreateBot} disabled={loading}>
          {loading ? '创建中...' : '创建Bot'}
        </button>
        {error && <div className="error">{error.message}</div>}
      </div>

      {/* Bot列表 */}
      <div className="bot-list">
        <h2>我的Bot</h2>
        {bots.map(bot => (
          <div key={bot.bot_id} className="bot-item">
            {/* ✅ 渲染用户输入时使用sanitizeHTML */}
            <div
              className="bot-name"
              dangerouslySetInnerHTML={{
                __html: sanitize(bot.name),
              }}
            />
            <div
              className="bot-description"
              dangerouslySetInnerHTML={{
                __html: sanitize(bot.description),
              }}
            />
            <button onClick={() => handleDeleteBot(bot.bot_id)}>
              删除
            </button>
          </div>
        ))}
      </div>
    </div>
  );
}

export default BotManagement;
```

## 常见问题

### Q1: 哪些API需要CSRF保护?

**A**: 所有写操作API(POST/PUT/DELETE/PATCH)都需要CSRF保护。读操作(GET/HEAD/OPTIONS)自动跳过验证。

### Q2: 如果XSS防护导致功能异常怎么办?

**A**: 检查以下几点:
1. 是否正确使用了`sanitizeHTML`(富文本场景)或`escapeHTML`(纯文本场景)
2. 是否允许了必要的HTML标签和属性
3. 考虑使用`stripTags`移除所有标签,只保留纯文本

### Q3: CSRF Token如何刷新?

**A**: Token的有效期为24小时,过期后需要重新获取。建议:
- 页面加载时调用`GET /api/csrf_token`获取Token
- 敏感操作前刷新Token
- Token存储在内存中,不要持久化到LocalStorage

### Q4: 如何处理第三方API集成?

**A**: 第三方API回调接口需要跳过CSRF验证:
```go
// 在路由中排除第三方回调
r.POST("/api/webhooks/third-party",
    middleware.CSRFMiddlewareSkip(),  // 跳过CSRF验证
    handler.WebhookHandler)
```
