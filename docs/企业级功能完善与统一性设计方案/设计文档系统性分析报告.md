func (e *HybridSearchEngine) HybridSearch(
    ctx context.Context,
    query string,
    config *HybridSearchConfig,
) ([]SearchResult, error) {
    // 1. 向量检索
    vectorResults, err := e.vectorSearch.Search(ctx, query, config.TopK)
    if err != nil {
        return nil, err
    }

    // 2. 关键词检索
    keywordResults, err := e.keywordSearch.Search(ctx, query, config.TopK)
    if err != nil {
        return nil, err
    }

    // 3. RRF融合
    fusedResults := e.fuseResults(vectorResults, keywordResults, config)

    // 4. Reranker (可选)
    if config.RerankerEnabled {
        fusedResults, err = e.reranker.Rerank(ctx, query, fusedResults)
        if err != nil {
            return nil, err
        }
    }

    return fusedResults, nil
}
```

type TemporaryGrant struct {
    ID           int64     `json:"id"`
    UserID       int64     `json:"user_id"`
    ResourceID   int64     `json:"resource_id"`
    Permission   string    `json:"permission"`
    GrantedAt    time.Time `json:"granted_at"`
    ExpiresAt    time.Time `json:"expires_at"`
    GrantedBy    int64     `json:"granted_by"`
    Reason       string    `json:"reason"`
}

func (s *PermissionService) CheckTemporaryGrant(
    ctx context.Context,
    userID, resourceID int64,
    permission string,
) (bool, error) {
    // 1. 查询临时授权
    grant, err := s.temporaryGrantRepo.FindValidGrant(
        ctx, userID, resourceID, permission,
    )
    if err != nil {
        return false, err
    }

    // 2. 检查是否过期
    if grant.ExpiresAt.Before(time.Now()) {
        return false, nil
    }

    return true, nil
}
```
