## 1. 模块概述

**记忆引擎** 负责存储和管理对话历史、用户偏好等记忆数据，通过**向量存储 + 语义检索**实现长期记忆和短期记忆管理。

**核心设计理念**：
- ✅ **分层记忆**：会话记忆、用户记忆、知识记忆
- ✅ **语义检索**：基于向量相似度的记忆检索
- ✅ **记忆压缩**：自动摘要和压缩长对话

**实现策略**：✅ 50% 编码（记忆框架） + 50% 配置（记忆策略）

## 2. 核心功能

### 2.1 记忆存储

**F1 - 会话记忆**：当前对话的上下文
**F2 - 用户记忆**：用户的长期偏好、习惯
**F3 - 知识记忆**：从对话中提炼的知识点

### 2.2 记忆检索

**F4 - 语义检索**：基于问题检索相关记忆
**F5 - 关键词检索**：精确匹配
**F6 - 时序检索**：按时间范围检索

## 3. 数据库设计

### 3.1 核心表结构

#### 3.1.1 对话记忆表 (conversation_memories)

```sql
CREATE TABLE conversation_memories (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '记忆ID',
    tenant_id VARCHAR(64) NOT NULL COMMENT '租户ID',
    user_id BIGINT NOT NULL COMMENT '用户ID',
    conversation_id VARCHAR(64) NOT NULL COMMENT '会话ID',

    memory_type ENUM('context', 'preference', 'knowledge') NOT NULL,
    content TEXT NOT NULL COMMENT '记忆内容',
    summary TEXT COMMENT '摘要',

    -- 向量化
    vector_id VARCHAR(64) COMMENT '向量ID (Milvus)',

    -- 元数据
    metadata JSON COMMENT '元数据',
    importance DECIMAL(3,2) DEFAULT 0.50 COMMENT '重要性 (0-1)',

    accessed_count INT DEFAULT 0 COMMENT '访问次数',
    last_accessed_at DATETIME COMMENT '最后访问时间',

    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

    INDEX idx_tenant_user_conv (tenant_id, user_id, conversation_id),
    INDEX idx_memory_type (memory_type),
    INDEX idx_vector_id (vector_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
COMMENT='对话记忆表';
```

## 4. 引擎设计

### 4.1 记忆引擎

```go
package memory

// MemoryEngine 记忆引擎
type MemoryEngine struct {
    memoryRepo    repository.MemoryRepository
    vectorStore   *vector.Store
    llmClient     *llm.Client
}

// StoreMemory 存储记忆
func (e *MemoryEngine) StoreMemory(
    ctx context.Context,
    memoryType string,
    content string,
    metadata map[string]interface{},
) error {
    // 1. 向量化内容
    vector, err := e.llmClient.Embed(ctx, content)
    if err != nil {
        return err
    }

    // 2. 存储到向量数据库
    vectorID, err := e.vectorStore.Store(ctx, vector, metadata)
    if err != nil {
        return err
    }

    // 3. 存储到数据库
    memory := &Memory{
        TenantID:    getTenantID(ctx),
        UserID:      getUserID(ctx),
        MemoryType:  memoryType,
        Content:     content,
        VectorID:    vectorID,
        Importance:  calculateImportance(content),
    }

    return e.memoryRepo.Create(ctx, memory)
}

// RetrieveMemories 检索相关记忆
func (e *MemoryEngine) RetrieveMemories(
    ctx context.Context,
    query string,
    topK int,
) ([]*Memory, error) {
    // 1. 向量化查询
    vector, err := e.llmClient.Embed(ctx, query)
    if err != nil {
        return nil, err
    }

    // 2. 向量搜索
    vectorIDs, err := e.vectorStore.Search(ctx, vector, topK)
    if err != nil {
        return nil, err
    }

    // 3. 加载记忆内容
    memories, err := e.memoryRepo.GetByVectorIDs(ctx, vectorIDs)
    if err != nil {
        return nil, err
    }

    // 4. 更新访问统计
    for _, mem := range memories {
        go func(id string) {
            e.memoryRepo.UpdateAccessCount(context.Background(), id)
        }(mem.ID)
    }

    return memories, nil
}

// SummarizeConversation 压缩对话
func (e *MemoryEngine) SummarizeConversation(
    ctx context.Context,
    conversationID string,
    maxTokens int,
) (string, error) {
    // 1. 加载对话历史
    messages, err := e.loadConversation(ctx, conversationID, 20)
    if err != nil {
        return "", err
    }

    // 2. 调用LLM摘要
    summary, err := e.llmClient.Summarize(ctx, messages, maxTokens)
    if err != nil {
        return "", err
    }

    // 3. 存储摘要为记忆
    e.StoreMemory(ctx, "context", summary, map[string]interface{}{
        "conversation_id": conversationID,
        "type": "summary",
    })

    return summary, nil
}
```
