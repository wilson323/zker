## 1. 模块概述

### 1.1 功能定义

**组织中心** 是企业组织架构、人员、岗位、权限的统一管理中心，提供：
- 多级组织管理（公司/部门/团队/小组）
- 岗位管理与能力建模
- 成员管理与生命周期
- 角色权限管理（RBAC）
- 批量导入导出

### 1.2 核心价值

- 🎯 **统一管理** - 企业组织架构、人员、权限一站式管理
- 🎯 **灵活扩展** - 支持3-5级组织树，满足不同企业需求
- 🎯 **精细权限** - 基于RBAC的细粒度权限控制
- 🎯 **高效运维** - 批量操作、数据导入导出、快速入职

### 1.3 业务术语

| 术语 | 英文 | 定义 |
|------|------|------|
| **组织** | Organization | 企业内的行政单位，如公司、部门、团队、小组 |
| **组织树** | Organization Tree | 组织的层级结构，以树形表示 |
| **岗位** | Position | 职位，如产品经理、软件工程师 |
| **成员** | Member | 企业员工 |
| **角色** | Role | 权限的集合，如超级管理员、普通用户 |
| **权限** | Permission | 对系统资源的访问权 |
| **工号** | Employee No | 员工的唯一编号 |

### 1.4 目标用户

| 用户类型 | 典型角色 | 核心需求 |
|---------|---------|----------|
| **超级管理员** | CEO、CTO | 全局管理，查看所有组织 |
| **组织管理员** | HR总监、部门经理 | 管理本组织及下属组织 |
| **普通员工** | - | 查看组织架构、同事信息 |

### 3.2 领域模型设计

```typescript
// 领域模型

// 组织 (聚合根)
class Organization {
  id: string
  tenantId: string
  name: string
  type: OrganizationType        // company | department | team | group
  parentId: string | null
  path: string                  // /org1/org2/org3
  level: number                 // 1 | 2 | 3 | 4 | 5
  status: OrganizationStatus    // active | inactive
  managerId?: string            // 负责人
  description?: string
  createdAt: Date
  updatedAt: Date

  // 方法
  addChild(child: Organization): void
  removeChild(childId: string): void
  moveTo(newParent: Organization): void
  getChildren(): Organization[]
  getAllDescendants(): Organization[]
}

// 岗位 (实体)
class Position {
  id: string
  tenantId: string
  organizationId: string
  name: string
  code: string                   // 岗位编码（如 PM-001）
  description?: string
  level: PositionLevel           // Junior | Middle | Senior | Lead
  capabilities: string[]         // 岗位能力要求
  permissions: string[]          // 岗位权限
  status: PositionStatus         // active | inactive
  createdAt: Date
  updatedAt: Date
}

// 成员 (实体)
class Member {
  id: string
  tenantId: string
  name: string
  email: string
  phone?: string
  employeeNo: string             // 工号
  avatar?: string
  organizationId: string
  positionId: string
  roleIds: string[]
  status: MemberStatus           // active | inactive | locked | resigned
  lastLoginAt?: Date
  joinedAt: Date
  resignedAt?: Date
  createdAt: Date
  updatedAt: Date

  // 方法
  assignRole(roleId: string): void
  revokeRole(roleId: string): void
  transferTo(organizationId: string): void
  resign(): void
}

// 角色 (实体)
class Role {
  id: string
  tenantId: string
  name: string
  code: string
  type: RoleType                 // system | custom
  description?: string
  permissions: string[]
  isDefault: boolean             // 是否默认角色
  createdAt: Date
  updatedAt: Date

  // 方法
  addPermission(permission: string): void
  removePermission(permission: string): void
  hasPermission(permission: string): boolean
}

// 权限 (值对象)
class Permission {
  id: string
  code: string                   // org:member:create
  name: string                   // 创建成员
  module: string                 // org | member | position | role
  action: string                 // create | read | update | delete
  resource: string               // member | organization
}
```

### 4.2 表结构设计

#### 4.2.1 组织表 (organizations)

```sql
CREATE TABLE organizations (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '组织ID',
    tenant_id VARCHAR(64) NOT NULL COMMENT '租户ID',
    parent_id BIGINT DEFAULT NULL COMMENT '父组织ID',
    name VARCHAR(255) NOT NULL COMMENT '组织名称',
    type ENUM('company', 'department', 'team', 'group') NOT NULL COMMENT '组织类型',
    path VARCHAR(1024) NOT NULL COMMENT '组织路径，如/org1/org2/org3',
    level INT NOT NULL COMMENT '组织层级，1-5',
    manager_id BIGINT DEFAULT NULL COMMENT '负责人ID',
    description TEXT COMMENT '组织描述',
    status ENUM('active', 'inactive') DEFAULT 'active' COMMENT '状态',
    sort_order INT DEFAULT 0 COMMENT '排序序号',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    deleted_at DATETIME DEFAULT NULL COMMENT '删除时间',

    UNIQUE KEY uk_tenant_parent_name (tenant_id, parent_id, name),
    INDEX idx_tenant_id (tenant_id),
    INDEX idx_parent_id (parent_id),
    INDEX idx_path (path(255)),
    INDEX idx_level (level),
    INDEX idx_status (status),
    INDEX idx_deleted_at (deleted_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='组织表';
```

### 4.3 索引优化策略

#### 4.3.1 组织查询优化

```sql
-- 常见查询1: 获取租户下所有组织
SELECT * FROM organizations
WHERE tenant_id = ? AND deleted_at IS NULL
ORDER BY path;

-- 索引: idx_tenant_id + idx_path

-- 常见查询2: 获取某个组织的所有子组织
SELECT * FROM organizations
WHERE tenant_id = ? AND parent_id = ? AND deleted_at IS NULL;

-- 索引: idx_tenant_id + idx_parent_id

-- 常见查询3: 获取组织的所有后代（递归）
-- 使用 MySQL 8.0 Recursive CTE
WITH RECURSIVE org_tree AS (
  SELECT * FROM organizations WHERE id = ?
  UNION ALL
  SELECT o.* FROM organizations o
  INNER JOIN org_tree ot ON o.parent_id = ot.id
)
SELECT * FROM org_tree;
```

#### 4.3.2 成员查询优化

```sql
-- 常见查询1: 搜索成员（姓名/邮箱/工号）
SELECT * FROM members
WHERE tenant_id = ?
  AND deleted_at IS NULL
  AND (name LIKE ? OR email LIKE ? OR employee_no LIKE ?)
LIMIT 20;

-- 索引: idx_tenant_id + idx_name + idx_employee_no
-- 或使用全文索引: FULLTEXT INDEX ft_name_email (name, email)

-- 常见查询2: 获取某组织下的所有成员
SELECT m.*, o.name as organization_name, p.name as position_name
FROM members m
LEFT JOIN organizations o ON m.organization_id = o.id
LEFT JOIN positions p ON m.position_id = p.id
WHERE m.tenant_id = ?
  AND m.organization_id = ?
  AND m.deleted_at IS NULL;

-- 索引: idx_tenant_id + idx_organization_id
```

```typescript
      const res = await organizationApi.list();
      setOrganizations(res.data.items);
    } catch (error) {
      console.error('Failed to fetch organizations:', error);
    } finally {
      setLoading(false);
    }
  };

  const createOrganization = async (data: CreateOrganizationDTO) => {
    return await organizationApi.create(data);
  };

  const updateOrganization = async (id: string, data: UpdateOrganizationDTO) => {
    return await organizationApi.update(id, data);
  };

  };

  const moveOrganization = async (id: string, newParentId: string, position: number) => {
    return await organizationApi.move(id, { new_parent_id: newParentId, position });
  };

  return {
    organizations,
    loading,
    fetchOrganizations,
    createOrganization,
```

```go
func (s *OrganizationService) Create(ctx context.Context, tenantID string, dto *CreateOrganizationDTO) (*Organization, error) {
    // 1. 验证父组织存在
    if dto.ParentID != nil {

        }
    }

    var path string
    var level int
    if dto.ParentID != nil {
        var parent Organization
        s.db.WithContext(ctx).First(&parent, *dto.ParentID)

        level = parent.Level + 1
    } else {
        path = "/"
        level = 1
    }

        TenantID:    tenantID,
        ParentID:    dto.ParentID,
        Name:        dto.Name,
        Type:        dto.Type,
        Path:        path,
        Level:       level,
        ManagerID:   dto.ManagerID,
        Description: dto.Description,
        Status:      "active",

        return nil, err

// Delete 删除组织
func (s *OrganizationService) Delete(ctx context.Context, tenantID string, id uint, force bool, migrateTo *uint) error {
    // 1. 查询组织
    var org Organization
    if err := s.db.WithContext(ctx).
        Where("id = ? AND tenant_id = ?", id, tenantID).
        First(&org).Error; err != nil {
        return err
    }

    // 2. 检查是否有子组织
    var childCount int64
    s.db.WithContext(ctx).
        Model(&Organization{}).
        Where("parent_id = ?", id).
        Count(&childCount)

    // 3. 检查是否有成员
    var memberCount int64
    s.db.WithContext(ctx).
        Model(&Member{}).
        Where("organization_id = ?", id).
        Count(&memberCount)

    // 4. 处理删除逻辑
    if childCount > 0 || memberCount > 0 {
        if !force {
            return errors.New("组织下有子组织或成员，请先迁移或强制删除")
        }

        // 强制删除：级联删除子组织
        if migrateTo != nil {
            // 迁移成员到新组织
            s.db.WithContext(ctx).
                Model(&Member{}).
                Where("organization_id IN (?)", append(getAllDescendantIDs(ctx, id), id)).
                Update("organization_id", migrateTo)

            // 迁移子组织到新父组织
            s.db.WithContext(ctx).
                Model(&Organization{}).
                Where("parent_id = ?", id).
                Update("parent_id", migrateTo)
        }
    }

    // 5. 软删除
    if err := s.db.WithContext(ctx).Delete(&org).Error; err != nil {
        return err
    }

    // 6. 记录审计日志
    s.auditLogService.Create(ctx, &AuditLog{
        TenantID:     tenantID,
        OperatorID:   ctx.Value("user_id").(uint),
        Module:       "organization",
        Action:       "delete",
        ResourceType: "organization",
        ResourceID:   org.ID,
        OldValue:     org,
    })

    return nil
}

// Move 移动组织（调整层级）
func (s *OrganizationService) Move(ctx context.Context, tenantID string, id uint, dto *MoveOrganizationDTO) error {
    // 1. 查询组织
    var org Organization
    if err := s.db.WithContext(ctx).
        Where("id = ? AND tenant_id = ?", id, tenantID).
        First(&org).Error; err != nil {
        return err
    }

    // 2. 查询新父组织
    var newParent Organization
    if err := s.db.WithContext(ctx).
        Where("id = ? AND tenant_id = ?", dto.NewParentID, tenantID).
        First(&newParent).Error; err != nil {
        return errors.New("新父组织不存在")
    }

    // 3. 验证不能移动到自己的子孙组织
    if strings.HasPrefix(newParent.Path, org.Path) {
        return errors.New("不能移动到自己的子孙组织")
    }

    // 4. 计算新的path和level
    newLevel := newParent.Level + 1
    newPath := newParent.Path + "/" + fmt.Sprintf("%d", org.ID)

    // 5. 更新组织和所有子孙组织的path和level
    levelDiff := newLevel - org.Level

    // 更新当前组织
    org.ParentID = &dto.NewParentID
    org.Path = newPath
    org.Level = newLevel
    s.db.WithContext(ctx).Save(&org)

    // 递归更新所有子孙组织
    descendantIDs := getAllDescendantIDs(ctx, id)
    for _, descendantID := range descendantIDs {
        s.db.WithContext(ctx).
            Model(&Organization{}).
            Where("id = ?", descendantID).
            Updates(map[string]interface{}{
                "path":  gorm.Expr("REPLACE(path, ?, ?)", org.Path, newPath),
                "level": gorm.Expr("level + ?", levelDiff),
            })
    }

    return nil
}

// GetAllDescendants 获取所有后代组织ID
func getAllDescendantIDs(ctx context.Context, id uint) []uint {
    var descendants []Organization
    var ids []uint

    s.db.WithContext(ctx).
        Where("parent_id = ?", id).
        Find(&descendants)

    for _, desc := range descendants {
        ids = append(ids, desc.ID)
        ids = append(ids, getAllDescendantIDs(ctx, desc.ID)...)
    }

    return ids
}
```

```go
    return func(ctx context.Context, c *app.RequestContext) {
        // 1. 获取用户ID
        userID := ctx.Value("user_id").(uint)
        if userID == 0 {

            return
        }

        tenantID := c.Param("tenant_id")

        hasPermission, err := m.permissionService.CheckPermission(ctx, userID, tenantID, permission)
        if err != nil {
            c.JSON(500, ErrorResponse(err.Error()))
            c.Abort()
            return
        }

        if !hasPermission {
            c.JSON(403, ErrorResponse("权限不足"))
            c.Abort()
            return
        }

        c.Next(ctx)
    }
}
```

```go
func (s *OrganizationService) GetTree(ctx context.Context, tenantID string) ([]*OrganizationTreeNode, error) {
    var nodes []*OrganizationTreeNode

    query := `
        WITH RECURSIVE org_tree AS (
            -- 根节点
            SELECT id, tenant_id, parent_id, name, type, path, level, 1 as depth
            FROM organizations
            WHERE tenant_id = ? AND parent_id IS NULL AND deleted_at IS NULL

            UNION ALL

            -- 递归子节点
            SELECT o.id, o.tenant_id, o.parent_id, o.name, o.type, o.path, o.level, ot.depth + 1
            FROM organizations o
            INNER JOIN org_tree ot ON o.parent_id = ot.id
            WHERE o.deleted_at IS NULL
        )
        SELECT * FROM org_tree ORDER BY path
    `

    if err := s.db.WithContext(ctx).Raw(query, tenantID).Scan(&nodes).Error; err != nil {
        return nil, err
    }

    return nodes, nil
}
```

```go
// pkg/permission/service/permission_service.go
package service

import (
    "context"
    "fmt"
    "gorm.io/gorm"
)

type PermissionService struct {
    db *gorm.DB
}

// Permission 权限级别
type Permission string

const (
    PermissionView   Permission = "VIEW"
    PermissionEdit   Permission = "EDIT"
    PermissionAdmin  Permission = "ADMIN"
    PermissionDelete Permission = "DELETE"
)

// ResourceType 资源类型
type ResourceType string

const (
    ResourceTypeBot           ResourceType = "BOT"
    ResourceTypeWorkflow      ResourceType = "WORKFLOW"
    ResourceTypeKnowledgeBase ResourceType = "KNOWLEDGE_BASE"
    ResourceTypePlugin        ResourceType = "PLUGIN"
    ResourceTypeDatabase      ResourceType = "DATABASE"
)

// CheckPermission 检查用户是否有资源权限
func (s *PermissionService) CheckPermission(
    ctx context.Context,
    userID int64,
    tenantID string,
    resourceType ResourceType,
    resourceID int64,
    requiredPermission Permission,
) (bool, error) {
    // 1. 检查是否是资源所有者
    var ownerID int64
    err := s.db.WithContext(ctx).
        Table(fmt.Sprintf("%ss", strings.ToLower(string(resourceType)))).
        Select("creator_id").
        Where("id = ? AND tenant_id = ? AND deleted_at IS NULL", resourceID, tenantID).
        Scan(&ownerID).Error

    if err == nil && ownerID == userID {
        // 资源所有者拥有所有权限
        return true, nil
    }

    // 2. 检查团队成员权限
    var count int64
    err = s.db.WithContext(ctx).
        Table("resource_permissions rp").
        Joins("INNER JOIN team_members tm ON tm.team_id = rp.team_id").
        Where("rp.tenant_id = ?", tenantID).
        Where("tm.user_id = ?", userID).
        Where("tm.status = ?", "ACTIVE").
        Where("rp.resource_type = ?", resourceType).
        Where("rp.resource_id = ?", resourceID).
        Where("rp.permission >= ?", requiredPermission). // 假设权限级别为数字: VIEW=1, EDIT=2, ADMIN=3, DELETE=4
        Where("(rp.expires_at IS NULL OR rp.expires_at > NOW())").
        Count(&count).Error

    if err != nil {
        return false, err
    }

    return count > 0, nil
}

// CheckPermissionByTeam 检查团队是否有资源权限
func (s *PermissionService) CheckPermissionByTeam(
    ctx context.Context,
    teamID int64,
    tenantID string,
    resourceType ResourceType,
    resourceID int64,
    requiredPermission Permission,

    var count int64
    err := s.db.WithContext(ctx).
        Table("resource_permissions").
        Where("tenant_id = ?", tenantID).
        Where("team_id = ?", teamID).
        Where("resource_type = ?", resourceType).
        Where("resource_id = ?", resourceID).
        Where("permission >= ?", requiredPermission).
        Where("(expires_at IS NULL OR expires_at > NOW())").
        Count(&count).Error

    if err != nil {
        return false, err
    }

    return count > 0, nil
}

// GrantPermission 授予团队资源权限
func (s *PermissionService) GrantPermission(
    ctx context.Context,
    teamID int64,
    tenantID string,
    resourceType ResourceType,
    resourceID int64,
    permission Permission,
    grantedBy int64,
    expiresIn *time.Duration,
) error {
    perm := &ResourcePermission{
        TenantID:     tenantID,
        TeamID:        teamID,
        ResourceType:  resourceType,
        ResourceID:    resourceID,

        GrantedBy:     grantedBy,
    }

    if expiresIn != nil {

        perm.ExpiresAt = &expiresAt
    }
```

```go
type PermissionMiddleware struct {

// RequirePermission 要求指定权限
func (m *PermissionMiddleware) RequirePermission(
    resourceType service.ResourceType,
    permission service.Permission,
) gin.HandlerFunc {
    return func(c *gin.Context) {
        // 1. 获取用户ID
        userIDVal, exists := c.Get("user_id")
        if !exists {
            c.JSON(http.StatusUnauthorized, gin.H{
                "code":    401,
                "message": "未登录",
            })
            c.Abort()
            return
        }

        userID := userIDVal.(int64)

        // 2. 获取租户ID
        tenantID := c.GetHeader("X-Tenant-ID")
        if tenantID == "" {
            c.JSON(http.StatusBadRequest, gin.H{
                "code":    400,
                "message": "缺少租户ID",
            })
            c.Abort()
            return
        }

        // 3. 获取资源ID
        resourceIDStr := c.Param("id")
        if resourceIDStr == "" {
            c.JSON(http.StatusBadRequest, gin.H{
                "code":    400,
                "message": "缺少资源ID",
            })
            c.Abort()
            return
        }

        resourceID, err := strconv.ParseInt(resourceIDStr, 10, 64)
        if err != nil {
            c.JSON(http.StatusBadRequest, gin.H{
                "code":    400,
                "message": "无效的资源ID",
            })
            c.Abort()
            return
        }

        // 4. 检查权限
        hasPermission, err := m.permService.CheckPermission(
            c.Request.Context(),
            userID,
            tenantID,
            resourceType,
            resourceID,
            permission,
        )

        if err != nil {
            c.JSON(http.StatusInternalServerError, gin.H{
                "code":    500,
                "message": "权限检查失败",
            })
            c.Abort()
            return
        }

        if !hasPermission {
            c.JSON(http.StatusForbidden, gin.H{
                "code":    403,
                "message": "权限不足",
            })
            c.Abort()
            return
        }

        // 5. 权限检查通过,继续处理请求
        c.Next()
    }
}

            "VIEWER": 1,
            "MEMBER": 2,
            "ADMIN":  3,
            "OWNER":  4,
        }

        requiredLevel := getRequiredPermissionLevel(permission)
        userLevel := roleHierarchy[memberRole]

        if userLevel < requiredLevel {
            c.JSON(http.StatusForbidden, gin.H{
                "code":    403,
                "message": "角色权限不足",
            })
            c.Abort()
            return
        }

        c.Next()
    }
}

// getRequiredPermissionLevel 获取权限所需的最低角色级别
func getRequiredPermissionLevel(permission service.Permission) int {
    switch permission {
    case service.PermissionView:
        return 1 // VIEWER及以上
    case service.PermissionEdit:
        return 2 // MEMBER及以上
    case service.PermissionAdmin:
        return 3 // ADMIN及以上
    case service.PermissionDelete:
        return 3 // ADMIN及以上
    default:
        return 4 // 默认OWNER
    }
}
```
