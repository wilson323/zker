## 1. 模块概述

**开发者平台** 是对标 **Dify**、**Langflow** 的开发者生态建设平台，提供**CLI 工具 + SDK + API 文档 + 开发者社区**，降低开发门槛，吸引第三方开发者。

**核心设计理念**：
- ✅ **开发者优先** - 提供完整的开发者工具链
- ✅ **低代码开发** - CLI + SDK + 模板，快速上手
- ✅ **API 优先** - RESTful API + OpenAPI 规范
- ✅ **社区驱动** - 开源策略、插件市场、贡献者激励

**实现策略**：✅ 30% 编码（工具框架） + 70% 内容建设（文档、教程、社区）

**对标产品**: [Dify](https://github.com/langgenius/dify)、[Langflow](https://github.com/logspace-ai/langflow)

### 3.2 CLI 实现

```typescript
// packages/cli/src/commands/bot.ts
import { Command } from 'commander';
import { createProject } from '../utils/project';
import { deployBot } from '../utils/deploy';

export const botCommand = new Command('bot');

// 创建 Bot
botCommand
  .command('create <name>')
  .description('创建新 Bot')
  .option('-t, --template <template>', '模板名称')
  .option('-d, --description <description>', 'Bot 描述')
  .action(async (name, options) => {
    console.log(`🤖 正在创建 Bot: ${name}`);

    const bot = await createProject({
      name,
      template: options.template,
      description: options.description,
    });

    console.log(`✅ Bot 创建成功!`);
    console.log(`📁 项目位置: ${bot.path}`);
    console.log(`\n下一步:`);
    console.log(`  cd ${name}`);
    console.log(`  jika dev`);
  });

// 部署 Bot
botCommand
  .command('deploy')
  .description('部署 Bot 到云端')
  .option('-e, --env <environment>', '环境 (dev|staging|prod)', 'dev')
  .action(async (options) => {
    console.log(`🚀 正在部署 Bot...`);

    const result = await deployBot({
      environment: options.environment,
    });

    console.log(`✅ 部署成功!`);
    console.log(`🔗 Bot URL: ${result.url}`);
  });
```

## 4. SDK 设计

### 4.1 Node.js SDK

```typescript
// packages/sdk/src/index.ts
export class ZekerClient {
  private config: ZekerConfig;
  private http: HttpClient;

  constructor(config: ZekerConfig) {
    this.config = {
      apiKey: config.apiKey,
      baseURL: config.baseURL || 'https://api.zeker.com',
      timeout: config.timeout || 30000,
    };
    this.http = new HttpClient(this.config);
  }

  // Bot 管理
  bots = {
    create: (data: CreateBotDTO) =>
      this.http.post('/api/v1/bots', data),

    get: (id: string) =>
      this.http.get(`/api/v1/bots/${id}`),

    update: (id: string, data: UpdateBotDTO) =>
      this.http.put(`/api/v1/bots/${id}`, data),

    delete: (id: string) =>
      this.http.delete(`/api/v1/bots/${id}`),

    list: (params: ListBotsParams) =>
      this.http.get('/api/v1/bots', { params }),
  };

  // 对话
  conversations = {
    create: (data: CreateConversationDTO) =>
      this.http.post('/api/v1/conversations', data),

    get: (id: string) =>
      this.http.get(`/api/v1/conversations/${id}`),

    sendMessage: async (conversationId: string, message: string) => {
      return this.http.post(`/api/v1/conversations/${conversationId}/messages`, {
        content: message,
        role: 'user',
      });
    },
  };

  // Bot 对话（快捷方式）
  async chat(botId: string, message: string, options?: ChatOptions) {
    // 1. 创建临时会话
    const conversation = await this.conversations.create({
      bot_id: botId,
      user_id: options?.userId,
    });

    // 2. 发送消息
    const response = await this.conversations.sendMessage(
      conversation.id,
      message,
    );

    return response;
  }
}
```

### 4.2 SDK 使用示例

```typescript
// examples/chat.ts
import { ZekerClient } from '@zeker/sdk';

// 初始化客户端
const client = new ZekerClient({
  apiKey: process.env.ZEKER_API_KEY,
});

// 创建 Bot
const bot = await client.bots.create({
  name: '客服助手',
  description: '智能客服助手',
  prompt_template: '你是一个专业的客服助手...',
});

console.log(`Bot 创建成功: ${bot.id}`);

// 对话
const response = await client.chat(bot.id, '你好，帮我查询订单');
console.log(`Bot 回复: ${response.content}`);

// 流式对话
const stream = await client.conversations.streamMessage(conversationId, '你好');
for await (const chunk of stream) {
  process.stdout.write(chunk.content);
}
```

## 5. API 文档平台

### 5.1 OpenAPI 规范

```yaml
# openapi.yaml
openapi: 3.0.0
info:
  title: ZKER API
  version: 1.0.0
  description: 企业级 AI Agent 平台 API
servers:
  - url: https://api.zeker.com
    description: 生产环境
  - url: https://api-staging.zeker.com
    description: 测试环境

paths:
  /api/v1/bots:
    get:
      summary: 列出 Bot
      tags:
        - Bots
      parameters:
        - name: page
          in: query
          schema:
            type: integer
          description: 页码
        - name: page_size
          in: query
          schema:
            type: integer
            default: 20
          description: 每页数量
      responses:
        '200':
          description: 成功
          content:
            application/json:
              schema:
                type: object
                properties:
                  code:
                    type: integer
                  message:
                    type: string
                  data:
                    type: object
                    properties:
                      total:
                        type: integer
                      items:
                        type: array
                        items:
                          $ref: '#/components/schemas/Bot'

    post:
      summary: 创建 Bot
      tags:
        - Bots
      requestBody:
        required: true
        content:
          application/json:
            schema:
              $ref: '#/components/schemas/CreateBotDTO'
      responses:
        '201':
          description: 创建成功
          content:
            application/json:
              schema:
                type: object
                properties:
                  code:
                    type: integer
                  data:
                    $ref: '#/components/schemas/Bot'

components:
  schemas:
    Bot:
      type: object
      properties:
        id:
          type: string
        name:
          type: string
        description:
          type: string
        created_at:
          type: string
          format: date-time

    CreateBotDTO:
      type: object
      required:
        - name
      properties:
        name:
          type: string
        description:
          type: string
        prompt_template:
          type: string
```

## 7. 数据库设计

### 7.1 开发者表 (developers)

```sql
CREATE TABLE developers (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '开发者ID',
    user_id BIGINT NOT NULL COMMENT '用户ID',
    developer_name VARCHAR(128) COMMENT '开发者名称',

    -- GitHub 信息
    github_id VARCHAR(128) COMMENT 'GitHub ID',
    github_url VARCHAR(512) COMMENT 'GitHub 主页',

    -- 统计
    total_commits INT DEFAULT 0 COMMENT '总提交数',
    total_prs INT DEFAULT 0 COMMENT '总 PR 数',
    total_issues_resolved INT DEFAULT 0 COMMENT '总解决问题数',
    reputation_score INT DEFAULT 0 COMMENT '声望分数',
    level ENUM('bronze', 'silver', 'gold', 'platinum') DEFAULT 'bronze' COMMENT '等级',

    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    UNIQUE KEY uk_user_id (user_id),
    INDEX idx_github_id (github_id),
    INDEX idx_level (level)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='开发者表';
```

### 7.2 插件市场表 (plugins_market)

```sql
CREATE TABLE plugins_market (
    id VARCHAR(64) PRIMARY KEY COMMENT '插件ID',
    developer_id BIGINT NOT NULL COMMENT '开发者ID',

    name VARCHAR(128) NOT NULL COMMENT '插件名称',
    description TEXT COMMENT '插件描述',
    category VARCHAR(64) COMMENT '分类',
    version VARCHAR(32) NOT NULL COMMENT '版本号',

    -- 仓库信息
    repository_url VARCHAR(512) NOT NULL COMMENT '仓库地址',
    homepage_url VARCHAR(512) COMMENT '主页地址',

    -- 统计
    downloads INT DEFAULT 0 COMMENT '下载量',
    stars INT DEFAULT 0 COMMENT '点赞数',
    rating DECIMAL(3,2) COMMENT '评分',

    -- 状态
    status ENUM('pending', 'approved', 'rejected') DEFAULT 'pending' COMMENT '审核状态',

    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    INDEX idx_developer_id (developer_id),
    INDEX idx_category (category),
    INDEX idx_status (status),
    INDEX idx_downloads (downloads)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='插件市场表';
```
