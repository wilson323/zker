## 1. 模块概述

### 1.1 功能定义

**智能路由引擎** 是 zker Enterprise AI 系统的"大脑中枢"，负责：
- **理解用户意图** - NLP 意图识别，提取关键实体
- **智能路由决策** - 根据意图、用户画像、历史行为选择最优处理路径
- **动态负载均衡** - 在多个候选服务间智能分配请求
- **持续学习优化** - 基于反馈数据优化路由策略

### 1.2 核心价值

- 🎯 **精准路由** - 95%+ 准确率，直达最优数字员工
- 🎯 **快速响应** - 路由决策 < 200ms，用户无感知
- 🎯 **智能分配** - 负载均衡，避免单点过载
- 🎯 **自我进化** - 持续学习，准确率持续提升

### 1.3 业务场景

| 用户输入 | 识别意图 | 路由目标 | 理由 |
|---------|---------|---------|------|
| "XX产品价格是多少？" | knowledge_query | 问答型数字员工 | 知识查询 |
| "帮我创建一个销售订单" | task_execution | 操作型数字员工 | 任务执行 |
| "上个月销售额" | data_analysis | ChatBI数字员工 | 数据分析 |
| "写一份产品周报" | content_generation | 慧笔模板 | 内容生成 |
| "我的待办事项" | todo_query | 待办中心 | 待办查询 |

### 1.4 技术栈

| 组件 | 技术选型 | 说明 |
|------|----------|------|
| **意图识别** | 通义千问/GPT-4 | LLM意图分类 |
| **实体抽取** | spaCy/HanLP | NLP实体识别 |
| **规则引擎** | Drools/Govaluate | 规则匹配 |
| **向量检索** | Milvus | 相似意图检索 |
| **缓存** | Redis | 意图缓存 |
| **消息队列** | NSQ | 异步处理 |

```go
// routingengine/engine.go
package routingengine

// RoutingEngine 路由引擎
type RoutingEngine struct {
    // 意图识别
    intentRecognizer *IntentRecognizer

    // 实体抽取
    entityExtractor *EntityExtractor

    // 路由决策
    routerDecision *RouterDecision

    // 负载均衡
    loadBalancer *LoadBalancer

    // 学习优化
    learner *Learner

    // 监控
    monitor *Monitor
}

// RouteRequest 路由请求
type RouteRequest struct {
    TenantID    string   // 租户ID
    UserID      string   // 用户ID
    Input       string   // 用户输入
    Context     *Context // 上下文信息（对话历史等）
    UserProfile *UserProfile // 用户画像
}

// RouteResponse 路由响应
type RouteResponse struct {
    Intent      *Intent           // 识别的意图
    Entities    map[string]string // 提取的实体
    Candidates  []*Candidate       // 候选服务列表
    Selected    *SelectedService  // 选中的最优服务
    Confidence  float64           // 决策置信度
    Reason      string            // 路由理由（可解释性）
}

// Route 执行路由
func (e *RoutingEngine) Route(ctx context.Context, req *RouteRequest) (*RouteResponse, error) {
    startTime := time.Now()

    // 1. 预处理
    cleanedInput := preprocess(req.Input)

    // 2. 意图识别
    intent, entities := e.recognizeIntent(ctx, cleanedInput)

    // 3. 获取候选服务
    candidates := e.getCandidates(ctx, req.TenantID, intent)

    // 4. 规则匹配
    candidates = e.applyRules(ctx, candidates, req, intent, entities)

    // 5. 用户画像匹配
    candidates = e.matchUserProfile(ctx, candidates, req.UserProfile)

    // 6. 状态检查
    candidates = e.checkHealth(ctx, candidates)

    // 7. 负载均衡
    selected := e.loadBalancer.Select(ctx, candidates)

    // 8. 记录决策日志
    e.monitor.RecordDecision(ctx, &DecisionLog{
        TenantID:   req.TenantID,
        UserID:     req.UserID,
        Input:      req.Input,
        Intent:     intent,
        Candidates: candidates,
        Selected:   selected,
        Duration:   time.Since(startTime),
    })

    // 9. 返回响应
    return &RouteResponse{
        Intent:     intent,
        Entities:   entities,
        Candidates: candidates,
        Selected:   selected,
        Reason:     e.explainDecision(selected, candidates),
    }, nil
}
```

```go
// routing/intent_recognizer.go
package routing

import (
    "context"
    "encoding/json"
    "fmt"
    "github.com/tmc/langchaingo/llms"
    "github.com/tmc/langchaingo/prompts"
)

type IntentRecognizer struct {
    llm         *llms.ChatModel
    vectorStore *VectorStore
    rules       *RuleEngine
    config      *IntentConfig
}

// RecognizeIntent 识别意图
func (r *IntentRecognizer) RecognizeIntent(ctx context.Context, input string) (*Intent, map[string]string) {
    // 1. 尝试规则匹配（快速路径）
    if r.config.EnableRule {
        if intent, entities, matched := r.rules.Match(input); matched {
            return intent, entities
        }
    }

    // 2. 尝试向量检索（相似意图）
    if r.config.EnableVector {
        if intent, entities, similarity := r.vectorStore.Search(input); similarity > 0.9 {
            return intent, entities
        }
    }

    // 3. LLM意图分类（默认路径）
    if r.config.EnableLLM {
        return r.recognizeByLLM(ctx, input)
    }

    // 4. 兜底：返回默认意图
    return &Intent{
        Type:       IntentSmallTalk,
        Confidence: 0.5,
        Entities:   make(map[string]interface{}),
    }, make(map[string]string)
}

// recognizeByLLM 使用LLM识别意图
func (r *IntentRecognizer) recognizeByLLM(ctx context.Context, input string) (*Intent, map[string]string) {
    // 构建提示词
    prompt := prompts.NewPromptTemplate(
        r.config.LLMPrompt,
        map[string]interface{}{
            "Input": input,
            "Intents": []string{
                IntentKnowledgeQuery,
                IntentTaskExecution,
                IntentDataAnalysis,
                IntentContentGeneration,
                IntentTodoQuery,
                IntentSmallTalk,
            },
        },
    )

    // 调用LLM
    messages := []llms.MessageContent{
        llms.SystemMessage("你是一个专业的意图识别专家"),
        llms.UserMessage(prompt.Format(input)),
    }

    response, err := llms.GenerateFromMessages(ctx, r.llm, messages)
    if err != nil {
        return nil, nil
    }

    // 解析LLM响应
    var result struct {
        Intent     string                 `json:"intent"`
        Confidence float64               `json:"confidence"`
        Entities   map[string]string     `json:"entities"`
    }

    if err := json.Unmarshal([]byte(response), &result); err != nil {
        return nil, nil
    }

    return &Intent{
        Type:       result.Intent,
        Confidence: result.Confidence,
        Entities:   make(map[string]interface{}),
    }, result.Entities
}
```

```go
// routing/entity_extractor.go
package routing

import (
    "context"
    "github.com/go-playground/validator/v10"
    "spacygo"
)

type EntityExtractor struct {
    nlp *spacygo.Language
}

// Entities 实体定义
type Entities struct {
    Person      string `json:"person"`       // 人名
    Organization string `json:"organization"` // 机构名
    Location    string `json:"location"`     // 地点
    Time        string `json:"time"`         // 时间
    Product     string `json:"product"`      // 产品名
    Department  string `json:"department"`   // 部门
    Role        string `json:"role"`          // 岗位
}

// Extract 抽取实体
func (e *EntityExtractor) Extract(ctx context.Context, input string) (*Entities, error) {
    // 使用spaCy进行NLP处理
    doc := e.nlp.Process(input)

    entities := &Entities{}

    for _, ent := range doc.Entities {
        switch ent.Label {
        case "PERSON":
            if entities.Person == "" {
                entities.Person = ent.Text
            }
        case "ORG":
            if entities.Organization == "" {
                entities.Organization = ent.Text
            }
        case "GPE":
            if entities.Location == "" {
                entities.Location = ent.Text
            }
        // ... 其他实体类型
        }
    }

    // 时间实体识别（使用正则）
    if time := extractTime(input); time != "" {
        entities.Time = time
    }

    return entities, nil
}

// extractTime 提取时间实体
func extractTime(input string) string {
    // 定义时间正则
    patterns := []string{
        `\d{4}年\d{1,2}月\d{1,2}日`,           // 2025年1月1日
        `\d{4}-\d{1,2}-\d{1,2}`,             // 2025-01-01
        `\d{1,2}月前`,                       // 1个月前
        `上个[星期周]`,                     // 上星期
        `今年|去年`,                         // 今年、去年
    }

    for _, pattern := range patterns {
        re := regexp.MustCompile(pattern)
        if matches := re.FindString(input); len(matches) > 0 {
            return matches[0]
        }
    }

    return ""
}
```

### 5.1 路由决策算法

```go
// routing/router_decision.go
package routing

type RouterDecision struct {
    ruleEngine    *RuleEngine
    userProfile   *UserProfileService
    historyService *HistoryService
    healthChecker  *HealthChecker
}

// Decide 路由决策
func (d *RouterDecision) Decide(
    ctx context.Context,
    req *RouteRequest,
    intent *Intent,
    candidates []*Candidate,
) ([]*Candidate, error) {

    // 1. 应用路由规则
    candidates = d.ruleEngine.Apply(ctx, candidates, req, intent)

    // 2. 用户画像匹配
    candidates = d.matchByUserProfile(ctx, candidates, req.UserProfile)

    // 3. 历史成功率过滤
    candidates = d.filterBySuccessRate(ctx, candidates, req.UserID)

    // 4. 实时状态检查
    candidates = d.checkRealTimeStatus(ctx, candidates)

    // 5. 评分排序
    candidates = d.scoreAndRank(ctx, candidates, req)

    return candidates, nil
}

// matchByUserProfile 用户画像匹配
func (d *RouterDecision) matchByUserProfile(
    ctx context.Context,
    candidates []*Candidate,
    profile *UserProfile,
) []*Candidate {

    scored := make([]*ScoredCandidate, len(candidates))

    for i, candidate := range candidates {
        score := 0.0

        // 岗位匹配
        if candidate.TargetPositions != nil {
            if contains(candidate.TargetPositions, profile.Position) {
                score += 30
            }
        }

        // 角色匹配
        if candidate.TargetRoles != nil {
            for _, role := range profile.Roles {
                if contains(candidate.TargetRoles, role) {
                    score += 20
                }
            }
        }

        // 历史偏好
        historyScore := d.historyService.GetPreferenceScore(
            ctx,
            profile.UserID,
            candidate.ServiceID,
        )
        score += historyScore * 20

        // 技能匹配度
        if candidate.RequiredSkills != nil {
            skillMatchRate := calculateSkillMatchRate(
                profile.Skills,
                candidate.RequiredSkills,
            )
            score += skillMatchRate * 30
        }

        scored[i] = &ScoredCandidate{
            Candidate: candidate,
            Score:     score,
        }
    }

    // 按评分排序
    sort.Slice(scored, func(i, j int) bool {
        return scored[i].Score > scored[j].Score
    })

    // 转换回Candidate数组
    result := make([]*Candidate, len(scored))
    for i, sc := range scored {
        result[i] = sc.Candidate
    }

    return result
}

// scoreAndRank 评分排序
func (d *RouterDecision) scoreAndRank(
    ctx context.Context,
    candidates []*Candidate,
    req *RouteRequest,
) []*Candidate {

    for _, candidate := range candidates {
        score := 0.0
        reasons := []string{}

        // 因素1: 用户画像匹配度 (30%)
        userMatchScore := calculateUserMatchScore(candidate, req.UserProfile)
        score += userMatchScore * 0.3
        reasons = append(reasons, fmt.Sprintf("用户匹配度: %.2f", userMatchScore))

        // 因素2: 历史成功率 (25%)
        successRate := d.historyService.GetSuccessRate(ctx, req.UserID, candidate.ServiceID)
        score += successRate * 0.25
        reasons = append(reasons, fmt.Sprintf("历史成功率: %.2f", successRate))

        // 因素3: 当前负载 (20%)
        loadScore := 1.0 - candidate.CurrentLoad
        score += loadScore * 0.2
        reasons = append(reasons, fmt.Sprintf("负载评分: %.2f", loadScore))

        // 因素4: 平均响应时间 (15%)
        responseTimeScore := calculateResponseTimeScore(candidate.AvgResponseTime)
        score += responseTimeScore * 0.15
        reasons = append(reasons, fmt.Sprintf("响应时间评分: %.2f", responseTimeScore))

        // 因素5: 服务可用性 (10%)
        availabilityScore := candidate.Availability
        score += availabilityScore * 0.1
        reasons = append(reasons, fmt.Sprintf("可用性评分: %.2f", availabilityScore))

        candidate.Score = score
        candidate.Reasons = reasons
    }

    // 按评分排序
    sort.Slice(candidates, func(i, j int) bool {
        return candidates[i].Score > candidates[j].Score
    })

    return candidates
}
```

```go
// Condition 条件
type Condition struct {
    Intent      []string `json:"intent"`       // 意图匹配
    Keyword     []string `json:"keyword"`      // 关键词匹配
    UserRoles   []string `json:"user_roles"`   // 用户角色
    Departments []string `json:"departments"` // 用户部门
    TimeRange   *TimeRange `json:"time_range"`  // 时间范围
    CustomExpr  string   `json:"custom_expr"`  // 自定义表达式
}

// Action 动作
type Action struct {
    TargetServiceID string  `json:"target_service_id"` // 指定服务ID
    FilterType      string  `json:"filter_type"`       // 筛选类型
    FilterValue     string  `json:"filter_value"`      // 筛选值
    Priority        int     `json:"priority"`           // 优先级调整
}

// Apply 应用规则
func (r *RuleEngine) Apply(
    ctx context.Context,
    candidates []*Candidate,
    req *RouteRequest,
    intent *Intent,
) []*Candidate {

    // 按优先级排序规则
    sort.Slice(r.rules, func(i, j int) bool {
        return r.rules[i].Priority < r.rules[j].Priority
    })

    // 应用每条规则
    for _, rule := range r.rules {
        if r.matchCondition(rule.Condition, req, intent) {
            candidates = r.applyAction(rule.Action, candidates)
            break // 规则匹配后不再继续
        }
    }

    return candidates
}

// matchCondition 检查条件是否匹配
func (r *RuleEngine) matchCondition(cond *Condition, req *RouteRequest, intent *Intent) bool {
    // 1. 意图匹配
    if len(cond.Intent) > 0 {
        if !contains(cond.Intent, intent.Type) {
            return false
        }
    }

    // 2. 关键词匹配
    if len(cond.Keyword) > 0 {
        inputLower := strings.ToLower(req.Input)
        matched := false
        for _, keyword := range cond.Keyword {
            if strings.Contains(inputLower, strings.ToLower(keyword)) {
                matched = true
                break
            }
        }
        if !matched {
            return false
        }
    }

    // 3. 用户角色匹配
    if len(cond.UserRoles) > 0 {
        profile := req.UserProfile
        if !hasAnyRole(profile.Roles, cond.UserRoles) {
            return false
        }
    }

    // 4. 部门匹配
    if len(cond.Departments) > 0 {
        profile := req.UserProfile
        if !contains(cond.Departments, profile.Department) {
            return false
        }
    }

    // 5. 时间范围匹配
    if cond.TimeRange != nil {
        now := time.Now()
        if now.Before(cond.TimeRange.Start) || now.After(cond.TimeRange.End) {
            return false
        }
    }

    // 6. 自定义表达式（支持简单的Lua表达式）
    if cond.CustomExpr != "" {
        if !r.evaluateCustomExpr(cond.CustomExpr, req, intent) {
            return false
        }
    }

    return true
}
```

```go
// routing/load_balancer.go
package routing

type LoadBalancer struct {
    strategy LoadBalanceStrategy
}

type LoadBalanceStrategy int

const (
    RoundRobin LoadBalanceStrategy = iota
    WeightedRoundRobin
    LeastConnections
    ResponseTimePriority
    StickySession
)

// Select 选择最优服务
func (lb *LoadBalancer) Select(
    ctx context.Context,
    candidates []*Candidate,
) *SelectedService {

    switch lb.strategy {
    case RoundRobin:
        return lb.roundRobinSelect(ctx, candidates)

    case WeightedRoundRobin:
        return lb.weightedRoundRobinSelect(ctx, candidates)

    case LeastConnections:
        return lb.leastConnectionsSelect(ctx, candidates)

    case ResponseTimePriority:
        return lb.responseTimePrioritySelect(ctx, candidates)

    case StickySession:
        return lb.stickySessionSelect(ctx, candidates)

    default:
        return lb.roundRobinSelect(ctx, candidates)
    }
}

// roundRobinSelect 轮询选择
func (lb *LoadBalancer) roundRobinSelect(
    ctx context.Context,
    candidates []*Candidate,
) *SelectedService {
    // 原子性递增计数器
    counter := lb.getCounter(ctx, "round_robin")
    index := int(counter % uint64(len(candidates)))
    counter++

    lb.setCounter(ctx, "round_robin", counter)

    return &SelectedService{
        ServiceID:   candidates[index].ServiceID,
        ServiceName: candidates[index].ServiceName,
        Reason:      "轮询调度",
    }
}

// weightedRoundRobinSelect 加权轮询
func (lb *LoadBalancer) weightedRoundRobinSelect(
    ctx context.Context,
    candidates []*Candidate,
) *SelectedService {
    // 构建加权列表
    weighted := []struct {
        service *Candidate
        weight  int
    }{}

    for _, candidate := range candidates {
        weighted = append(weighted, struct {
            service: candidate,
            weight:  candidate.Weight,
        })
    }

    // 计算总权重
    totalWeight := 0
    for _, w := range weighted {
        totalWeight += w.weight
    }

    // 随机选择
    rand := lb.rand.Intn(totalWeight)
    currentWeight := 0

    for _, w := range weighted {
        currentWeight += w.weight
        if rand < currentWeight {
            return &SelectedService{
                ServiceID:   w.service.ServiceID,
                ServiceName: w.service.ServiceName,
                Reason:      fmt.Sprintf("加权轮询 (权重: %d)", w.weight),
            }
        }
    }

    // 兜底：返回第一个
    return &SelectedService{
        ServiceID:   weighted[0].service.ServiceID,
        ServiceName: weighted[0].service.ServiceName,
        Reason:      "加权轮询 (兜底)",
    }
}

// leastConnectionsSelect 最少连接选择
func (lb *LoadBalancer) leastConnectionsSelect(
    ctx context.Context,
    candidates []*Candidate,
) *SelectedService {
    var selected *Candidate
    minConnections := int(^uint(0) >> 1)

    for _, candidate := range candidates {
        if candidate.CurrentConnections < minConnections {
            minConnections = candidate.CurrentConnections
            selected = candidate
        }
    }

    return &SelectedService{
        ServiceID:   selected.ServiceID,
        ServiceName: selected.ServiceName,
        Reason:      fmt.Sprintf("最少连接 (连接数: %d)", minConnections),
    }
}

// responseTimePrioritySelect 响应时间优先
func (lb *LoadBalancer) responseTimePrioritySelect(
    ctx context.Context,
    candidates []*Candidate,
) *SelectedService {
    var selected *Candidate
    minResponseTime := time.Duration(^uint64(0) >> 1)

    for _, candidate := range candidates {
        if candidate.AvgResponseTime < minResponseTime {
            minResponseTime = candidate.AvgResponseTime
            selected = candidate
        }
    }

    return &SelectedService{
        ServiceID:   selected.ServiceID,
        ServiceName: selected.ServiceName,
        Reason:      fmt.Sprintf("响应时间优先 (响应时间: %v)", minResponseTime),
    }
}

// stickySessionSelect 粘性会话
func (lb *LoadBalancer) stickySessionSelect(
    ctx context.Context,
    candidates []*Candidate,
) *SelectedService {
    // 尝试从会话中获取上次选择的服务
    sessionID := ctx.Value("session_id").(string)
    if sessionID != "" {
        lastServiceID := lb.getLastSelected(ctx, sessionID)
        if lastServiceID != "" {
            // 查找该服务是否在候选列表中
            for _, candidate := range candidates {
                if candidate.ServiceID == lastServiceID {
                    return &SelectedService{
                        ServiceID:   candidate.ServiceID,
                        ServiceName: candidate.ServiceName,
                        Reason:      "粘性会话 (复用上次服务)",
                    }
                }
            }
        }
    }

    // 如果没有找到，使用轮询
    return lb.roundRobinSelect(ctx, candidates)
}
```

```go
func (o *ModelOptimizer) calculateSuccessRateByIntent(feedbacks []*RoutingFeedback) map[string]float64 {
    stats := make(map[string]*IntentStats)

    for _, feedback := range feedbacks {
        if _, ok := stats[feedback.Intent]; !ok {
            stats[feedback.Intent] = &IntentStats{}
        }
        stats[feedback.Intent].Total++
        if feedback.Success {
            stats[feedback.Intent].Success++
        }
    }

    result := make(map[string]float64)
    for intent, stat := range stats {
        result[intent] = float64(stat.Success) / float64(stat.Total)
    }

    return result
}
```

```go
func (m *RoutingMonitor) calculateIntentAccuracy(ctx context.Context, tenantID string, timeRange time.Duration) float64 {
    query := `
        SELECT
            COUNT(CASE WHEN predicted_intent = actual_intent THEN 1 END) * 1.0 / COUNT(*) as accuracy
        FROM routing_feedback
        WHERE tenant_id = ? AND created_at >= now() - ?
    `

    var accuracy float64
    err := m.metricsDB.Query(ctx, query, tenantID, timeRange).Scan(&accuracy)
    if err != nil {
        return 0
    }

    return accuracy
}

// checkAlerts 检查告警
func (m *RoutingMonitor) checkAlerts(ctx context.Context, tenantID string, metrics *RoutingMetrics) error {
    alerts := []Alert{}

    // 告警1: 意图识别准确率低于90%
    if metrics.IntentAccuracy < 0.9 {
        alerts = append(alerts, Alert{
            Type:        "low_intent_accuracy",
            Severity:    "warning",
            Message:     fmt.Sprintf("意图识别准确率过低: %.2f%% (阈值: 90%%)", metrics.IntentAccuracy*100),
            TenantID:    tenantID,
        })
    }

    // 告警2: 路由成功率低于95%
    if metrics.RoutingSuccessRate < 0.95 {
        alerts = append(alerts, Alert{
            Type:        "low_routing_success",
            Severity:    "critical",
            Message:     fmt.Sprintf("路由成功率过低: %.2f%% (阈值: 95%%)", metrics.RoutingSuccessRate*100),
            TenantID:    tenantID,
        })
    }

    // 告警3: 平均响应时间超过1秒
    if metrics.AvgResponseTime > 1000 {
        alerts = append(alerts, Alert{
            Type:        "high_response_time",
            Severity:    "warning",
            Message:     fmt.Sprintf("平均响应时间过长: %.0fms (阈值: 1000ms)", metrics.AvgResponseTime),
            TenantID:    tenantID,
        })
    }

    // 发送告警
    for _, alert := range alerts {
        m.alertService.SendAlert(ctx, alert)
    }

    return nil
}
```
