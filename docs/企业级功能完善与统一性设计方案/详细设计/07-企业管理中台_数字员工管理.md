## 1. 模块概述

### 1.1 功能定义

**数字员工管理** 是ZKER企业AI智能体的统一管理平台,提供:
- 数字员工(Bot)的创建与配置
- 三种Bot类型支持(问答型/操作型/综合型)
- 技能与知识库的关联配置
- Bot发布上架与版本管理
- Bot使用监控与数据分析
- Bot权限控制与发布范围

### 1.2 核心价值

- 🎯 **统一管理** - 企业所有数字员工集中管理,统一配置
- 🎯 **灵活扩展** - 支持三种类型Bot,满足不同业务场景
- 🎯 **快速部署** - 可视化配置,快速创建发布Bot
- 🎯 **数据驱动** - 完整的使用监控与效果分析

### 1.3 业务术语

| 术语 | 英文 | 定义 |
|------|------|------|
| **数字员工** | Bot / Digital Employee | AI智能体,可执行特定任务的虚拟员工 |
| **问答型Bot** | Q&A Bot | 基于知识库回答问题的Bot |
| **操作型Bot** | Action Bot | 调用技能/插件执行操作的Bot |
| **综合型Bot** | Comprehensive Bot | 同时具备问答和操作能力的Bot |
| **技能** | Skill | Bot可执行的具体能力,如查询天气、发送邮件 |
| **知识库** | Knowledge Base | Bot的知识来源,如文档、FAQ |
| **发布** | Publish | 将Bot从草稿状态变为可用状态 |
| **上架** | Listing | 将Bot发布到"发现"页,供所有用户使用 |

### 1.4 目标用户

| 用户类型 | 典型角色 | 核心需求 |
|---------|---------|----------|
| **企业管理员** | CEO、CTO | 管理所有Bot,查看全局数据 |
| **部门经理** | HR总监、研发总监 | 管理本部门Bot,配置权限 |
| **开发者** | AI工程师 | 创建配置Bot,测试调试 |
| **普通用户** | 企业员工 | 使用已发布的Bot |

### 2.2 Bot类型详解

#### 2.2.1 问答型Bot (Q&A Bot)

**定义**: 基于知识库回答问题的Bot

**典型场景**:
- 企业规章制度问答
- 产品知识问答
- FAQ问答
- 培训资料问答

**核心能力**:
- 知识库检索(RAG)
- 自然语言理解
- 答案生成
- 答案溯源

**配置要求**:
- 必须关联至少1个知识库
- 不需要配置技能

#### 2.2.2 操作型Bot (Action Bot)

**定义**: 调用技能/插件执行操作的Bot

**典型场景**:
- 查询数据(天气、股票、物流)
- 发送通知(邮件、短信、钉钉)
- 调用API(CRM、ERP、OA系统)
- 生成文档(合同、报告)

**核心能力**:
- 意图识别
- 实体提取
- 技能调用
- 参数填充

**配置要求**:
- 必须关联至少1个技能
- 不需要知识库(可选)

#### 2.2.3 综合型Bot (Comprehensive Bot)

**定义**: 同时具备问答和操作能力的Bot

**典型场景**:
- 智能客服(回答问题+办理业务)
- 助理Bot(查询信息+执行操作)
- 协作Bot(知识问答+流程审批)

**核心能力**:
- 知识库检索
- 技能调用
- 任务编排
- 上下文管理

**配置要求**:
- 至少关联1个知识库或1个技能
- 推荐同时配置

  type: BotType

  welcomeMessage?: string          // 开场白

  maxTokens?: number              // 最大Token数

  // 状态

  isListed: boolean               // 是否上架到发现页
  publishedAt?: Date
  publishedBy?: string

  // 统计
  usageCount: number              // 使用次数
  activeUsers: number             // 活跃用户数

  createdAt: Date
  updatedAt: Date

  // 关联
  skills: BotSkill[]
  knowledgeBases: BotKnowledge[]
  versions: BotVersion[]

  // 方法
  publish(): void
  unpublish(): void
  list(): void
  unlist(): void
  addSkill(skillId: string): void
  removeSkill(skillId: string): void
  addKnowledgeBase(kbId: string): void
  removeKnowledgeBase(kbId: string): void
  createVersion(): BotVersion
  updateStats(stats: BotStatistics): void
}

// BotSkill (Bot技能关联实体)
class BotSkill {
  id: string
  botId: string

}

// BotKnowledge (Bot知识库关联实体)
class BotKnowledge {
  id: string
  botId: string

  id: string

  description?: string

class BotStatistics {

### 4.2 表结构设计

#### 4.2.1 数字员工表 (bots)

```sql
CREATE TABLE bots (
    id VARCHAR(64) PRIMARY KEY COMMENT 'Bot ID',
    tenant_id VARCHAR(64) NOT NULL COMMENT '租户ID',
    creator_id BIGINT NOT NULL COMMENT '创建者ID',

    -- 基本信息
    name VARCHAR(100) NOT NULL COMMENT 'Bot名称',
    description VARCHAR(500) COMMENT 'Bot描述',
    avatar VARCHAR(512) COMMENT 'Bot头像URL',

    -- Bot类型: qa(问答型) | action(操作型) | comprehensive(综合型)
    type ENUM('qa', 'action', 'comprehensive') NOT NULL COMMENT 'Bot类型',

    -- 配置
    prompt TEXT COMMENT '系统提示词',
    welcome_message VARCHAR(500) COMMENT '开场白',
    conversation_style JSON COMMENT '对话风格配置',

    -- 高级配置
    temperature DECIMAL(3,2) DEFAULT 0.7 COMMENT '温度值(0-1)',
    max_tokens INT DEFAULT 2000 COMMENT '最大Token数',
    stream_output BOOLEAN DEFAULT TRUE COMMENT '是否流式输出',

    -- 状态
    status ENUM('draft', 'published', 'archived') DEFAULT 'draft' COMMENT '状态',
    is_listed BOOLEAN DEFAULT FALSE COMMENT '是否上架到发现页',
    published_at DATETIME COMMENT '发布时间',
    published_by BIGINT COMMENT '发布者ID',

    -- 统计
    usage_count INT DEFAULT 0 COMMENT '使用次数',
    active_users INT DEFAULT 0 COMMENT '活跃用户数',

    -- 时间戳
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    deleted_at DATETIME DEFAULT NULL COMMENT '删除时间',

    UNIQUE KEY uk_tenant_name (tenant_id, name),
    INDEX idx_tenant_id (tenant_id),
    INDEX idx_creator_id (creator_id),
    INDEX idx_type (type),
    INDEX idx_status (status),
    INDEX idx_is_listed (is_listed),
    INDEX idx_deleted_at (deleted_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='数字员工表';
```

#### 4.2.2 Bot技能关联表 (bot_skills)

### 4.3 索引优化策略

#### 4.3.1 Bot列表查询优化

```sql
-- 常见查询1: 获取租户下的Bot列表
SELECT * FROM bots
WHERE tenant_id = ? AND deleted_at IS NULL
ORDER BY updated_at DESC
LIMIT 20;

-- 索引: idx_tenant_id + idx_updated_at

-- 常见查询2: 按类型筛选Bot
SELECT * FROM bots
WHERE tenant_id = ? AND type = ? AND deleted_at IS NULL;

-- 索引: idx_tenant_id + idx_type

-- 常见查询3: 搜索Bot(名称/描述)
SELECT * FROM bots
WHERE tenant_id = ?
  AND deleted_at IS NULL
  AND (name LIKE ? OR description LIKE ?)
LIMIT 20;

-- 索引: idx_tenant_id + FULLTEXT INDEX ft_name_desc (name, description)
```

#### 4.3.2 Bot统计查询优化

```sql
-- 常见查询1: 获取Bot最近7天的统计数据
SELECT * FROM bot_statistics
WHERE bot_id = ?
  AND stat_date >= DATE_SUB(CURDATE(), INTERVAL 7 DAY)
ORDER BY stat_date DESC;

-- 索引: idx_bot_id + idx_stat_date

-- 常见查询2: 获取租户下所有Bot的汇总统计
SELECT
    bot_id,
    SUM(total_conversations) as total_conversations,
    SUM(total_messages) as total_messages,
    AVG(avg_response_time) as avg_response_time
FROM bot_statistics
WHERE tenant_id = ? AND stat_date >= ?
GROUP BY bot_id;

-- 索引: idx_tenant_id + idx_stat_date
```

// src/hooks/useBot.ts
import { useState, useEffect } from 'react';
import { botApi } from '@/api/bot';

export function useBot() {
  const [bots, setBots] = useState<Bot[]>([]);
  const [loading, setLoading] = useState(false);
  const [pagination, setPagination] = useState({
    current: 1,
    pageSize: 20,
    total: 0,
  });

  const fetchBots = async (params: FetchBotsParams = {}) => {
    setLoading(true);
    try {
      const res = await botApi.list({
        page: pagination.current,
        page_size: pagination.pageSize,
        ...params,
      });
      setBots(res.data.items);
      setPagination({
        ...pagination,
        total: res.data.total,
      });
    } catch (error) {
      console.error('Failed to fetch bots:', error);
    } finally {
      setLoading(false);
    }
  };

  const createBot = async (data: CreateBotDTO) => {
    return await botApi.create(data);
  };

  const updateBot = async (id: string, data: UpdateBotDTO) => {
    return await botApi.update(id, data);
  };

  const deleteBot = async (id: string) => {
    return await botApi.delete(id);
  };

  const publishBot = async (id: string, createVersion: boolean, versionDesc?: string) => {
    return await botApi.publish(id, { create_version: createVersion, version_description: versionDesc });
  };

  const unpublishBot = async (id: string) => {
    return await botApi.unpublish(id);
  };

  const listBot = async (id: string) => {
    return await botApi.list(id);
  };

  const unlistBot = async (id: string) => {
    return await botApi.unlist(id);
  };

  const copyBot = async (id: string, data: CopyBotDTO) => {

    pagination,
    fetchBots,
    createBot,

    publishBot,

// List 获取Bot列表
func (s *BotService) List(ctx context.Context, tenantID string, params *ListParams) (*ListResult, error) {
    var bots []Bot
    var total int64

    query := s.db.WithContext(ctx).
        Where("tenant_id = ?", tenantID).
        Where("deleted_at IS NULL")

    // 类型筛选
    if params.Type != "" {
        query = query.Where("type = ?", params.Type)
    }

    // 状态筛选
    if params.Status != "" {
        query = query.Where("status = ?", params.Status)
    }

    // 关键词搜索
    if params.Keyword != "" {
        query = query.Where("name LIKE ? OR description LIKE ?",
            "%"+params.Keyword+"%", "%"+params.Keyword+"%")
    }

    // 创建者筛选
    if params.CreatorID != "" {
        query = query.Where("creator_id = ?", params.CreatorID)
    }

    // 统计总数
    query.Model(&Bot{}).Count(&total)

    // 分页
    if params.PageSize > 0 && params.Page > 0 {
        offset := (params.Page - 1) * params.PageSize
        query = query.Offset(offset).Limit(params.PageSize)
    }

    // 排序
    query = query.Order("updated_at DESC")

    // 预加载关联
    query = query.
        Preload("Skills", func(db *gorm.DB) *gorm.DB {
            return db.Order("sort_order ASC")
        }).
        Preload("KnowledgeBases", func(db *gorm.DB) *gorm.DB {
            return db.Order("priority DESC")
        })

    // 查询
    if err := query.Find(&bots).Error; err != nil {
        return nil, err
    }

    return &ListResult{
        Total: total,
        Items: bots,
    }, nil
}

// Create 创建Bot
func (s *BotService) Create(ctx context.Context, tenantID string, creatorID uint, dto *CreateBotDTO) (*Bot, error) {
    // 1. 验证Bot名称唯一性
    var existingBot Bot
    err := s.db.WithContext(ctx).
        Where("tenant_id = ? AND name = ?", tenantID, dto.Name).
        First(&existingBot).Error
    if err == nil {
        return nil, errors.New("Bot名称已存在")
    }
    if err != gorm.ErrRecordNotFound {
        return nil, err
    }

    // 2. 根据Bot类型验证配置
    if dto.Type == "qa" && len(dto.KnowledgeBases) == 0 {
        return nil, errors.New("问答型Bot必须关联至少1个知识库")
    }
    if dto.Type == "action" && len(dto.Skills) == 0 {
        return nil, errors.New("操作型Bot必须关联至少1个技能")
    }
    if dto.Type == "comprehensive" && len(dto.Skills) == 0 && len(dto.KnowledgeBases) == 0 {
        return nil, errors.New("综合型Bot至少需要关联1个技能或知识库")
    }

    // 3. 验证技能存在
    for _, skillDTO := range dto.Skills {
        _, err := s.skillService.GetByID(ctx, skillDTO.SkillID)
        if err != nil {
            return nil, fmt.Errorf("技能不存在: %s", skillDTO.SkillID)
        }
    }

    // 4. 验证知识库存在
    for _, kbDTO := range dto.KnowledgeBases {
        kb, err := s.knowledgeBaseService.GetByID(ctx, kbDTO.KnowledgeBaseID)
        if err != nil || kb.TenantID != tenantID {
            return nil, fmt.Errorf("知识库不存在: %s", kbDTO.KnowledgeBaseID)
        }
    }

    // 5. 生成Bot ID
    botID := fmt.Sprintf("bot-%s", uuid.New().String())

    // 6. 创建Bot
    bot := &Bot{
        ID:              botID,
        TenantID:        tenantID,
        CreatorID:       creatorID,
        Name:            dto.Name,
        Description:     dto.Description,
        Avatar:          dto.Avatar,
        Type:            dto.Type,
        Prompt:          dto.Prompt,
        WelcomeMessage:  dto.WelcomeMessage,
        Temperature:     0.7,
        MaxTokens:       2000,
        StreamOutput:    true,
        Status:          "draft",
        IsListed:        false,
    }

    if dto.Temperature != nil {
        bot.Temperature = *dto.Temperature
    }
    if dto.MaxTokens != nil {
        bot.MaxTokens = *dto.MaxTokens
    }
    if dto.StreamOutput != nil {
        bot.StreamOutput = *dto.StreamOutput
    }

    if dto.ConversationStyle != nil {
        styleJSON, _ := json.Marshal(dto.ConversationStyle)
        conversationStyleStr := string(styleJSON)
        bot.ConversationStyle = &conversationStyleStr
    }

    if err := s.db.WithContext(ctx).Create(bot).Error; err != nil {
        return nil, err
    }

    // 7. 关联技能
    for _, skillDTO := range dto.Skills {
        botSkill := &BotSkill{
            BotID:     botID,
            SkillID:   skillDTO.SkillID,
            SortOrder: skillDTO.SortOrder,
            IsEnabled: skillDTO.IsEnabled,
        }
        if skillDTO.Config != nil {
            configJSON, _ := json.Marshal(skillDTO.Config)
            botSkill.Config = string(configJSON)
        }
        s.db.WithContext(ctx).Create(botSkill)
    }

    // 8. 关联知识库
    for _, kbDTO := range dto.KnowledgeBases {
        botKB := &BotKnowledgeBase{
            BotID:           botID,
            KnowledgeBaseID: kbDTO.KnowledgeBaseID,
            Priority:        kbDTO.Priority,
            IsEnabled:       kbDTO.IsEnabled,
        }
        s.db.WithContext(ctx).Create(botKB)
    }

    // 9. 记录审计日志
    s.auditLogService.Create(ctx, &AuditLog{
        TenantID:     tenantID,
        OperatorID:   creatorID,
        Module:       "bot",
        Action:       "create",
        ResourceType: "bot",
        ResourceID:   botID,
        NewValue:     bot,
    })

    return bot, nil
}

// Publish 发布Bot
func (s *BotService) Publish(ctx context.Context, tenantID string, userID uint, botID string, createVersion bool, versionDesc string) error {
    // 1. 查询Bot
    var bot Bot
    if err := s.db.WithContext(ctx).
        Where("id = ? AND tenant_id = ?", botID, tenantID).
        First(&bot).Error; err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return errors.New("Bot不存在")
        }
        return err
    }

    // 2. 检查Bot状态
    if bot.Status == "published" {
        return errors.New("Bot已发布")
    }

    // 3. 验证Bot配置完整
    if bot.Type == "qa" {
        var kbCount int64
        s.db.WithContext(ctx).Model(&BotKnowledgeBase{}).Where("bot_id = ?", botID).Count(&kbCount)
        if kbCount == 0 {
            return errors.New("问答型Bot必须关联至少1个知识库")
        }
    }
    if bot.Type == "action" {
        var skillCount int64
        s.db.WithContext(ctx).Model(&BotSkill{}).Where("bot_id = ?", botID).Count(&skillCount)
        if skillCount == 0 {
            return errors.New("操作型Bot必须关联至少1个技能")
        }
    }

    // 4. 更新Bot状态
    bot.Status = "published"
    now := time.Now()
    bot.PublishedAt = &now
    bot.PublishedBy = &userID
    if err := s.db.WithContext(ctx).Save(&bot).Error; err != nil {
        return err
    }

    // 5. 创建版本(可选)
    if createVersion {
        _, err := s.versionService.Create(ctx, tenantID, userID, botID, versionDesc)
        if err != nil {
            return err
        }
    }

    // 6. 清除Bot缓存
    s.invalidateCache(ctx, botID)

    // 7. 记录审计日志
    s.auditLogService.Create(ctx, &AuditLog{
        TenantID:     tenantID,
        OperatorID:   userID,
        Module:       "bot",
        Action:       "publish",
        ResourceType: "bot",
        ResourceID:   botID,
    })

    return nil
}

// Copy 复制Bot
func (s *BotService) Copy(ctx context.Context, tenantID string, userID uint, botID string, dto *CopyBotDTO) (*Bot, error) {
    // 1. 查询原Bot
    var originalBot Bot
    if err := s.db.WithContext(ctx).
        Where("id = ? AND tenant_id = ?", botID, tenantID).
        Preload("Skills").
        Preload("KnowledgeBases").
        First(&originalBot).Error; err != nil {
        return nil, err
    }

    // 2. 创建副本
    newBotID := fmt.Sprintf("bot-%s", uuid.New().String())
    newBot := &Bot{
        ID:             newBotID,
        TenantID:       tenantID,
        CreatorID:      userID,
        Name:           dto.Name,
        Description:    originalBot.Description,
        Avatar:         originalBot.Avatar,
        Type:           originalBot.Type,
        Prompt:         originalBot.Prompt,
        WelcomeMessage: originalBot.WelcomeMessage,
        ConversationStyle: originalBot.ConversationStyle,
        Temperature:    originalBot.Temperature,
        MaxTokens:      originalBot.MaxTokens,
        StreamOutput:   originalBot.StreamOutput,
        Status:         "draft",
        IsListed:       false,
    }

    if err := s.db.WithContext(ctx).Create(newBot).Error; err != nil {
        return nil, err
    }

    // 3. 复制技能关联
    if dto.CopySkills {
        for _, skill := range originalBot.Skills {
            newSkill := &BotSkill{
                BotID:     newBotID,
                SkillID:   skill.SkillID,
                SortOrder: skill.SortOrder,
                IsEnabled: skill.IsEnabled,
                Config:    skill.Config,
            }
            s.db.WithContext(ctx).Create(newSkill)
        }
    }

    // 4. 复制知识库关联
    if dto.CopyKnowledgeBases {

                KnowledgeBaseID: kb.KnowledgeBaseID,
                Priority:        kb.Priority,
                IsEnabled:       kb.IsEnabled,
            }
            s.db.WithContext(ctx).Create(newKB)
        }
    }

    return newBot, nil
}

    ragEngine *RAGEngine
    skillCaller *SkillCaller
}

func (o *TaskOrchestrator) Orchestrate(ctx context.Context, input string, bot *Bot) (*OrchestrationResult, error) {
    // 1. 意图识别
    intent := o.recognizeIntent(ctx, input)

    // 2. 根据意图执行不同流程
    switch intent.Type {
    case "knowledge_query":
        // 知识查询流程
        return o.ragEngine.Query(ctx, input, bot)

    case "action_execution":
        // 操作执行流程
        return o.skillCaller.Execute(ctx, intent.SkillID, intent.Parameters)

    case "hybrid":
        // 混合流程:先RAG,再执行操作
        ragResult, _ := o.ragEngine.Query(ctx, input, bot)

        // 从RAG结果中提取参数
        params := o.extractParams(ragResult)

        // 执行操作
        actionResult, _ := o.skillCaller.Execute(ctx, intent.SkillID, params)

        // 融合结果
        return o.mergeResults(ragResult, actionResult)

    default:
        // 通用对话
        return o.generalChat(ctx, input, bot)
    }
}

func (s *BotVersionService) Create(ctx context.Context, tenantID string, userID uint, botID string, description string) (*BotVersion, error) {
    // 1. 查询Bot完整配置
    var bot Bot
    if err := s.db.WithContext(ctx).

        Preload("KnowledgeBases").

    if err != nil {
        return nil, err
    }

            Avatar:      bot.Avatar,
        },

        Prompt: bot.Prompt,
        ConversationStyle: bot.ConversationStyle,
        Temperature: bot.Temperature,
        MaxTokens: bot.MaxTokens,
    }

    for _, skill := range bot.Skills {
        config.Skills = append(config.Skills, SkillConfig{
            SkillID:   skill.SkillID,
            IsEnabled: skill.IsEnabled,
            Config:    skill.Config,
        })
    }

    for _, kb := range bot.KnowledgeBases {
        config.KnowledgeBases = append(config.KnowledgeBases, KnowledgeBaseConfig{
            KnowledgeBaseID: kb.KnowledgeBaseID,
            Priority:        kb.Priority,
            IsEnabled:       kb.IsEnabled,
        })
    }

    configJSON, _ := json.Marshal(config)

    // 4. 创建版本记录
    version := &BotVersion{
        ID:             fmt.Sprintf("version-%s", uuid.New().String()),
        BotID:          botID,
        VersionNumber:  versionNumber,

        Config:         string(configJSON),
        CreatedBy:      userID,
    }

    if err := s.db.WithContext(ctx).Create(version).Error; err != nil {

    }

// generateNextVersionNumber 生成下一个版本号
func (s *BotVersionService) generateNextVersionNumber(ctx context.Context, botID string) (string, error) {
    var latestVersion BotVersion
    err := s.db.WithContext(ctx).
        Where("bot_id = ?", botID).
        Order("created_at DESC").
        First(&latestVersion).Error

    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return "v1.0.0", nil
        }
        return "", err
    }

    // 解析版本号并递增
    parts := strings.TrimPrefix(latestVersion.VersionNumber, "v")
    nums := strings.Split(parts, ".")
    major, _ := strconv.Atoi(nums[0])
    minor, _ := strconv.Atoi(nums[1])
    patch, _ := strconv.Atoi(nums[2])

    // 递增补丁版本
    patch++

    return fmt.Sprintf("v%d.%d.%d", major, minor, patch), nil
}

// Rollback 回滚到指定版本
func (s *BotVersionService) Rollback(ctx context.Context, tenantID string, userID uint, botID string, versionID string) error {
    // 1. 查询版本配置
    var version BotVersion
    if err := s.db.WithContext(ctx).
        Where("id = ? AND bot_id = ?", versionID, botID).
        First(&version).Error; err != nil {
        return err
    }

    // 2. 解析配置
    var config BotConfig
    if err := json.Unmarshal([]byte(version.Config), &config); err != nil {
        return err
    }

    // 3. 更新Bot配置
    var bot Bot
    if err := s.db.WithContext(ctx).
        Where("id = ? AND tenant_id = ?", botID, tenantID).
        First(&bot).Error; err != nil {
        return err
    }

    bot.Name = config.BasicInfo.Name
    bot.Description = config.BasicInfo.Description
    bot.Avatar = config.BasicInfo.Avatar
    bot.Prompt = config.Prompt
    bot.ConversationStyle = config.ConversationStyle
    bot.Temperature = config.Temperature
    bot.MaxTokens = config.MaxTokens

    if err := s.db.WithContext(ctx).Save(&bot).Error; err != nil {
        return err
    }

    // 4. 重建技能关联
    s.db.WithContext(ctx).Where("bot_id = ?", botID).Delete(&BotSkill{})
    for _, skillConfig := range config.Skills {
        botSkill := &BotSkill{
            BotID:     botID,
            SkillID:   skillConfig.SkillID,
            IsEnabled: skillConfig.IsEnabled,
            Config:    skillConfig.Config,
        }
        s.db.WithContext(ctx).Create(botSkill)
    }

    // 5. 重建知识库关联
    s.db.WithContext(ctx).Where("bot_id = ?", botID).Delete(&BotKnowledgeBase{})
    for _, kbConfig := range config.KnowledgeBases {
        botKB := &BotKnowledgeBase{
            BotID:           botID,
            KnowledgeBaseID: kbConfig.KnowledgeBaseID,
            Priority:        kbConfig.Priority,
            IsEnabled:       kbConfig.IsEnabled,
        }
        s.db.WithContext(ctx).Create(botKB)
    }

    // 6. 创建回滚版本记录
    s.Create(ctx, tenantID, userID, botID, fmt.Sprintf("回滚到版本 %s", version.VersionNumber))

    return nil
}

// 使用索引优化Bot列表查询
func (s *BotService) ListOptimized(ctx context.Context, tenantID string, params *ListParams) (*ListResult, error) {
    var bots []Bot
    var total int64

    // 1. 使用索引: idx_tenant_id + idx_status + idx_type
    query := s.db.WithContext(ctx).
        Where("tenant_id = ? AND status = ? AND type = ?", tenantID, params.Status, params.Type).
        Where("deleted_at IS NULL")

    // 2. 只查询必要字段(覆盖索引)
    query = query.Select("id, name, description, avatar, type, status, is_listed, usage_count, active_users, created_at, updated_at")

    // 3. 延迟加载关联数据(按需加载)
    // 不使用Preload,在需要时单独查询

    // 4. 分页
    offset := (params.Page - 1) * params.PageSize
    query = query.Offset(offset).Limit(params.PageSize)

    // 5. 执行查询
    if err := query.Find(&bots).Error; err != nil {
        return nil, err
    }

    // 6. 单独统计总数(避免COUNT(*)在大数据集时的性能问题)
    s.db.WithContext(ctx).
        Model(&Bot{}).
        Where("tenant_id = ? AND status = ? AND type = ?", tenantID, params.Status, params.Type).
        Count(&total)

    return &ListResult{
        Total: total,
        Items: bots,
    }, nil
}

func (s *BotService) GetByIDCached(ctx context.Context, botID string) (*Bot, error) {
    // 1. 尝试从Redis获取
    cacheKey := fmt.Sprintf("bot:%s", botID)
    cached, err := s.redis.Get(ctx, cacheKey).Result()
    if err == nil {
        var bot Bot
        if err := json.Unmarshal([]byte(cached), &bot); err == nil {
            return &bot, nil
        }
    }

    // 2. 从数据库查询
    bot, err := s.GetByID(ctx, botID)
    if err != nil {
        return nil, err
    }

    // 3. 缓存到Redis(1小时过期)
    data, _ := json.Marshal(bot)
    s.redis.Set(ctx, cacheKey, data, 1*time.Hour)

    return bot, nil
}
