## 1. 模块概述

**数据看板** 是 ZKER 企业级 SaaS 平台的企业管理中台模块，通过**拖拽式框架 + JSON配置**的方式，提供可视化数据看板能力。

**核心设计理念**：
- ✅ **拖拽式框架**：10% 可视化拖拽组件
- ✅ **JSON配置**：90% 看板布局和数据源配置
- ✅ **实时数据**：支持WebSocket实时更新

**实现策略**：✅ 10% 编码（拖拽框架） + 90% 配置（JSON配置）

```tsx
      <div className={styles.toolbar}>
        <Button onClick={addComponent}>添加组件</Button>
        <Button onClick={saveDashboard}>保存</Button>

      <RGL
        className={styles.grid}
        layout={layout}
        onLayoutChange={handleLayoutChange}
        cols={12}
        rowHeight={60}
        isDraggable={true}
        isResizable={true}
      >
        {layout.map((item: any) => (
          <div key={item.i} className={styles.gridItem}>
            <DashboardComponent component={item} />
          </div>
        ))}
      </RGL>
    </div>
  );
};

// 根据类型渲染不同组件
const DashboardComponent: React.FC<{ component: any }> = ({ component }) => {
  switch (component.type) {
    case 'metric-card':
      return <MetricCard config={component} />;
    case 'line-chart':
      return <LineChart config={component} />;
    case 'table':
      return <Table config={component} />;
    default:
      return <div>未知组件类型</div>;
  }
};
```

## 7. 核心代码实现

### 7.1 数据加载服务

```go
package dashboard

// DataService 数据服务
type DataService struct {
    db          *gorm.DB
    cache       *redis.Client
}

// LoadDashboardData 加载看板数据
func (s *DataService) LoadDashboardData(
    ctx context.Context,
    dashboardID string,
) (map[string]interface{}, error) {
    // 1. 加载看板配置
    dashboard, err := s.getDashboard(ctx, dashboardID)
    if err != nil {
        return nil, err
    }

    result := make(map[string]interface{})

    // 2. 并发加载所有组件数据
    var wg sync.WaitGroup
    var mu sync.Mutex

    for _, component := range dashboard.Components {
        wg.Add(1)
        go func(comp DashboardComponent) {
            defer wg.Done()

            data, err := s.loadComponentData(ctx, comp)
            if err != nil {
                s.logger.Error("failed to load component data",
                    zap.String("component_id", comp.ID),
                    zap.Error(err))
                return
            }

            mu.Lock()
            result[comp.ID] = data
            mu.Unlock()
        }(component)
    }

    wg.Wait()
    return result, nil
}

// loadComponentData 加载组件数据
func (s *DataService) loadComponentData(
    ctx context.Context,
    component DashboardComponent,
) (interface{}, error) {
    ds := component.DataSource

    switch ds.Type {
    case "sql":
        return s.executeSQL(ctx, ds.Query)
    case "api":
        return s.callAPI(ctx, ds.URL)
    case "websocket":
        return s.subscribeWebSocket(ctx, ds.Channel)
    default:
        return nil, fmt.Errorf("unknown data source type: %s", ds.Type)
    }
}
```

#### 8.2.1 Bot指标表 (bot_metrics)

**设计目的**: 存储Bot的核心运行指标,支持按时间序列查询和聚合分析。

```sql
CREATE TABLE bot_metrics (

    resource_points_consumed BIGINT DEFAULT 0 COMMENT '资源点消耗',

    -- 满意度指标

    feedback_count INT DEFAULT 0 COMMENT '反馈数量',

    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    UNIQUE KEY uk_tenant_bot_date_hour (tenant_id, bot_id, date, hour),
    INDEX idx_tenant_id (tenant_id),

    INDEX idx_date (date),
    INDEX idx_date_hour (date, hour)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
COMMENT='Bot指标数据表-按小时/天汇总';

-- 创建分区表(按月分区,保留12个月数据)
ALTER TABLE bot_metrics PARTITION BY RANGE (TO_DAYS(date)) (
    PARTITION p202501 VALUES LESS THAN (TO_DAYS('2025-02-01')),
    PARTITION p202502 VALUES LESS THAN (TO_DAYS('2025-03-01')),
    PARTITION p202503 VALUES LESS THAN (TO_DAYS('2025-04-01')),
```

**字段说明**：
- **time_granularity**: 支持`hour`(小时级)和`day`(日级)两种粒度

- **分区策略**: 按月分区,自动清理超过12个月的数据
- **索引优化**: `tenant_id + bot_id + date` 组合索引,加速多租户查询

```go
// pkg/analytics/collector/collector.go
package collector

import (
    "context"
    "time"
)

// MetricsCollector 指标采集器接口
type MetricsCollector interface {
    // CollectConversationMetrics 采集对话指标
    CollectConversationMetrics(ctx context.Context, event *ConversationEvent) error

    // CollectUserBehavior 采集用户行为
    CollectUserBehavior(ctx context.Context, behavior *UserBehavior) error

    // CollectPerformanceMetrics 采集性能指标
    CollectPerformanceMetrics(ctx context.Context, metrics *PerformanceMetrics) error

    // Flush 刷新缓存中的数据到数据库
    Flush(ctx context.Context) error
}

// ConversationEvent 对话事件
type ConversationEvent struct {
    TenantID        string
    BotID           int64
    UserID          int64
    ConversationID  string
    MessageType     string // text, image, audio, etc.
    TokenCount      int
    ResponseTimeMs  int
    ModelID         string
    IsError         bool
    FeedbackScore   *int
    Timestamp       time.Time
}

// UserBehavior 用户行为
type UserBehavior struct {
    TenantID     string
    UserID       int64
    SessionID    string
    Action       string // bot_view, bot_chat, etc.
    EntityType   string // bot, workflow, etc.
    EntityID     int64
    Content      map[string]interface{}
    IPAddress    string
    UserAgent    string
    Referrer     string
    DeviceType   string
    DurationMs   int
    Timestamp    time.Time
}

// PerformanceMetrics 性能指标
type PerformanceMetrics struct {
    TenantID       string
    BotID          int64
    AvgResponseTime float64
    P95ResponseTime float64
    P99ResponseTime float64
    ErrorRate      float64
    Timestamp      time.Time
}
```

#### 8.3.2 MetricsCollector 实现

```go
package collector

import (
    "context"
    "encoding/json"
    "fmt"

type MetricsCollectorImpl struct {

func NewMetricsCollector(db *gorm.DB) *MetricsCollectorImpl {

func (mc *MetricsCollectorImpl) CollectConversationMetrics(
    ctx context.Context,
    event *ConversationEvent,
) error {
    // 1. 写入缓冲区
    mc.bufferMutex.Lock()
    mc.buffer = append(mc.buffer, event)
    shouldFlush := len(mc.buffer) >= mc.bufferSize
    mc.bufferMutex.Unlock()

    // 2. 缓冲区满则立即刷新
    if shouldFlush {
        return mc.Flush(ctx)
    }

    return nil
}

// Flush 刷新缓冲区数据到数据库
func (mc *MetricsCollectorImpl) Flush(ctx context.Context) error {
    mc.bufferMutex.Lock()
    if len(mc.buffer) == 0 {
        mc.bufferMutex.Unlock()
        return nil
    }

    // 复制缓冲区并清空
    events := make([]*ConversationEvent, len(mc.buffer))
    copy(events, mc.buffer)
    mc.buffer = mc.buffer[:0]
    mc.bufferMutex.Unlock()

    // 批量写入数据库
    return mc.batchInsert(ctx, events)
}

// batchInsert 批量插入指标数据
func (mc *MetricsCollectorImpl) batchInsert(
    ctx context.Context,
    events []*ConversationEvent,
) error {
    tx := mc.db.WithContext(ctx).Begin()
    defer func() {
        if r := recover(); r != nil {
            tx.Rollback()
        }
    }()

    for _, event := range events {
        // 1. 写入 bot_metrics 表(按小时汇总)
        date := event.Timestamp.Format("2006-01-02")
        hour := event.Timestamp.Hour()

        metrics := &BotMetric{
            TenantID:         event.TenantID,
            BotID:            event.BotID,
            Date:             date,
            Hour:             &hour,
            ConversationCount: 1,
            MessageCount:      1,
            TokenConsumption:  event.TokenCount,
        }

        // 使用 INSERT ... ON DUPLICATE KEY UPDATE 实现累加
        err := tx.Table("bot_metrics").
            Where("tenant_id = ? AND bot_id = ? AND date = ? AND hour = ?",
                event.TenantID, event.BotID, date, hour).
            Assign(metrics).
            FirstOrCreate(metrics).Error

        if err != nil {
            tx.Rollback()
            return fmt.Errorf("failed to insert metrics: %w", err)
        }

        // 2. 写入 conversation_logs 表(详细日志)
        contentBytes, _ := json.Marshal(event)
        log := &ConversationLog{
            TenantID:        event.TenantID,
            BotID:           event.BotID,
            UserID:          event.UserID,
            ConversationID:  event.ConversationID,
            Role:            "user",
            TokenCount:      event.TokenCount,
            ResponseTimeMs:  event.ResponseTimeMs,
            ModelID:         event.ModelID,
            IsError:         event.IsError,
            FeedbackScore:   event.FeedbackScore,
            CreatedAt:       event.Timestamp,
        }

        if err := tx.Create(log).Error; err != nil {
            tx.Rollback()
            return fmt.Errorf("failed to insert log: %w", err)
        }
    }

    return tx.Commit().Error
}

// backgroundFlush 后台定时刷新
func (mc *MetricsCollectorImpl) backgroundFlush() {
    ticker := time.NewTicker(mc.flushInterval)
    defer ticker.Stop()

    for range ticker.C {
        _ = mc.Flush(context.Background())
    }
}

// CollectUserBehavior 采集用户行为
func (mc *MetricsCollectorImpl) CollectUserBehavior(
    ctx context.Context,
    behavior *UserBehavior,
) error {
    // 直接写入 user_behaviors 表
    contentBytes, _ := json.Marshal(behavior.Content)

    record := &UserBehavior{
        TenantID:    behavior.TenantID,
        UserID:      behavior.UserID,
        SessionID:   &behavior.SessionID,
        Action:      behavior.Action,
        EntityType:  &behavior.EntityType,
        EntityID:    &behavior.EntityID,
        Content:     string(contentBytes),
        IPAddress:   &behavior.IPAddress,
        UserAgent:   &behavior.UserAgent,
        Referrer:    &behavior.Referrer,
        DeviceType:  behavior.DeviceType,
        DurationMs:  &behavior.DurationMs,
        CreatedAt:   behavior.Timestamp,
    }

    return mc.db.WithContext(ctx).Create(record).Error
}

// CollectPerformanceMetrics 采集性能指标
func (mc *MetricsCollectorImpl) CollectPerformanceMetrics(
    ctx context.Context,
    metrics *PerformanceMetrics,
) error {
    date := metrics.Timestamp.Format("2006-01-02")
    hour := metrics.Timestamp.Hour()

    // 更新 bot_metrics 表的性能字段
    return mc.db.WithContext(ctx).
        Table("bot_metrics").
        Where("tenant_id = ? AND bot_id = ? AND date = ? AND hour = ?",
            metrics.TenantID, metrics.BotID, date, hour).
        Updates(map[string]interface{}{
            "avg_response_time": metrics.AvgResponseTime,
            "p95_response_time": metrics.P95ResponseTime,
            "p99_response_time": metrics.P99ResponseTime,
            "error_rate":       metrics.ErrorRate,
        }).Error
}
```
