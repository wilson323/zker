## 🗄️ 数据库设计（扩展）

### 6.1 成员表扩展 (members)

```sql
-- 扩展现有 members 表
ALTER TABLE members
-- 试用期管理
ADD COLUMN probation_end_date DATE COMMENT '试用期结束日期',
ADD COLUMN probation_status ENUM('pending', 'passed', 'failed', 'extended') COMMENT '试用期状态',
ADD COLUMN probation_feedback TEXT COMMENT '试用期反馈',

-- 职级薪酬
ADD COLUMN job_level VARCHAR(32) COMMENT '职级 (P1-P7, M1-M5)',
ADD COLUMN salary_grade VARCHAR(32) COMMENT '薪级',

-- 档案信息
ADD COLUMN gender ENUM('male', 'female', 'other') COMMENT '性别',
ADD COLUMN birth_date DATE COMMENT '出生日期',
ADD COLUMN id_card VARCHAR(64) COMMENT '身份证号（加密）',
ADD COLUMN marital_status ENUM('single', 'married', 'divorced', 'widowed') COMMENT '婚姻状况',
ADD COLUMN nationality VARCHAR(32) DEFAULT '中国' COMMENT '国籍',
ADD COLUMN native_place VARCHAR(128) COMMENT '籍贯',
ADD COLUMN address VARCHAR(512) COMMENT '居住地址',

-- 紧急联系人
ADD COLUMN emergency_contact_name VARCHAR(64) COMMENT '紧急联系人姓名',
ADD COLUMN emergency_contact_phone VARCHAR(20) COMMENT '紧急联系人电话（加密）',
ADD COLUMN emergency_contact_relation VARCHAR(32) COMMENT '紧急联系人关系',

-- 工作信息
ADD COLUMN work_email VARCHAR(255) COMMENT '工作邮箱',
ADD COLUMN office_location VARCHAR(128) COMMENT '办公地点',
ADD COLUMN work_phone VARCHAR(20) COMMENT '工作电话',

-- 状态
ADD COLUMN entry_status ENUM('probation', 'regular', 'resigned', 'blacklist') DEFAULT 'probation' COMMENT '入职状态',
ADD COLUMN leave_date DATE COMMENT '实际离职日期',
ADD COLUMN leave_reason TEXT COMMENT '离职原因',
ADD COLUMN rehireable BOOLEAN DEFAULT TRUE COMMENT '是否可再入职',

-- 扩展索引
ADD INDEX idx_entry_status (entry_status),
ADD INDEX idx_probation_end (probation_end_date),
ADD INDEX idx_job_level (job_level);
```

### 6.2 工作经历表 (work_experiences)

```sql
CREATE TABLE work_experiences (
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT 'ID',
    member_id BIGINT NOT NULL COMMENT '成员ID',
    company_name VARCHAR(255) NOT NULL COMMENT '公司名称',
    industry VARCHAR(128) COMMENT '行业',
    position VARCHAR(128) COMMENT '职位',
    start_date DATE NOT NULL COMMENT '开始日期',
    end_date DATE COMMENT '结束日期',
    description TEXT COMMENT '工作描述',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',

    INDEX idx_member_id (member_id),
    INDEX idx_start_date (start_date),
    FOREIGN KEY (member_id) REFERENCES members(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='工作经历表';
```

```sql
CREATE TABLE temporary_grants (

    INDEX idx_grantor (grantor_id),
    INDEX idx_time_range (start_time, end_time),
    INDEX idx_status (is_revoked, start_time, end_time),
    FOREIGN KEY (grantor_id) REFERENCES members(id) ON DELETE CASCADE,
    FOREIGN KEY (grantee_id) REFERENCES members(id) ON DELETE CASCADE
```

```sql
CREATE TABLE position_changes (
```

```go
// service/data_permission_service.go
package service

type DataPermissionService struct {
    db *gorm.DB
}

// CheckDataPermission 检查数据权限
func (s *DataPermissionService) CheckDataPermission(
    ctx context.Context,
    userID uint,
    tenantID string,
    resourceType string,
    resourceOwnerID uint,  // 资源拥有者ID（如成员ID）
) (bool, error) {

    // 1. 获取用户的所有角色
    var roleIDs []uint
    s.db.WithContext(ctx).
        Table("member_roles").
        Select("role_id").
        Where("member_id = ?", userID).
        Pluck("role_id", &roleIDs)

    if len(roleIDs) == 0 {
        return false, nil
    }

    // 2. 查询角色的数据权限配置
    var dataPermissions []DataPermission
    s.db.WithContext(ctx).
        Where("role_id IN ?", roleIDs).
        Where("resource_type = ?", resourceType).
        Find(&dataPermissions)

    // 3. 检查权限（取最宽松的权限）
    for _, dp := range dataPermissions {
        switch dp.PermissionScope {
        case "all":
            return true, nil
        case "self":
            if resourceOwnerID == userID {
                return true, nil
            }
        case "department":
            // 检查是否同部门
            if s.isSameDepartment(ctx, userID, resourceOwnerID) {
                return true, nil
            }
        case "department_and_below":
            // 检查是否是本部门或下级部门
            if s.isDepartmentOrBelow(ctx, userID, resourceOwnerID) {
                return true, nil
            }
        case "custom":
            // 检查是否在自定义组织列表中
            var customOrgIDs []uint
            json.Unmarshal([]byte(dp.CustomOrgIDs), &customOrgIDs)
            if s.isInCustomOrgs(ctx, userID, resourceOwnerID, customOrgIDs) {
                return true, nil
            }
        }
    }

    return false, nil
}

// isSameDepartment 检查是否同部门
func (s *DataPermissionService) isSameDepartment(ctx context.Context, userID1, userID2 uint) bool {
    var orgID1, orgID2 uint
    s.db.WithContext(ctx).
        Model(&Member{}).
        Where("id = ?", userID1).
        Pluck("organization_id", &orgID1)
    s.db.WithContext(ctx).
        Model(&Member{}).
        Where("id = ?", userID2).
        Pluck("organization_id", &orgID2)

    return orgID1 == orgID2
}

// isDepartmentOrBelow 检查是否是本部门或下级部门
func (s *DataPermissionService) isDepartmentOrBelow(ctx context.Context, managerID, memberID uint) bool {
    // 获取manager的组织路径
    var manager Member
    s.db.WithContext(ctx).First(&manager, managerID)

    var managerOrg Organization
    s.db.WithContext(ctx).First(&managerOrg, manager.OrganizationID)

    // 获取member的组织路径
    var member Member
    s.db.WithContext(ctx).First(&member, memberID)

    var memberOrg Organization
    s.db.WithContext(ctx).First(&memberOrg, member.OrganizationID)

    // 检查member的组织路径是否以manager的组织路径开头
    return strings.HasPrefix(memberOrg.Path, managerOrg.Path)
}
```

### 9.2 字段权限验证

```go
// service/field_permission_service.go
package service

type FieldPermissionService struct {
    db *gorm.DB
}

// GetFieldPermissions 获取字段权限配置
func (s *FieldPermissionService) GetFieldPermissions(
    ctx context.Context,
    userID uint,
    tenantID string,
    resourceType string,
) (map[string]FieldPermissionConfig, error) {

    // 1. 获取用户的所有角色
    var roleIDs []uint
    s.db.WithContext(ctx).
        Table("member_roles").
        Select("role_id").
        Where("member_id = ?", userID).
        Pluck("role_id", &roleIDs)

    // 2. 查询字段权限
    var fieldPerms []FieldPermission
    s.db.WithContext(ctx).
        Where("role_id IN ?", roleIDs).
        Where("resource_type = ?", resourceType).
        Find(&fieldPerms)

    // 3. 合并权限（同一字段取最高权限：visible > masked > hidden）
    permMap := make(map[string]FieldPermissionConfig)
    for _, fp := range fieldPerms {
        existing, ok := permMap[fp.FieldName]
        if !ok {
            permMap[fp.FieldName] = FieldPermissionConfig{
                FieldName: fp.FieldName,
                Permission: fp.Permission,
                MaskRule:  fp.MaskRule,
            }
        } else {
            // 如果当前权限高于已存在权限，则更新
            if s.isHigherPermission(fp.Permission, existing.Permission) {
                existing.Permission = fp.Permission
                existing.MaskRule = fp.MaskRule
            }
        }
    }

    return permMap, nil
}

// isHigherPermission 判断权限是否更高
func (s *FieldPermissionService) isHigherPermission(perm1, perm2 string) bool {
    order := map[string]int{
        "visible": 3,
        "masked":  2,
        "hidden":  1,
    }
    return order[perm1] > order[perm2]
}

// ApplyFieldPermissions 应用字段权限（脱敏）
func (s *FieldPermissionService) ApplyFieldPermissions(
    ctx context.Context,
    userID uint,
    tenantID string,
    resourceType string,
    data map[string]interface{},
) map[string]interface{} {

    // 1. 获取字段权限配置
    permConfigs, err := s.GetFieldPermissions(ctx, userID, tenantID, resourceType)
    if err != nil {
        return data
    }

    // 2. 应用权限
    for fieldName, value := range data {
        if permConfig, ok := permConfigs[fieldName]; ok {
            switch permConfig.Permission {
            case "visible":
                // 保持原值
                data[fieldName] = value
            case "masked":
                // 应用脱敏规则
                if strValue, ok := value.(string); ok {
                    data[fieldName] = s.maskField(strValue, permConfig.MaskRule)
                }
            case "hidden":
                // 隐藏字段
                delete(data, fieldName)
            }
        }
    }

    return data
}

// maskField 字段脱敏
func (s *FieldPermissionService) maskField(value, maskRule string) string {
    if maskRule == "" {
        maskRule = "####****####"  // 默认脱敏规则
    }

    runes := []rune(value)
    result := []rune{}
    ruleIndex := 0

    for i, r := range runes {
        if ruleIndex >= len(maskRule) {
            ruleIndex = len(maskRule) - 1
        }

        ruleChar := maskRule[ruleIndex]
        if ruleChar == '#' {
            // 保留原字符
            result = append(result, r)
        } else if ruleChar == '*' {
            // 替换为*
            result = append(result, '*')
        }

        ruleIndex++
    }

    return string(result)
}
```
