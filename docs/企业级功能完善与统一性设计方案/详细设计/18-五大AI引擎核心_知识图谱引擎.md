## 1. 模块概述

**知识图谱引擎** 负责构建和管理企业知识图谱，通过**实体识别 + 关系抽取 + 图数据库**，实现知识的结构化存储和推理查询。

**核心设计理念**：
- ✅ **图谱构建**：自动从文档中抽取实体和关系
- ✅ **图存储**：使用 Neo4j 或 NebulaGraph
- ✅ **图谱推理**：基于图结构的推理

**实现策略**：✅ 60% 编码（图谱引擎） + 40% 配置（抽取规则）

---

## 2. 核心功能

### 2.1 图谱构建

**F1 - 实体识别**：从文本中识别人名、地名、组织等
**F2 - 关系抽取**：识别实体间的关系
**F3 - 知识融合**：合并重复实体

### 2.2 图谱查询

**F4 - 图谱遍历**：查找关联实体
**F5 - 路径查询**：查找实体间路径
**F6 - 推理查询**：基于规则的推理

## 4. 引擎设计

### 4.1 知识图谱引擎

```go
package knowledgegraph

// KnowledgeGraphEngine 知识图谱引擎
type KnowledgeGraphEngine struct {
    neo4jDriver *neo4j.Driver
    entityRepo  repository.EntityRepository
    llmClient   *llm.Client
}

// ExtractEntities 实体抽取
func (e *KnowledgeGraphEngine) ExtractEntities(
    ctx context.Context,
    text string,
) ([]*Entity, error) {
    // 1. 使用LLM进行实体识别
    prompt := fmt.Sprintf("从以下文本中识别实体（人名、地名、组织、时间等）：\n%s", text)

    resp, err := e.llmClient.Chat(ctx, &llm.ChatRequest{
        Model: "gpt-4",
        Messages: []llm.Message{
            {Role: "system", Content: "你是一个实体识别专家"},
            {Role: "user", Content: prompt},
        },
    })
    if err != nil {
        return nil, err
    }

    // 2. 解析LLM输出
    entities := e.parseEntities(resp.Content)

    return entities, nil
}

// ExtractRelationships 关系抽取
func (e *KnowledgeGraphEngine) ExtractRelationships(
    ctx context.Context,
    entities []*Entity,
    text string,
) ([]*Relationship, error) {
    // 使用LLM识别实体间关系
    entityNames := make([]string, len(entities))
    for i, e := range entities {
        entityNames[i] = e.Name
    }

    prompt := fmt.Sprintf(
        "识别实体间的关系。实体：%v\n文本：%s",
        entityNames, text,
    )

    resp, err := e.llmClient.Chat(ctx, &llm.ChatRequest{
        Model: "gpt-4",
        Messages: []llm.Message{
            {Role: "user", Content: prompt},
        },
    })

    if err != nil {
        return nil, err
    }

    return e.parseRelationships(resp.Content), nil
}

// BuildGraph 构建图谱
func (e *KnowledgeGraphEngine) BuildGraph(
    ctx context.Context,
    entities []*Entity,
    relationships []*Relationship,
) error {
    session := e.neo4jDriver.NewSession(neo4j.AccessModeWrite)
    defer session.Close()

    // 1. 创建实体节点
    for _, entity := range entities {
        _, err := session.Run(
            `MERGE (e:Entity {id: $id, type: $type, name: $name, properties: $props})`,
            map[string]interface{}{
                "id":   entity.ID,
                "type": entity.Type,
                "name": entity.Name,
                "props": entity.Properties,
            },
        )
        if err != nil {
            return err
        }
    }

    // 2. 创建关系边
    for _, rel := range relationships {
        _, err := session.Run(
            `MATCH (from:Entity {id: $fromId}), (to:Entity {id: $toId})
             MERGE (from)-[r:RELATIONSHIP {type: $type, properties: $props}]->(to)`,
            map[string]interface{}{
                "fromId":  rel.FromEntity,
                "toId":    rel.ToEntity,
                "type":    rel.RelationType,
                "props":   rel.Properties,
            },
        )
        if err != nil {
            return err
        }
    }

    return nil
}
```
