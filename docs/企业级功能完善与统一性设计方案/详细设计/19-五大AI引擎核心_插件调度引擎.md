## 1. 模块概述

**插件调度引擎** 负责智能调用和管理插件/技能，通过**意图识别 + 参数提取 + 插件执行**的流程，实现AI能力的扩展。

**核心设计理念**：
- ✅ **智能调度**：根据意图自动选择合适的插件
- ✅ **并行执行**：支持多插件并行调用
- ✅ **结果聚合**：整合多个插件的结果

**实现策略**：✅ 50% 编码（调度框架） + 50% 配置（插件配置）

---

## 2. 核心功能

### 2.1 插件调度

**F1 - 意图识别**：识别用户意图，匹配插件
**F2 - 参数提取**：从用户输入中提取插件参数
**F3 - 插件执行**：调用插件API
**F4 - 结果格式化**：格式化插件输出

### 2.2 插件管理

**F5 - 插件注册**：动态注册新插件
**F6 - 插件健康检查**：监控插件可用性
**F7 - 插件限流**：防止插件滥用

```sql
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '执行ID',
    tenant_id VARCHAR(64) NOT NULL COMMENT '租户ID',
    user_id BIGINT NOT NULL COMMENT '用户ID',
    conversation_id VARCHAR(64) COMMENT '会话ID',
    message_id VARCHAR(64) COMMENT '消息ID',

    -- 执行信息
    input_params JSON COMMENT '输入参数',
    output_data JSON COMMENT '输出数据',
    status ENUM('pending', 'running', 'success', 'failed', 'timeout') NOT NULL,
    error_message TEXT COMMENT '错误信息',

    -- 性能
    start_time DATETIME NOT NULL COMMENT '开始时间',
    end_time DATETIME COMMENT '结束时间',
    duration_ms INT COMMENT '耗时(毫秒)',

    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,

    INDEX idx_tenant_id (tenant_id),
    INDEX idx_plugin_id (plugin_id),
    INDEX idx_status (status),
    INDEX idx_start_time (start_time)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
COMMENT='插件执行日志表';
```

## 4. 引擎设计

### 4.1 插件调度引擎

```go
package plugin

// PluginScheduler 插件调度引擎
type PluginScheduler struct {
    pluginRepo     repository.PluginRepository
    executionRepo  repository.ExecutionRepository
    intentEngine   *intent.IntentEngine
    httpClient     *http.Client
    logger         *zap.Logger
}

// SchedulePlugin 调度插件
func (s *PluginScheduler) SchedulePlugin(
    ctx context.Context,
    userQuery string,
    conversationID string,
) (*PluginExecutionResult, error) {
    // 1. 意图识别
    intent, err := s.intentEngine.RecognizeIntent(ctx, userQuery)
    if err != nil {
        return nil, err
    }

    // 2. 匹配插件
    plugin, err := s.pluginRepo.GetByIntent(ctx, intent.Intent)
    if err != nil {
        return nil, fmt.Errorf("no plugin found for intent: %s", intent.Intent)
    }

    // 3. 参数提取
    params, err := s.extractParameters(ctx, plugin, intent.Entities)
    if err != nil {
        return nil, err
    }

    // 4. 创建执行记录
    execution := &PluginExecution{
        TenantID:       getTenantID(ctx),
        UserID:         getUserID(ctx),
        ConversationID: conversationID,
        PluginID:       plugin.ID,
        PluginName:     plugin.Name,
        InputParams:    params,
        Status:         "pending",
        StartTime:      time.Now(),
    }

    s.executionRepo.Create(ctx, execution)

    // 5. 执行插件
    result, err := s.executePlugin(ctx, plugin, params)
    execution.EndTime = time.Now()
    execution.DurationMs = int(time.Since(execution.StartTime).Milliseconds())

    if err != nil {
        execution.Status = "failed"
        execution.ErrorMessage = err.Error()
        s.executionRepo.Update(ctx, execution)
        return nil, err
    }

    // 6. 保存结果
    execution.Status = "success"
    execution.OutputData = result
    s.executionRepo.Update(ctx, execution)

    return &PluginExecutionResult{
        PluginID:   plugin.ID,
        PluginName: plugin.Name,
        Result:     result,
    }, nil
}

// executePlugin 执行插件
func (s *PluginScheduler) executePlugin(
    ctx context.Context,
    plugin *Plugin,
    params map[string]interface{},
) (map[string]interface{}, error) {
    // 1. 构建插件API请求
    pluginAPI := plugin.Config["api_url"].(string)

    reqBody, _ := json.Marshal(map[string]interface{}{
        "action": plugin.Config["action"],
        "params": params,
    })

    // 2. 调用插件API
    req, err := http.NewRequestWithContext(
        ctx,
        "POST",
        pluginAPI,
        bytes.NewReader(reqBody),
    )
    if err != nil {
        return nil, err
    }

    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("Authorization", "Bearer "+plugin.Config["api_key"].(string))

    // 3. 设置超时
    client := &http.Client{Timeout: 30 * time.Second}
    resp, err := client.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    // 4. 解析响应
    var result map[string]interface{}
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return nil, err
    }

    if resp.StatusCode != 200 {
        return nil, fmt.Errorf("plugin returned error: %s", result["error"])
    }

    return result, nil
}
```
