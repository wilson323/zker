## 1. 模块概述

**人机协同引擎** 是五大AI引擎的核心，通过**AI决策 + 人工审核**的协同机制，在关键决策点引入人工干预，提升AI输出质量和安全性。

**核心设计理念**：
- ✅ **智能触发**：AI自动识别需要人工干预的场景
- ✅ **协同工作流**：人工审核、修正、反馈
- ✅ **持续学习**：人工反馈用于模型微调

**实现策略**：✅ 40% 编码（协同框架） + 60% 配置（规则配置）

## 2. 核心功能

### 2.1 协同触发

**F1 - 自动识别**
- F1.1 置信度低于阈值
- F1.2 敏感内容检测
- F1.3 复杂业务逻辑
- F1.4 异常行为检测

**F2 - 手动触发**
- F2.1 用户请求人工介入
- F2.2 管理员强制审核

### 2.2 人工审核

**F3 - 审核工作台**
- F3.1 待审核队列
- F3.2 快速审核（通过/拒绝）
- F3.3 修正AI输出
- F3.4 添加备注

**F4 - 反馈学习**
- F4.1 记录人工决策
- F4.2 模型微调
- F4.3 规则优化

## 4. 引擎设计

### 4.1 人机协同引擎

```go
package engine

// HumanInLoopEngine 人机协同引擎
type HumanInLoopEngine struct {
    ruleRepo      repository.HumanLoopRuleRepository
    taskRepo      repository.HumanLoopTaskRepository
    notifier      *NotificationService
    llmClient     *llm.Client
}

// ShouldTriggerHumanLoop 判断是否需要人工介入
func (e *HumanInLoopEngine) ShouldTriggerHumanLoop(
    ctx context.Context,
    aiOutput string,
    confidence float64,
    metadata map[string]interface{},
) (bool, string) {
    // 1. 检查置信度规则
    rules, err := e.ruleRepo.GetActiveRules(ctx, getTenantID(ctx))
    if err != nil {
        return false, ""
    }

    // 2. 评估每条规则
    for _, rule := range rules {
        triggered, reason := e.evaluateRule(rule, aiOutput, confidence, metadata)
        if triggered {
            return true, reason
        }
    }

    return false, ""
}

// evaluateRule 评估规则
func (e *HumanInLoopEngine) evaluateRule(
    rule *HumanLoopRule,
    aiOutput string,
    confidence float64,
    metadata map[string]interface{},
) (bool, string) {
    switch rule.RuleType {
    case "confidence":
        threshold := rule.Config["confidence"].(map[string]interface{})["threshold"].(float64)
        if confidence < threshold {
            return true, fmt.Sprintf("置信度 %.2f 低于阈值 %.2f", confidence, threshold)
        }

    case "keyword":
        keywords := rule.Config["keywords"].(map[string]interface{})["match"].([]interface{})
        for _, kw := range keywords {
            if strings.Contains(aiOutput, kw.(string)) {
                return true, fmt.Sprintf("包含敏感关键词: %s", kw)
            }
        }
    }

    return false, ""
}

// CreateHumanLoopTask 创建人机协同任务
func (e *HumanInLoopEngine) CreateHumanLoopTask(
    ctx context.Context,
    req *CreateTaskRequest,
) (*HumanLoopTask, error) {
    task := &HumanLoopTask{
        ID:           generateTaskID(),
        TenantID:     getTenantID(ctx),
        TaskType:     req.TaskType,
        AIOutput:     req.AIOutput,
        Confidence:   req.Confidence,
        TriggerReason: req.TriggerReason,
        Status:       "pending",
        Priority:     req.Priority,
    }

    if err := e.taskRepo.Create(ctx, task); err != nil {
        return nil, err
    }

    // 自动分配审核人
    if req.AssignTo > 0 {
        task.AssignedTo = req.AssignTo
        e.taskRepo.Update(ctx, task)
        // 发送通知
        e.notifier.SendNotification(ctx, req.AssignTo, "新的人工审核任务")
    }

    return task, nil
}

// ProcessHumanDecision 处理人工决策
func (e *HumanInLoopEngine) ProcessHumanDecision(
    ctx context.Context,
    taskID string,
    decision string,
    reviewerID int64,
) error {
    // 1. 更新任务状态
    task, err := e.taskRepo.GetByID(ctx, taskID)
    if err != nil {
        return err
    }

    task.Status = "approved"
    task.HumanDecision = decision
    task.ReviewerID = reviewerID
    task.ReviewedAt = time.Now()

    if err := e.taskRepo.Update(ctx, task); err != nil {
        return err
    }

    // 2. 反馈学习（用于模型优化）
    go e.recordFeedbackForLearning(context.Background(), task)

    return nil
}
```
