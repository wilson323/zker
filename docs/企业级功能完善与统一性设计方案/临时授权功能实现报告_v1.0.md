### 2. temporary_grants临时授权表

```sql
CREATE TABLE `opencoze`.`temporary_grants` (
    `grant_id` BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    `grant_code` VARCHAR(100) UNIQUE NOT NULL COMMENT '授权码（一次性使用）',
    `grantee_id` VARCHAR(36) NOT NULL COMMENT '被授权人用户ID',
    `grantor_id` VARCHAR(36) NOT NULL COMMENT '授权人用户ID',
    `tenant_id` VARCHAR(36) NOT NULL COMMENT '租户ID',
    `permission_type` ENUM('role', 'data_permission', 'field_permission') NOT NULL,
    `permission_data` JSON NOT NULL COMMENT '权限数据',
    `expires_at` BIGINT UNSIGNED NOT NULL COMMENT '过期时间（毫秒时间戳）',
    `is_used` BOOLEAN DEFAULT FALSE COMMENT '是否已使用',
    `is_revoked` BOOLEAN DEFAULT FALSE COMMENT '是否已撤销',
    `used_at` BIGINT UNSIGNED NULL COMMENT '使用时间',
    `revoked_at` BIGINT UNSIGNED NULL COMMENT '撤销时间',
    `reason` VARCHAR(500) COMMENT '授权原因',
    `request_id` VARCHAR(36) COMMENT '关联请求ID',
    `created_at` BIGINT UNSIGNED NOT NULL DEFAULT 0,
    `updated_at` BIGINT UNSIGNED NOT NULL DEFAULT 0,

    -- 外键约束
    FOREIGN KEY (`tenant_id`) REFERENCES `tenants`(`tenant_id`) ON DELETE CASCADE,
    FOREIGN KEY (`grantee_id`) REFERENCES `users`(`user_id`) ON DELETE CASCADE,
    FOREIGN KEY (`grantor_id`) REFERENCES `users`(`user_id`) ON DELETE CASCADE,

    -- 索引
    UNIQUE INDEX `uk_grant_code` (`grant_code`),
    INDEX `idx_grantee_id` (`grantee_id`),
    INDEX `idx_grantor_id` (`grantor_id`),
    INDEX `idx_tenant_id` (`tenant_id`),
    INDEX `idx_expires_at` (`expires_at`),
    INDEX `idx_is_used` (`is_used`),
    INDEX `idx_is_revoked` (`is_revoked`),
    INDEX `idx_status_expires` (`is_used`, `is_revoked`, `expires_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```

**核心特性**:
- 授权码唯一索引
- 复合索引优化查询性能
- 外键级联删除保证数据一致性
- 状态字段支持软删除和审计

#### 3.2 授权码生成算法

```go
// 生成16位授权码,格式: XXXX-XXXX-XXXX-XXXX
// 使用Base32编码,避免歧义字符(0/O, 1/I/l)
func (s *TemporaryGrantService) generateGrantCode() (string, error) {
    // 1. 生成随机字节(16字节 = 128位)
    b := make([]byte, 16)
    rand.Read(b)

    // 2. Base32编码(自定义字符集,避免歧义)
    encoded := base32HexEncoding.EncodeToString(b)

    // 3. 格式化为 XXXX-XXXX-XXXX-XXXX
    return fmt.Sprintf("%s-%s-%s-%s",
        encoded[0:4], encoded[4:8], encoded[8:12], encoded[12:16]), nil
}
```

**字符集**: `0123456789ABCDEFGHJKMNPQRSTVWXYZ` (去除0/O, 1/I/l等歧义字符)

#### 3.3 授权验证逻辑

```go
func (s *TemporaryGrantService) UseTemporaryGrant(
    ctx context.Context,
    grantCode string,
    userID string,
) (*entity.UserRole, error) {
    // 1. 查询授权码
    grant, err := s.grantRepo.GetByGrantCode(ctx, grantCode)

    // 2. 验证有效性
    if !grant.IsValid() {
        if grant.IsExpired() {
            return nil, fmt.Errorf("授权码已过期")
        }
        if grant.IsUsed {
            return nil, fmt.Errorf("授权码已被使用")
        }
        if grant.IsRevoked {
            return nil, fmt.Errorf("授权码已被撤销")
        }
    }

    // 3. 验证被授权人
    if grant.GranteeID != userID {
        return nil, fmt.Errorf("授权码不属于当前用户")
    }

    // 4. 根据权限类型分配角色
    return s.useRoleGrant(ctx, grant)
}
```

### 5. 定时清理任务

**文件**: `backend/application/permission/temporary_grant_cleanup.go`

```go
type TemporaryGrantCleanupJob struct {
    temporaryGrantService *TemporaryGrantService
    cleanupInterval       time.Duration // 清理间隔(默认1小时)
    historyRetentionDays  int           // 历史记录保留天数(默认90天)
}

func (j *TemporaryGrantCleanupJob) Start(ctx context.Context) {
    ticker := time.NewTicker(j.cleanupInterval)
    defer ticker.Stop()

    for {
        select {
        case <-ctx.Done():
            return
        case <-ticker.C:
            j.cleanup(ctx) // 清理过期授权和历史
        }
    }
}
```

**清理逻辑**:
1. 清理过期的临时授权
2. 清理90天前的历史记录
3. 自动删除过期的UserRole
4. 记录清理日志

### 常见问题

1. **授权码不存在**
   - 检查: grant_code是否正确
   - 解决: 验证授权码格式和大小写
