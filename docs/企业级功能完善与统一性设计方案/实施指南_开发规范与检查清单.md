```go
    var subscription Subscription
    err := db.Where("id = ?", id).First(&subscription).Error
    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return nil, ErrNotFound
        }
        return nil, errorx.Wrapf(err, "failed to get subscription, id=%d", id)
    }
    return &subscription, nil
}
```

```go
// ❌ 错误 - SQL 注入风险
func GetSubscriptionByName(name string) *Subscription {

    return &subscription
}

// ✅ 正确 - 使用参数化查询
func GetSubscriptionByName(name string) (*Subscription, error) {
    var subscription Subscription
    err := db.Where("name = ?", name).First(&subscription).Error
    return &subscription, err
}
```

#### 8.3 并发安全

```go
// ❌ 错误 - 并发不安全
var counter int

    counter++  // 非原子操作
}

// ✅ 正确 - 使用 sync/atomic
import "sync/atomic"

var counter int64

func Increment() {
    atomic.AddInt64(&counter, 1)
}
```
