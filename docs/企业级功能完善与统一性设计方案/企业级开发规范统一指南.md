## 1. 架构设计规范

### 1.1 DDD分层架构

**四层架构模型**:

```
┌─────────────────────────────────────┐
│         API Layer (接口层)           │  HTTP/gRPC处理器
│  - 接收请求                          │
│  - 参数验证                          │
│  - 调用应用服务                      │
│  - 返回响应                          │
├─────────────────────────────────────┤
│    Application Layer (应用层)        │  用例编排、DTO
│  - 应用服务                          │
│  - DTO转换                          │
│  - 事务管理                          │
│  - 权限检查                          │
├─────────────────────────────────────┤
│      Domain Layer (领域层)           │  实体、仓储接口、领域服务
│  - 领域实体（Entity）                │
│  - 值对象（Value Object）            │
│  - 仓储接口（Repository Interface）  │
│  - 领域服务（Domain Service）        │
│  - 领域事件（Domain Event）          │
├─────────────────────────────────────┤
│   Infrastructure Layer (基础设施层)  │  仓储实现、外部服务
│  - 仓储实现（Repository Impl）       │
│  - 外部服务适配器                    │
│  - 数据库访问                        │
│  - 缓存实现                          │
└─────────────────────────────────────┘
```

**依赖规则**:

```go
// ✅ 正确的依赖方向
api → application → domain
infra → domain  // 实现领域层定义的接口

// ❌ 错误的依赖方向
domain → infra   // 领域层不能依赖基础设施层
domain → application // 领域层不能依赖应用层
```

**目录结构**:

```
backend/
├── api/                    # 接口层
│   └── handler/           # HTTP处理器
├── application/           # 应用层
│   ├── dto/              # 数据传输对象
│   └── service/          # 应用服务
├── domain/               # 领域层
│   ├── entity/           # 领域实体
│   ├── repository/       # 仓储接口
│   └── service/          # 领域服务
└── infra/                # 基础设施层
    ├── repository/       # 仓储实现
```

### 1.3 领域驱动设计原则

**实体（Entity）**:

- 具有唯一标识
- 包含业务逻辑
- 通过方法访问字段
- 保证业务规则不变性

**值对象（Value Object）**:

- 不可变
- 无唯一标识
- 通过值比较相等性

**聚合根（Aggregate Root）**:

- 管理内部对象的一致性
- 提供修改内部对象的接口
- 发布领域事件

**代码示例**:

```go
// 值对象示例
type TenantID struct {
    value string
}

func NewTenantID(value string) (TenantID, error) {
    if value == "" {
        return TenantID{}, ErrInvalidTenantID
    }
    return TenantID{value: value}, nil
}

func (id TenantID) String() string {
    return id.value
}

// 实体示例（聚合根）
type Tenant struct {
    id           TenantID
    name         string
    status       TenantStatus
    events       []interface{}  // 领域事件
}

func (t *Tenant) Activate() error {
    if t.status == TenantStatusActive {
        return nil // 已经是活跃状态
    }
    t.status = TenantStatusActive

    // 记录领域事件
    t.recordEvent(TenantActivatedEvent{
        TenantID: t.id.String(),
    })
    return nil
}

func (t *Tenant) recordEvent(event interface{}) {
    t.events = append(t.events, event)
}

func (t *Tenant) GetEvents() []interface{} {
    events := t.events
    t.events = []interface{}{} // 清空事件
    return events
}
```

### 2.4 错误处理规范

**自定义错误类型**:

```go
// domain/errors/errors.go
package errors

type DomainError struct {
    Code    string
    Message string
    Cause   error
}

func (e *DomainError) Error() string {
    if e.Cause != nil {
        return fmt.Sprintf("%s: %s (caused by: %v)", e.Code, e.Message, e.Cause)
    }
    return fmt.Sprintf("%s: %s", e.Code, e.Message)
}

// 预定义错误
var (
    ErrTenantNotFound     = &DomainError{Code: "TENANT_NOT_FOUND", Message: "tenant not found"}
    ErrTenantNameRequired = &DomainError{Code: "TENANT_NAME_REQUIRED", Message: "tenant name is required"}
    ErrQuotaExceeded      = &DomainError{Code: "QUOTA_EXCEEDED", Message: "quota exceeded"}
)
```

**错误处理模式**:

```go
// ✅ 正确：始终检查错误
func (s *BotService) CreateBot(ctx context.Context, req *CreateBotRequest) (*Bot, error) {
    // 验证
    if err := validateRequest(req); err != nil {
        return nil, fmt.Errorf("validation failed: %w", err)
    }

    // 创建实体
    bot, err := entity.NewBot(req.Name, req.Description)
    if err != nil {
        return nil, fmt.Errorf("failed to create bot: %w", err)
    }

    // 持久化
    if err := s.botRepo.Save(ctx, bot); err != nil {
        return nil, fmt.Errorf("failed to save bot: %w", err)
    }

    return bot, nil
}

// ❌ 错误：忽略错误
func (s *BotService) CreateBot(ctx context.Context, req *CreateBotRequest) (*Bot, error) {
    bot, _ := entity.NewBot(req.Name, req.Description) // 忽略错误
    s.botRepo.Save(ctx, bot) // 忽略错误
    return bot, nil
}
```

**错误包装**:

```go
// 使用 %w 包装错误，保留原始错误
if err := s.botRepo.Save(ctx, bot); err != nil {
    return nil, fmt.Errorf("failed to save bot: %w", err)
}

// 使用 errors.Is 检查错误类型
if errors.Is(err, errors.ErrTenantNotFound) {
    // 处理租户不存在的情况
}

// 使用 errors.As 获取错误详情
```

```go
    data  map[string]interface{}
}

func (c *Cache) Get(key string) (interface{}, bool) {
    c.mu.RLock() // 读锁
    defer c.mu.RUnlock()
    val, ok := c.data[key]
    return val, ok

    defer c.mu.Unlock()
    c.data[key] = val
```

**自定义Hook结构**:

```tsx
// use-bots.ts
import { useState, useEffect } from 'react';
import { botApi } from '@/services/api';
import type { Bot } from './bot.types';

interface UseBotsOptions {
  page?: number;
  pageSize?: number;
  searchTerm?: string;
}

interface UseBotsResult {
  bots: Bot[];
  loading: boolean;
  error: Error | null;
  refetch: () => void;
}

/**
 * 获取Bot列表的自定义Hook
 */
export function useBots(
  tenantId: string,
  options: UseBotsOptions = {},
): UseBotsResult {
  const [bots, setBots] = useState<Bot[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);

  const { page = 1, pageSize = 20, searchTerm } = options;

  const fetchBots = async () => {
    setLoading(true);
    setError(null);

    try {
      const result = await botApi.listBots({
        tenantId,
        page,
        pageSize,
        searchTerm,
      });
      setBots(result.bots);
    } catch (err) {
      setError(err as Error);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchBots();
  }, [tenantId, page, pageSize, searchTerm]);

  return {
    bots,
    loading,
    error,
    refetch: fetchBots,
  };
}
```

### 3.6 状态管理规范

**使用Zustand进行状态管理**:

```tsx
// stores/botStore.ts
import { create } from 'zustand';
import { devtools } from 'zustand/middleware';
import type { Bot } from './bot.types';

interface BotState {
  bots: Bot[];
  selectedBot: Bot | null;
  loading: boolean;

  // Actions
  setBots: (bots: Bot[]) => void;
  selectBot: (bot: Bot | null) => void;
  addBot: (bot: Bot) => void;
  updateBot: (id: string, updates: Partial<Bot>) => void;
  removeBot: (id: string) => void;
}

export const useBotStore = create<BotState>()(
  devtools((set) => ({
    // Initial state
    bots: [],
    selectedBot: null,
    loading: false,

    // Actions
    setBots: (bots) => set({ bots }),

    selectBot: (bot) => set({ selectedBot: bot }),

    addBot: (bot) => set((state) => ({
      bots: [...state.bots, bot],
    })),

    updateBot: (id, updates) => set((state) => ({
      bots: state.bots.map((bot) =>
        bot.id === id ? { ...bot, ...updates } : bot
      ),
    })),

    removeBot: (id) => set((state) => ({
      bots: state.bots.filter((bot) => bot.id !== id),
    })),
  }))
);
```

```tsx
/**

export const botApi = {

    return request.post('/api/v1/bots', data);

   * 更新Bot
   */

    return request.put(`/api/v1/bots/${id}`, data);

  /**
```

## 4. 数据库设计规范

### 4.1 表命名规范

**规则**:
- 表名使用小写字母和下划线
- 表名应该是复数名词
- 多对多关系表使用`_`连接两个表名

**示例**:

```sql
-- ✅ 正确
CREATE TABLE tenants ( );
CREATE TABLE bot_categories ( );
CREATE TABLE user_roles ( );  -- 多对多关系表

-- ❌ 错误
CREATE TABLE Tenant ( );
CREATE TABLE botCategories ( );
CREATE TABLE UserRole ( );
```

### 4.2 字段命名规范

**规则**:
- 字段名使用小写字母和下划线
- 布尔字段使用`is_`前缀
- 时间字段使用`_at`后缀
- 外键字段使用`_id`后缀

**示例**:

```sql
-- ✅ 正确
CREATE TABLE bots (
    id BIGINT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    is_published BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL,
    tenant_id VARCHAR(64) NOT NULL,
    creator_id VARCHAR(64) NOT NULL
);

-- ❌ 错误
CREATE TABLE bots (
    Id BIGINT PRIMARY KEY,
    botName VARCHAR(100) NOT NULL,
    Published BOOLEAN DEFAULT FALSE,
    createTime TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    TenantID VARCHAR(64) NOT NULL
);
```

### 4.3 标准字段规范

**所有表必须包含的字段**:

```sql
CREATE TABLE example (
    -- 主键
    id BIGINT PRIMARY KEY AUTO_INCREMENT,

    -- 租户ID（多租户表必填）
    tenant_id VARCHAR(64) NOT NULL COMMENT '租户ID',

    -- 审计字段
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    deleted_at TIMESTAMP NULL COMMENT '删除时间（软删除）',
    created_by VARCHAR(64) COMMENT '创建人',
    updated_by VARCHAR(64) COMMENT '更新人',

    -- 版本字段（乐观锁）
    version INT DEFAULT 0 COMMENT '版本号',

    -- 索引
    UNIQUE KEY uk_id (id),
    KEY idx_tenant_id (tenant_id),
    KEY idx_deleted_at (deleted_at),
    KEY idx_tenant_deleted (tenant_id, deleted_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='示例表';
```

### 4.4 索引命名规范

**规则**:
- 主键索引：`pk_表名`
- 唯一索引：`uk_字段名`
- 普通索引：`idx_字段名`
- 全文索引：`ft_字段名`
- 外键：`fk_表名_字段名`

**示例**:

```sql
CREATE TABLE bots (
    id BIGINT PRIMARY KEY,
    tenant_id VARCHAR(64) NOT NULL,
    name VARCHAR(100) NOT NULL,
    creator_id VARCHAR(64) NOT NULL,

    -- 主键（自动命名为PRIMARY）
    -- UNIQUE KEY
    UNIQUE KEY uk_id (id),

    -- 普通索引
    KEY idx_tenant_id (tenant_id),
    KEY idx_name (name),
    KEY idx_creator_id (creator_id),

    -- 联合索引
    KEY idx_tenant_name (tenant_id, name),

    -- 外键
    CONSTRAINT fk_bots_tenant
        FOREIGN KEY (tenant_id) REFERENCES tenants(id),
    CONSTRAINT fk_bots_creator
        FOREIGN KEY (creator_id) REFERENCES users(id)
);
```

### 4.5 外键约束规范

**规则**:
- 所有外键必须有约束
- 外键字段命名：`关联表名_id`
- 级联删除谨慎使用

**示例**:

```sql
-- ✅ 正确：有外键约束
ALTER TABLE bots
ADD CONSTRAINT fk_bots_tenant
    FOREIGN KEY (tenant_id)
    REFERENCES tenants(id)
    ON DELETE RESTRICT  -- 禁止级联删除
    ON UPDATE CASCADE;

-- ❌ 错误：无外键约束
ALTER TABLE bots
ADD COLUMN tenant_id VARCHAR(64) NOT NULL;  -- 没有外键约束
```

### 7.1 认证授权

**JWT认证**:

```go
// middleware/auth.go
func AuthMiddleware(secret string) app.HandlerFunc {
    return func(ctx *app.RequestContext) {
        token := ctx.GetHeader("Authorization")
        if token == "" {
            ctx.JSON(401, map[string]string{"error": "missing token"})
            ctx.Abort()
            return
        }

        // 验证JWT
        claims, err := validateJWT(token, secret)
        if err != nil {
            ctx.JSON(401, map[string]string{"error": "invalid token"})
            ctx.Abort()
            return
        }

        // 将用户信息注入上下文
        ctx.Set("user_id", claims.UserID)
        ctx.Set("tenant_id", claims.TenantID)
        ctx.Next(ctx)
    }
}
```

```go
func RequirePermission(permission string) app.HandlerFunc {
```
