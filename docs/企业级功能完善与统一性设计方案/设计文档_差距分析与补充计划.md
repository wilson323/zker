```go
func (m *Middleware) PermissionCheck(resourceType string, requiredPerm Permission) gin.HandlerFunc {
    return func(c *gin.Context) {
        userID := m.GetUserID(c)
        resourceID := c.Param("id")

        hasPermission, err := m.permissionService.CheckPermission(
            c.Request.Context(),
            userID,
            resourceType,
            resourceID,
            requiredPerm,
        )

        if err != nil || !hasPermission {
            c.JSON(http.StatusForbidden, gin.H{"error": "权限不足"})
            c.Abort()
            return
        }

        c.Next()
    }
}

// 使用示例
router.PUT("/api/bots/:id",
    middleware.AuthRequired(),
    middleware.PermissionCheck("BOT", Permission.EDIT),
    botHandler.UpdateBot,
)
```
