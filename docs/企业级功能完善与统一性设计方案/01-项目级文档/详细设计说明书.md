```go
// Register 注册新租户
func (s *TenantService) Register(ctx context.Context, req *RegisterTenantRequest) (*Tenant, error) {
    // 1. 参数校验
    if err := s.validateRegisterRequest(req); err != nil {
        return nil, err
    }

    // 2. 检查子域名唯一性
    exists, err := s.tenantRepository.ExistsBySubdomain(ctx, req.Subdomain)
    if err != nil {
        return nil, err
    }
    if exists {
        return nil, ErrSubdomainAlreadyExists
    }

    // 3. 检查企业名称唯一性
    exists, err = s.tenantRepository.ExistsByName(ctx, req.Name)
    if err != nil {
        return nil, err
    }
    if exists {
        return nil, ErrTenantNameAlreadyExists
    }

    // 4. 获取套餐信息
    plan, err := s.billingService.GetPlanByID(ctx, req.PlanID)
    if err != nil {
        return nil, err
    }

    // 5. 创建租户实体
    tenant := &Tenant{
        TenantID:    generateTenantID(),
        Name:        req.Name,
        Subdomain:   req.Subdomain,
        Industry:    req.Industry,
        Scale:       req.Scale,
        ContactPerson: req.ContactPerson,
        ContactEmail: req.ContactEmail,
        ContactPhone: req.ContactPhone,
        Status:      "active",
        PlanID:      req.PlanID,
        TenantType:  req.TenantType,
        CreatedAt:   time.Now(),
        UpdatedAt:   time.Now(),
    }

    // 6. 保存到数据库
    if err := s.tenantRepository.Create(ctx, tenant); err != nil {
        return nil, err
    }

    // 7. 初始化租户数据
    if err := s.initializeTenant(ctx, tenant); err != nil {
        // 回滚租户创建
        s.tenantRepository.Delete(ctx, tenant.TenantID)
        return nil, err
    }

    // 8. 加载配额信息
    quota, _ := s.quotaService.GetTenantQuota(ctx, tenant.TenantID)
    tenant.Quota = quota

    return tenant, nil
}

// initializeTenant 初始化租户数据
func (s *TenantService) initializeTenant(ctx context.Context, tenant *Tenant) error {
    // 1. 创建默认组织架构
    if err := s.createDefaultOrganization(ctx, tenant); err != nil {
        return err
    }

    // 2. 创建管理员用户
    adminUser, err := s.createAdminUser(ctx, tenant)
    if err != nil {
        return err
    }

    // 3. 分配默认角色
    if err := s.assignDefaultRole(ctx, adminUser); err != nil {
        return err
    }

    // 4. 初始化配额
    if err := s.quotaService.InitializeTenantQuota(ctx, tenant.TenantID, tenant.PlanID); err != nil {
        return err
    }

    return nil
}

// createDefaultOrganization 创建默认组织架构
func (s *TenantService) createDefaultOrganization(ctx context.Context, tenant *Tenant) error {
    org := &Organization{
        OrgID:      generateOrgID(),
        TenantID:   tenant.TenantID,
        Name:       tenant.Name,
        ParentID:   "",
        Level:      1,
        Path:       "/" + tenant.Name,
        CreatedAt:  time.Now(),
    }

    return s.organizationRepository.Create(ctx, org)
}

// createAdminUser 创建管理员用户
func (s *TenantService) createAdminUser(ctx context.Context, tenant *Tenant) (*User, error) {
    password := generateRandomPassword(16)

    user := &User{
        UserID:     generateUserID(),
        TenantID:   tenant.TenantID,
        Username:   "admin",
        Email:      tenant.ContactEmail,
        Phone:      tenant.ContactPhone,
        Password:   hashPassword(password),
        Nickname:   tenant.ContactPerson,
        RoleID:     "", // 将在assignDefaultRole中设置
        Status:     "active",
        CreatedAt:  time.Now(),
    }

    if err := s.userRepository.Create(ctx, user); err != nil {
        return nil, err
    }

    // 发送欢迎邮件(包含临时密码)
    s.sendWelcomeEmail(tenant.ContactEmail, password)

    return user, nil
}

// assignDefaultRole 分配默认角色
func (s *TenantService) assignDefaultRole(ctx context.Context, user *User) error {
    // 获取租户的超级管理员角色
    role, err := s.roleService.GetSuperAdminRole(ctx, user.TenantID)
    if err != nil {
        return err
    }

    user.RoleID = role.RoleID
    return s.userRepository.Update(ctx, user)
}
```

```go
type tenantRepositoryImpl struct {
    db *gorm.DB
}

func (r *tenantRepositoryImpl) Create(ctx context.Context, tenant *Tenant) error {
    return r.db.WithContext(ctx).Create(tenant).Error
}

func (r *tenantRepositoryImpl) GetByID(ctx context.Context, id string) (*Tenant, error) {
    var tenant Tenant
    err := r.db.WithContext(ctx).
        Where("tenant_id = ?", id).
        First(&tenant).Error
    if err != nil {
        return nil, err
    }
    return &tenant, nil
}

func (r *tenantRepositoryImpl) GetBySubdomain(ctx context.Context, subdomain string) (*Tenant, error) {
    var tenant Tenant
    err := r.db.WithContext(ctx).
        Where("subdomain = ?", subdomain).
        First(&tenant).Error
    if err != nil {
        return nil, err
    }
    return &tenant, nil
}

func (r *tenantRepositoryImpl) ExistsBySubdomain(ctx context.Context, subdomain string) (bool, error) {
    var count int64
    err := r.db.WithContext(ctx).
        Model(&Tenant{}).
        Where("subdomain = ?", subdomain).
        Count(&count).Error
    return count > 0, err
}
```

```go
// CheckUserQuota 检查用户配额
func (s *QuotaService) CheckUserQuota(ctx context.Context, tenantID string) error {
    quota, err := s.GetTenantQuota(ctx, tenantID)
    if err != nil {
        return err
    }

    if quota.Users.Used >= quota.Users.Limit {
        return ErrQuotaExceeded{QuotaType: "user", Limit: quota.Users.Limit}
    }

    return nil
}

// IncrementUserUsage 增加用户使用量
func (s *QuotaService) IncrementUserUsage(ctx context.Context, tenantID string) error {
    // 使用Redis原子操作确保并发安全
    key := fmt.Sprintf("quota:%s:users", tenantID)

    pipe := s.redisClient.Pipeline()
    pipe.Incr(ctx, key)
    pipe.Expire(ctx, key, 24*time.Hour)

    results, err := pipe.Exec(ctx)
    if err != nil {
        return err
    }

    newUsed := results[0].(*redis.IntCmd).Val()

    // 异步同步到数据库
    go func() {
        s.quotaRepository.UpdateUsersUsed(context.Background(), tenantID, int(newUsed))
    }()

    return nil
}
```

```go
// Login 用户登录
func (s *UserService) Login(ctx context.Context, req *LoginRequest) (*LoginResponse, error) {
    // 1. 参数校验
    if req.Username == "" || req.Password == "" {
        return nil, ErrInvalidCredentials
    }

    // 2. 查询用户
    user, err := s.userRepo.GetByUsername(ctx, req.Username)
    if err != nil {
        if errors.Is(err, gorm.ErrRecordNotFound) {
            return nil, ErrInvalidCredentials
        }
        return nil, err
    }

    // 3. 检查用户状态
    if user.Status != "active" {
        return nil, ErrUserInactive
    }

    // 4. 验证密码
    if err := s.validatePassword(user, req.Password); err != nil {
        return nil, ErrInvalidCredentials
    }

    // 5. 加载用户角色和权限
    role, err := s.roleService.GetByID(ctx, user.RoleID)
    if err != nil {
        return nil, err
    }
    user.Role = role

    // 6. 生成JWT Token
    tokens, err := s.generateTokens(ctx, user)
    if err != nil {
        return nil, err
    }

    // 7. 记录登录日志
    s.loginLogRepository.Create(ctx, &LoginLog{
        UserID:    user.UserID,
        TenantID:  user.TenantID,
        IP:        req.ClientIP,
        UserAgent: req.UserAgent,
        LoginAt:   time.Now(),
    })

    return &LoginResponse{
        AccessToken:  tokens.AccessToken,
        RefreshToken: tokens.RefreshToken,
        TokenType:    "Bearer",
        ExpiresIn:    3600,
        User:         user,
    }, nil
}

// validatePassword 验证密码
func (s *UserService) validatePassword(user *User, password string) error {
    // 使用bcrypt验证密码
    err := bcrypt.CompareHashAndPassword([]byte(user.Password), []byte(password))
    if err != nil {
        return err
    }
    return nil
}

// generateTokens 生成JWT Token
func (s *UserService) generateTokens(ctx context.Context, user *User) (*TokenPair, error) {
    // 生成Access Token (1小时有效期)
    accessClaims := jwt.MapClaims{
        "user_id":   user.UserID,
        "tenant_id": user.TenantID,
        "role_id":   user.RoleID,
        "exp":       time.Now().Add(time.Hour * 1).Unix(),
        "type":      "access",
    }
    accessToken, err := jwt.NewWithClaims(jwt.SigningMethodHS256, accessClaims).
        SignedString([]byte(s.jwtSecret))
    if err != nil {
        return nil, err
    }

    // 生成Refresh Token (30天有效期)
    refreshClaims := jwt.MapClaims{
        "user_id":   user.UserID,
        "tenant_id": user.TenantID,
        "exp":       time.Now().Add(time.Hour * 24 * 30).Unix(),
        "type":      "refresh",
    }
    refreshToken, err := jwt.NewWithClaims(jwt.SigningMethodHS256, refreshClaims).
        SignedString([]byte(s.jwtSecret))
    if err != nil {
        return nil, err
    }

    // 存储Refresh Token到Redis
    key := fmt.Sprintf("refresh_token:%s", user.UserID)
    s.redisClient.Set(ctx, key, refreshToken, 30*24*time.Hour)

    return &TokenPair{
        AccessToken:  accessToken,
        RefreshToken: refreshToken,
    }, nil
}
```

```go
// Publish 发布Bot
func (s *BotService) Publish(ctx context.Context, botID string, note string) error {
    // 1. 获取Bot
    bot, err := s.botRepo.GetByID(ctx, botID)
    if err != nil {
        return err
    }

    // 2. 检查Bot状态
    if bot.Status == "published" {
        return ErrBotAlreadyPublished
    }

    // 3. 验证LLM配置
    if err := s.validateLLMConfig(ctx, bot.LLMConfig); err != nil {
        return err
    }

    // 4. 验证知识库
    for _, kbID := range bot.KnowledgeBaseIDs {
        kb, err := s.kbService.GetByID(ctx, kbID)
        if err != nil {
            return fmt.Errorf("知识库%s不存在", kbID)
        }
        if kb.Status != "ready" {
            return fmt.Errorf("知识库%s未就绪", kbID)
        }
    }

    // 5. 验证工作流(如果有)
    if bot.WorkflowID != "" {
        workflow, err := s.workflowService.GetByID(ctx, bot.WorkflowID)
        if err != nil {
            return err
        }
        if workflow.Status != "published" {
            return ErrWorkflowNotPublished
        }
    }

    // 6. 创建Bot版本
    version, err := s.createBotVersion(ctx, bot)
    if err != nil {
        return err
    }

    // 7. 更新Bot状态
    bot.Status = "published"
    bot.Version = version.Version
    bot.PublishedAt = &time.Time{}
    *bot.PublishedAt = time.Now()

    if err := s.botRepo.Update(ctx, bot); err != nil {
        return err
    }

    // 8. 清除缓存
    s.clearBotCache(ctx, botID)

    return nil
}
```

```go
// UploadDocument 上传文档
func (s *KnowledgeBaseService) UploadDocument(ctx context.Context, kbID string, file *multipart.FileHeader, metadata map[string]string) (*Document, error) {
    // 1. 检查知识库是否存在
    kb, err := s.kbRepo.GetByID(ctx, kbID)
    if err != nil {
        return nil, err
    }

    // 2. 存储文件到MinIO
    fileKey := fmt.Sprintf("kb/%s/%s", kbID, file.Filename)
    reader, err := file.Open()
    if err != nil {
        return nil, err
    }
    defer reader.Close()

    fileURL, err := s.fileStorageSvc.Upload(ctx, fileKey, reader, file.Header.Get("Content-Type"))
    if err != nil {
        return nil, err
    }

    // 3. 创建文档记录
    doc := &Document{
        DocumentID: generateDocumentID(),
        KBID:       kbID,
        FileName:   file.Filename,
        FileURL:    fileURL,
        FileSize:   file.Size,
        Status:     "processing",
        Metadata:   metadata,
        CreatedAt:  time.Now(),
    }

    if err := s.docRepo.Create(ctx, doc); err != nil {
        return nil, err
    }

    // 4. 异步处理文档(分块、向量化、索引)
    go s.processDocument(context.Background(), doc)

    return doc, nil
}

// processDocument 处理文档
func (s *KnowledgeBaseService) processDocument(ctx context.Context, doc *Document) error {
    // 1. 下载文件
    reader, err := s.fileStorageSvc.Download(ctx, doc.FileURL)
    if err != nil {
        return err
    }
    defer reader.Close()

    // 2. 提取文本(根据文件类型)
    text, err := s.extractText(reader, doc.FileName)
    if err != nil {
        return err
    }

    // 3. 文本分块
    chunks := s.splitText(text)

    // 4. 向量化
    vectors, err := s.embeddingSvc.EmbedChunks(ctx, chunks)
    if err != nil {
        return err
    }

    // 5. 索引到Milvus
    chunkIDs := make([]string, len(chunks))
    for i := range chunks {
        chunkIDs[i] = generateChunkID()
    }

    if err := s.milvusSvc.InsertChunks(ctx, doc.KBID, chunkIDs, chunks, vectors); err != nil {
        return err
    }

    // 6. 更新文档状态
    doc.Status = "ready"
    doc.ChunksCount = len(chunks)
    s.docRepo.Update(ctx, doc)

    return nil
}

// splitText 文本分块
func (s *KnowledgeBaseService) splitText(text string) []string {
    // 按段落分块
    paragraphs := strings.Split(text, "\n\n")

    var chunks []string
    currentChunk := ""

    for _, paragraph := range paragraphs {
        if len(currentChunk) + len(paragraph) > s.chunkSize {
            if currentChunk != "" {
                chunks = append(chunks, currentChunk)
            }
            currentChunk = paragraph
        } else {
            currentChunk += paragraph + "\n\n"
        }
    }

    if currentChunk != "" {
        chunks = append(chunks, currentChunk)
    }

    return chunks
}

// Search 语义检索
func (s *KnowledgeBaseService) Search(ctx context.Context, kbID string, query string, topK int, threshold float64) (*SearchResults, error) {
    // 1. 查询向量化
    queryVector, err := s.embeddingSvc.EmbedQuery(ctx, query)
    if err != nil {
        return nil, err
    }

    // 2. Milvus向量检索
    searchResults, err := s.milvusSvc.Search(ctx, kbID, queryVector, topK)
    if err != nil {
        return nil, err
    }

    // 3. 过滤低分结果
    var filteredResults []SearchResult
    for _, result := range searchResults {
        if result.Score >= threshold {
            filteredResults = append(filteredResults, result)
        }
    }

    // 4. 加载文档元数据
    for i := range filteredResults {
        doc, _ := s.docRepo.GetByID(ctx, filteredResults[i].DocumentID)
        if doc != nil {
            filteredResults[i].DocumentName = doc.FileName
            filteredResults[i].Metadata = doc.Metadata
        }
    }

    return &SearchResults{
        Query:   query,
        Results: filteredResults,
    }, nil
}
```

### 7.2 租户识别策略

#### 7.2.1 识别方式

支持三种租户识别方式:

1. **子域名方式** (推荐)
   - URL: `https://{tenant_subdomain}.saas.coze.com/...`
   - 优先级: 最高

2. **路径方式**
   - URL: `https://saas.coze.com/t/{tenant_id}/...`
   - 优先级: 中等

3. **Header方式** (用于API调用)
   - Header: `X-Tenant-ID: {tenant_id}`
   - 优先级: 最低

#### 7.2.2 TenantMiddleware实现

```go
// TenantMiddleware 租户中间件
func TenantMiddleware() app.HandlerFunc {
    return func(ctx context.Context, c *app.RequestContext) {
        tenantID, err := extractTenantID(c)
        if err != nil {
            c.JSON(400, H{"code": 40001, "message": "无法识别租户"})
            c.Abort()
            return
        }

        // 验证租户是否存在
        tenant, err := tenantService.GetByID(ctx, tenantID)
        if err != nil {
            c.JSON(404, H{"code": 40401, "message": "租户不存在"})
            c.Abort()
            return
        }

        // 检查租户状态
        if tenant.Status != "active" {
            c.JSON(403, H{"code": 40301, "message": "租户已被停用"})
            c.Abort()
            return
        }

        // 注入租户上下文
        ctx = context.WithValue(ctx, "tenant_id", tenantID)
        ctx = context.WithValue(ctx, "tenant", tenant)

        c.Next(ctx)
    }
}

// extractTenantID 提取租户ID
func extractTenantID(c *app.RequestContext) (string, error) {
    // 1. 优先从子域名提取
    host := string(c.Host())
    parts := strings.Split(host, ".")
    if len(parts) > 1 && parts[1] == "saas" && parts[2] == "coze" && parts[3] == "com" {
        subdomain := parts[0]
        // 通过子域名查询租户ID
        tenant, err := tenantService.GetBySubdomain(context.Background(), subdomain)
        if err == nil {
            return tenant.TenantID, nil
        }
    }

    // 2. 从路径提取
    path := string(c.Path())
    if matches := regexp.MustCompile(`^/t/([^/]+)/`).FindStringSubmatch(path); len(matches) > 1 {
        return matches[1], nil
    }

    // 3. 从Header提取
    tenantID := c.GetHeader("X-Tenant-ID")
    if tenantID != "" {
        return tenantID, nil
    }

    return "", ErrTenantNotIdentified
}
```

```go
    if startNode == nil {
        return nil, ErrNoStartNode
    }

    // 3. 执行节点链
    currentNode := startNode
    executionHistory := []NodeExecution{}

    for currentNode != nil {
        // 3.1 创建节点实例
        node, err := e.createNode(currentNode)
        if err != nil {
            return nil, err
        }

        // 3.2 执行节点
        startTime := time.Now()
        result, err := node.Execute(execCtx)
        duration := time.Since(startTime)

        // 3.3 记录执行历史
        executionHistory = append(executionHistory, NodeExecution{
            NodeID:   currentNode.ID,
            NodeName: currentNode.Name,
            Status:   result.Status,
            Duration: duration,
            Error:    err,
        })

        if err != nil {
            return &ExecutionResult{
                Status:          "failed",
                ExecutionHistory: executionHistory,
                Error:           err,
            }, err
        }

        // 3.4 设置输出变量到上下文
        for k, v := range result.Output {
            execCtx.SetVariable(k, v)
        }

        // 3.5 确定下一个节点
        if result.NextNodeID != "" {
            // 节点内部指定了下一个节点(如条件节点)
            currentNode = e.findNodeByID(workflow, result.NextNodeID)
        } else {
            // 根据边查找下一个节点
            currentNode = e.findNextNode(workflow, currentNode.ID, execCtx)
        }

        // 3.6 检查是否到达结束节点
        if currentNode != nil && currentNode.Type == "end" {
            break
        }
    }

    return &ExecutionResult{
        Status:          "success",
        Output:          execCtx.GetAllVariables(),
        ExecutionHistory: executionHistory,
    }, nil
}

// findNextNode 查找下一个节点
func (e *WorkflowEngine) findNextNode(workflow *Workflow, currentNodeID string, execCtx *ExecutionContext) *Node {
    // 查找从当前节点出发的边
    for _, edge := range workflow.Definition.Edges {
        if edge.Source == currentNodeID {
            // 如果有条件,评估条件
            if edge.Condition != nil {
                if !execCtx.EvaluateCondition(edge.Condition.Expression) {
                    continue // 条件不满足,跳过这条边
                }
            }

            // 返回目标节点
            return e.findNodeByID(workflow, edge.Target)
        }
    }
    return nil
}
```

```go
// ModelProvider AI模型提供者接口
type ModelProvider interface {
    // Chat 单次对话
    Chat(ctx context.Context, req *ChatRequest) (*ChatResponse, error)

    // ChatStream 流式对话
    ChatStream(ctx context.Context, req *ChatRequest) (<-chan *StreamChunk, error)

    // GetModelInfo 获取模型信息
    GetModelInfo() ModelInfo
}

// QwenProvider 通义千问提供者
type QwenProvider struct {
    apiKey    string
    baseURL   string
    client    *http.Client
}

func (p *QwenProvider) Chat(ctx context.Context, req *ChatRequest) (*ChatResponse, error) {
    // 构造请求
    payload := map[string]interface{}{
        "model":       req.Model,
        "messages":    req.Messages,
        "temperature": req.Temperature,
        "max_tokens":  req.MaxTokens,
    }

    body, _ := json.Marshal(payload)
    httpReq, _ := http.NewRequestWithContext(ctx, "POST", p.baseURL+"/chat/completions", bytes.NewReader(body))
    httpReq.Header.Set("Authorization", "Bearer "+p.apiKey)
    httpReq.Header.Set("Content-Type", "application/json")

    // 发送请求
    resp, err := p.client.Do(httpReq)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    // 解析响应
    var result struct {
        Choices []struct {
            Message Message `json:"message"`
        } `json:"choices"`
        Usage struct {
            TotalTokens int `json:"total_tokens"`
        } `json:"usage"`
    }
    json.NewDecoder(resp.Body).Decode(&result)

    return &ChatResponse{
        Content: result.Choices[0].Message.Content,
        Usage: Usage{
            TotalTokens: result.Usage.TotalTokens,
        },
    }, nil
}

func (p *QwenProvider) ChatStream(ctx context.Context, req *ChatRequest) (<-chan *StreamChunk, error) {
    chunkChan := make(chan *StreamChunk)

    go func() {
        defer close(chunkChan)

        payload := map[string]interface{}{
            "model":       req.Model,
            "messages":    req.Messages,
            "temperature": req.Temperature,
            "max_tokens":  req.MaxTokens,
            "stream":      true,
        }

        body, _ := json.Marshal(payload)
        httpReq, _ := http.NewRequestWithContext(ctx, "POST", p.baseURL+"/chat/completions", bytes.NewReader(body))
        httpReq.Header.Set("Authorization", "Bearer "+p.apiKey)
        httpReq.Header.Set("Content-Type", "application/json")

        resp, err := p.client.Do(httpReq)
        if err != nil {
            chunkChan <- &StreamChunk{Error: err}
            return
        }
        defer resp.Body.Close()

        // 读取SSE流
        scanner := bufio.NewScanner(resp.Body)
        for scanner.Scan() {
            line := scanner.Text()
            if strings.HasPrefix(line, "data: ") {
                data := strings.TrimPrefix(line, "data: ")
                if data == "[DONE]" {
                    break
                }

                var chunk StreamChunk
                if err := json.Unmarshal([]byte(data), &chunk); err == nil {
                    chunkChan <- &chunk
                }
            }
        }
    }()

    return chunkChan, nil
}
```

```go
type aiServiceImpl struct {
    modelProviders map[string]ModelProvider
    modelConfigRepo ModelConfigRepository
}

func (s *aiServiceImpl) ChatStream(ctx context.Context, req *ChatRequest) (<-chan *StreamChunk, error) {
    // 1. 选择模型提供者
    provider, err := s.selectModelProvider(req.Model)
    if err != nil {
        return nil, err
    }

    // 2. 调用提供者
    chunkChan, err := provider.ChatStream(ctx, req)
    if err != nil {
        return nil, err
    }

    // 3. 包装错误处理和超时控制
    wrappedChan := make(chan *StreamChunk)
    go func() {
        defer close(wrappedChan)

        timeout := time.NewTimer(5 * time.Minute)
        defer timeout.Stop()

        for {
            select {
            case chunk, ok := <-chunkChan:
                if !ok {
                    return
                }
                if chunk.Error != nil {
                    wrappedChan <- &StreamChunk{Error: chunk.Error}
                    return
                }
                wrappedChan <- chunk
            case <-timeout.C:
                wrappedChan <- &StreamChunk{Error: ErrTimeout}
                return
            case <-ctx.Done():
                return
            }
        }
    }()

    return wrappedChan, nil
}

func (s *aiServiceImpl) selectModelProvider(model string) (ModelProvider, error) {
    // 根据模型名称选择提供者
    if strings.HasPrefix(model, "qwen") {
        return s.modelProviders["qwen"], nil
    } else if strings.HasPrefix(model, "gpt") {
        return s.modelProviders["openai"], nil
    } else if strings.HasPrefix(model, "claude") {
        return s.modelProviders["anthropic"], nil
    }
    return nil, ErrModelNotSupported
}
```
