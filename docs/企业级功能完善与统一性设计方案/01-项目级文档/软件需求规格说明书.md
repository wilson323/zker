## 1. 引言

### 1.1 编写目的

本文档是 zker Enterprise Edition 系统的软件需求规格说明书 (SRS),旨在:

### 1.2 项目背景

zker Enterprise Edition 是基于开源 zker 打造的企业级 Multi-Tenant SaaS 平台,核心特性:

✅ **Multi-Tenant SaaS**: 支持N个企业(租户)独立使用,每个企业可有大量员工
✅ **完全隔离**: 租户间数据、计算、缓存、存储100%隔离
✅ **对齐鲸智百应**: 功能完整度100%对齐
✅ **开源可定制**: 基于开源,可深度定制
✅ **成本可控**: 私有化部署,无订阅费用

#### 3.1.2 租户识别

**需求ID**: SRS-FUNC-MT-002

**功能描述**: 系统自动识别当前请求所属的租户,确保租户隔离

**识别方式**:

**方式1: 子域名识别 (优先级最高)**
```
URL: https://tenant-a.saas.coze.com/
提取: tenant-a
```

**方式2: 路径识别**
```
URL: https://saas.coze.com/tenant-a/
提取: tenant-a
```

**方式3: Header识别**
```
Header: X-Tenant-ID: tenant-a
提取: tenant-a
```

**处理逻辑**:
```go
type TenantIdentificationMiddleware struct {
    tenantService *TenantService
}

func (m *TenantIdentificationMiddleware) Handle(ctx context.Context, c *app.RequestContext) {
    var tenantID string

    // 1. 从子域名提取
    host := c.Host()
    if strings.Contains(host, ".") {
        parts := strings.Split(host, ".")
        if len(parts) > 1 && parts[1] == "saas" && parts[2] == "coze" && parts[3] == "com" {
            tenantID = parts[0]
        }
    }

    // 2. 从路径提取
    if tenantID == "" {
        path := c.Path()
        if strings.HasPrefix(path, "/t/") {
            parts := strings.Split(strings.TrimPrefix(path, "/t/"), "/")
            if len(parts) > 0 {
                tenantID = parts[0]
            }
        }
    }

    // 3. 从Header提取
    if tenantID == "" {
        tenantID = c.GetHeader("X-Tenant-ID")
    }

    // 4. 验证租户ID
    if tenantID == "" {
        c.JSON(400, map[string]string{"error": "无法识别租户"})
        c.Abort()
        return
    }

    // 5. 查询租户信息
    tenant, err := m.tenantService.GetByID(ctx, tenantID)
    if err != nil {
        c.JSON(404, map[string]string{"error": "租户不存在"})
        c.Abort()
        return
    }

    // 6. 检查租户状态

    skills, _ := s.botSkillRepo.GetByBotID(ctx, botID)
    if len(skills) == 0 {
        return errors.New("Bot至少需要1个技能")
    }

    // 4. 更新Bot状态
    bot.Status = "published"
    bot.PublishedAt = time.Now()
    bot.PublishedBy = userID
    err = s.botRepo.Update(ctx, bot)
    if err != nil {
        return err
    }

    // 5. 清除Bot缓存 (如果有)
    s.cache.Delete(fmt.Sprintf("bot:%s", botID))

    // 6. 记录审计日志
    s.auditLogService.Create(ctx, &AuditLog{
        TenantID: tenantID,
        UserID:   userID,
        Action:   "bot.publish",
        Details:  fmt.Sprintf("发布Bot: %s", bot.Name),
    })

    return nil
}
```

```go
func (s *KnowledgeBaseService) UploadDocument(ctx context.Context, tenantID string, userID string, kbID string, file *multipart.FileHeader) (*Document, error) {
    // 1. 验证知识库存在且属于当前租户
    kb, err := s.kbRepo.GetByID(ctx, kbID)
    if err != nil || kb.TenantID != tenantID {
        return nil, errors.New("知识库不存在")
    }

    // 2. 验证文件大小
    if file.Size > 100*1024*1024 { // 100MB
        return nil, errors.New("文件大小超过限制(100MB)")
    }

    // 3. 验证文件格式
    ext := filepath.Ext(file.Filename)
    allowedExts := map[string]bool{
        ".pdf": true, ".doc": true, ".docx": true,
        ".xls": true, ".xlsx": true, ".ppt": true,
        ".pptx": true, ".md": true, ".txt": true,
    }
    if !allowedExts[ext] {
        return nil, errors.New("不支持的文件格式")
    }

    // 4. 上传文件到MinIO
    objectKey := fmt.Sprintf("tenant/%s/knowledge/%s/%s", tenantID, kbID, file.Filename)
    reader, _ := file.Open()
    defer reader.Close()
    err = s.minioClient.PutObject(ctx, "documents", objectKey, reader, file.Size, minio.PutObjectOptions{
        ContentType: file.Header.Get("Content-Type"),
    })
    if err != nil {
        return nil, err
    }

    // 5. 创建文档记录
    docID := uuid.New().String()
    document := &Document{
        ID:             docID,
        TenantID:       tenantID,
        KnowledgeBaseID: kbID,
        FileName:       file.Filename,
        FileSize:       file.Size,
        FileType:       ext,
        FileURL:        fmt.Sprintf("https://storage.coze.com/documents/%s", objectKey),
        UploadedBy:     userID,
        Status:         "processing",
        CreatedAt:      time.Now(),
    }
    err = s.documentRepo.Create(ctx, document)
    if err != nil {
        return nil, err
    }

    // 6. 异步解析文档
    go func() {
        // 6.1 下载文件
        // 6.2 提取文本内容
        content, _ := s.extractor.Extract(document.FileURL)

        // 6.3 分块 (Chunking)
        chunks := s.chunker.Split(content, 500) // 每块500字符

        // 6.4 向量化
        for i, chunk := range chunks {
            embedding, _ := s.embeddingClient.Embed(chunk)

            // 6.5 存储到Milvus
            s.milvusClient.Insert(ctx, &milvus.Entity{
                CollectionName: "knowledge_chunks",
                Data: []interface{}{
                    docID,
                    i,
                    chunk,
                    embedding,
                },
            })
        }

        // 6.6 更新文档状态
        document.Status = "ready"
        document.ChunkCount = len(chunks)
        document.ProcessedAt = time.Now()
        s.documentRepo.Update(ctx, document)
    }()

    return document, nil
}
```

**输出**:
```typescript

  documentId: string;
  fileName: string;
  fileSize: number;
  status: 'processing' | 'ready' | 'failed';
}
```

- ✅ 文档解析准确率 > 95%

func (s *KnowledgeBaseService) Search(ctx context.Context, tenantID string, botID string, query string, topK int) ([]*KnowledgeChunk, error) {
    // 1. 对问题进行向量化
    queryEmbedding, err := s.embeddingClient.Embed(query)
    if err != nil {
        return nil, err
    }

    // 2. 在Milvus中进行向量检索
    searchResult, err := s.milvusClient.Search(ctx, &milvus.SearchRequest{
        CollectionName: "knowledge_chunks",
        Vectors:        [][][]float32{queryEmbedding},
        TopK:           topK,
        Filter:         fmt.Sprintf("tenant_id == '%s' && bot_id == '%s'", tenantID, botID),
    })
    if err != nil {
        return nil, err
    }

    // 3. 格式化结果
    var chunks []*KnowledgeChunk
    for _, result := range searchResult {
        chunk := &KnowledgeChunk{
            DocumentID: result.Fields["document_id"].(string),
            ChunkIndex: int(result.Fields["chunk_index"].(int32)),
            Content:    result.Fields["content"].(string),
            Score:      result.Score,
        }
        chunks = append(chunks, chunk)
    }

    return chunks, nil
}

  chunkIndex: number;
  content: string;
