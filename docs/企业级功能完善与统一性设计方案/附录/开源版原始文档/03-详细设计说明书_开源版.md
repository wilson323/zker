```typescript
import { HTTPClient } from '../client/http';
import type { Agent, CreateAgentParams, UpdateAgentParams } from '../types/agent';

export class AgentAPIImpl implements AgentAPI {

  }

    const response = await this.client.get<Agent>(`/api/v1/agents/${id}`);
    return response.data;
  }

  async updateAgent(id: string, params: UpdateAgentParams): Promise<Agent> {
    const response = await this.client.put<Agent>(`/api/v1/agents/${id}`, params);
    return response.data;
  }

    await this.client.delete(`/api/v1/agents/${id}`);
  }

  async listAgents(filters: AgentFilters): Promise<AgentListResponse> {
    const response = await this.client.get<AgentListResponse>('/api/v1/agents', {
      params: filters
    });
    return response.data;
  }

  async publishAgent(id: string, params: PublishParams): Promise<PublishInfo> {
    const response = await this.client.post<PublishInfo>(
      `/api/v1/agents/${id}/publish`,
      params
    );
    return response.data;
  }
```

```typescript
// src/client/http.ts
export interface HTTPClientConfig {
  baseURL: string;
  timeout?: number;
  headers?: Record<string, string>;
}

export interface RequestConfig {
  headers?: Record<string, string>;
  params?: Record<string, any>;
  timeout?: number;
}

export class HTTPClient {
  private config: HTTPClientConfig;
  private defaultHeaders: Record<string, string>;

  constructor(config: HTTPClientConfig) {
    this.config = config;
    this.defaultHeaders = {
      'Content-Type': 'application/json',
      'Accept': 'application/json',
      ...config.headers
    };
  }

  /**
   * 设置认证 Token
   */
  setAuthToken(token: string) {
    this.defaultHeaders['Authorization'] = `Bearer ${token}`;
  }

  /**
   * 清除认证 Token
   */
  clearAuthToken() {
    delete this.defaultHeaders['Authorization'];
  }

  /**
   * GET 请求
   */
  async get<T>(url: string, config?: RequestConfig): Promise<ApiResponse<T>> {
    return this.request<T>('GET', url, undefined, config);
  }

  /**
   * POST 请求
   */
  async post<T>(url: string, data?: any, config?: RequestConfig): Promise<ApiResponse<T>> {
    return this.request<T>('POST', url, data, config);
  }

  /**
   * PUT 请求
   */
  async put<T>(url: string, data?: any, config?: RequestConfig): Promise<ApiResponse<T>> {
    return this.request<T>('PUT', url, data, config);
  }

  /**
   * DELETE 请求
   */
  async delete<T>(url: string, config?: RequestConfig): Promise<ApiResponse<T>> {
    return this.request<T>('DELETE', url, undefined, config);
  }

  /**
   * 通用请求方法
   */
  private async request<T>(
    method: string,
    url: string,
    data?: any,
    config?: RequestConfig
  ): Promise<ApiResponse<T>> {
    const requestId = this.generateRequestId();

    try {
      const startTime = Date.now();

      const response = await fetch(`${this.config.baseURL}${url}`, {
        method,
        headers: {
          ...this.defaultHeaders,
          'X-Request-ID': requestId,
          ...config?.headers
        },
        body: data ? JSON.stringify(data) : undefined,
        params: config?.params
      });

      const duration = Date.now() - startTime;

      if (!response.ok) {
        throw new APIError(response.status, response.statusText, requestId);
      }

      const result = await response.json();

      // 记录性能日志
      this.logPerformance(method, url, duration, response.status);

      return result;
    } catch (error) {
      // 错误处理
      this.logError(method, url, error, requestId);
      throw error;
    }
  }

  private generateRequestId(): string {
    return `req_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  }

  private logPerformance(method: string, url: string, duration: number, status: number) {
    // 性能日志记录
    console.log(`[HTTP] ${method} ${url} - ${status} - ${duration}ms`);
  }

  private logError(method: string, url: string, error: Error, requestId: string) {
    // 错误日志记录
    console.error(`[HTTP Error] ${method} ${url} - ${requestId}`, error);
  }
}

export interface ApiResponse<T> {
  code: number;
  message: string;
  data: T;
  requestId: string;
}

export class APIError extends Error {
  constructor(
    public status: number,
    message: string,
    public requestId: string
  ) {
    super(message);
    this.name = 'APIError';
  }
}
```

```typescript
  token: string | null;

// ============================================
// 空间 Store
// ============================================
interface SpaceState {
  currentSpace: Space | null;
  spaces: Space[];

  // Actions
  setCurrentSpace: (space: Space) => void;
  setSpaces: (spaces: Space[]) => void;
  addSpace: (space: Space) => void;
  updateSpace: (id: string, updates: Partial<Space>) => void;
  removeSpace: (id: string) => void;
}

export const useSpaceStore = create<SpaceState>()(
  devtools(
    (set, get) => ({
      currentSpace: null,
      spaces: [],

      setCurrentSpace: (space) => set({ currentSpace: space }),

      setSpaces: (spaces) => set({ spaces }),

      addSpace: (space) => set((state) => ({ spaces: [...state.spaces, space] })),

      updateSpace: (id, updates) =>
        set((state) => ({
          spaces: state.spaces.map((s) => (s.id === id ? { ...s, ...updates } : s)),
          currentSpace:
            state.currentSpace?.id === id
              ? { ...state.currentSpace, ...updates }
              : state.currentSpace
        })),

      removeSpace: (id) =>
        set((state) => ({
          spaces: state.spaces.filter((s) => s.id !== id),
          currentSpace: state.currentSpace?.id === id ? null : state.currentSpace
        }))
    }),
    { name: 'SpaceStore' }
  )
);

// ============================================
// Agent Store
// ============================================
interface AgentState {
  agents: Record<string, Agent>; // 按 ID 索引
  agentList: string[]; // ID 列表（保持顺序）
  currentAgent: Agent | null;
  loading: boolean;
  error: string | null;

  // Actions
  setAgents: (agents: Agent[]) => void;
  setCurrentAgent: (agent: Agent | null) => void;
  addAgent: (agent: Agent) => void;
  updateAgent: (id: string, updates: Partial<Agent>) => void;
  removeAgent: (id: string) => void;
  setLoading: (loading: boolean) => void;
  setError: (error: string | null) => void;
}

export const useAgentStore = create<AgentState>()(
  devtools(
    (set, get) => ({
      agents: {},
      agentList: [],
      currentAgent: null,
      loading: false,
      error: null,

      setAgents: (agents) =>
        set({
          agents: Object.fromEntries(agents.map((a) => [a.id, a])),
          agentList: agents.map((a) => a.id)
        }),

      setCurrentAgent: (agent) => set({ currentAgent: agent }),

      addAgent: (agent) =>
        set((state) => ({
          agents: { ...state.agents, [agent.id]: agent },
          agentList: [agent.id, ...state.agentList]
        })),

      updateAgent: (id, updates) =>
        set((state) => {
          const existing = state.agents[id];
          if (!existing) return state;

          return {
            agents: {
              ...state.agents,
              [id]: { ...existing, ...updates, updatedAt: new Date().toISOString() }
            },
            currentAgent:
              state.currentAgent?.id === id
                ? { ...state.currentAgent, ...updates, updatedAt: new Date().toISOString() }
                : state.currentAgent
          };
        }),

      removeAgent: (id) =>
        set((state) => {
          const newAgents = { ...state.agents };
          delete newAgents[id];

          return {
            agents: newAgents,
            agentList: state.agentList.filter((agentId) => agentId !== id),
            currentAgent: state.currentAgent?.id === id ? null : state.currentAgent
          };
        }),

      setLoading: (loading) => set({ loading }),
      setError: (error) => set({ error })
    }),
    { name: 'AgentStore' }
  )
);

// 对话 Store

interface ConversationState {
  conversations: Record<string, Conversation>;
  currentConversationId: string | null;
  messages: Record<string, Message[]>; // conversationId -> messages
  loading: boolean;
  streaming: boolean;

  // Actions
  setConversations: (conversations: Conversation[]) => void;
  setCurrentConversation: (id: string | null) => void;
  addConversation: (conversation: Conversation) => void;
  setMessages: (conversationId: string, messages: Message[]) => void;
  addMessage: (conversationId: string, message: Message) => void;
  updateMessage: (conversationId: string, messageId: string, updates: Partial<Message>) => void;
  setLoading: (loading: boolean) => void;
  setStreaming: (streaming: boolean) => void;
}

export const useConversationStore = create<ConversationState>()(
  devtools(
    (set) => ({
      conversations: {},
      currentConversationId: null,
      messages: {},
      loading: false,
      streaming: false,

      setConversations: (conversations) =>
        set({
          conversations: Object.fromEntries(conversations.map((c) => [c.id, c]))
        }),

      setCurrentConversation: (id) => set({ currentConversationId: id }),

      addConversation: (conversation) =>
        set((state) => ({
          conversations: {
            ...state.conversations,
            [conversation.id]: conversation
          },
          messages: {
            ...state.messages,
            [conversation.id]: []
          }
        })),

      setMessages: (conversationId, messages) =>
        set((state) => ({
          messages: { ...state.messages, [conversationId]: messages }
        })),

      addMessage: (conversationId, message) =>
        set((state) => ({
          messages: {
            ...state.messages,
            [conversationId]: [...(state.messages[conversationId] || []), message]
          }
        })),

      updateMessage: (conversationId, messageId, updates) =>
        set((state) => {
          const messages = state.messages[conversationId] || [];
          return {
            messages: {
              ...state.messages,
              [conversationId]: messages.map((m) =>
                m.id === messageId ? { ...m, ...updates } : m
              )
            }
          };
        }),

      setLoading: (loading) => set({ loading }),
      setStreaming: (streaming) => set({ streaming })
    }),
    { name: 'ConversationStore' }
  )
);
```

```typescript
// src/components/WorkflowCanvas.tsx
import React, { useRef, useEffect, useState } from 'react';
import { WorkflowEditor } from '@flowgram-adapter/free-layout-editor';
import type { Node, Edge, Workflow } from '@coze-workflow/base';

interface WorkflowCanvasProps {
  workflow: Workflow;
  readonly?: boolean;
  onNodesChange?: (nodes: Node[]) => void;
  onEdgesChange?: (edges: Edge[]) => void;
  onSave?: () => void;
  onExecute?: () => void;
}

export const WorkflowCanvas: React.FC<WorkflowCanvasProps> = ({
  workflow,
  readonly = false,
  onNodesChange,
  onEdgesChange,
  onSave,
  onExecute
}) => {
  const editorRef = useRef<WorkflowEditor>(null);
  const [hasUnsavedChanges, setHasUnsavedChanges] = useState(false);

  // 自动保存（30秒间隔）
  useEffect(() => {
    if (!readonly) {
      const interval = setInterval(() => {
        if (hasUnsavedChanges && onSave) {
          onSave();
          setHasUnsavedChanges(false);
        }
      }, 30000);

      return () => clearInterval(interval);
    }
  }, [hasUnsavedChanges, readonly, onSave]);

  // 处理节点变化
  const handleNodesChange = (nodes: Node[]) => {
    setHasUnsavedChanges(true);
    onNodesChange?.(nodes);
  };

  // 处理连线变化
  const handleEdgesChange = (edges: Edge[]) => {
    setHasUnsavedChanges(true);
    onEdgesChange?.(edges);
  };

  // 工具栏操作
  const toolbarActions = {
    undo: () => editorRef.current?.undo(),
    redo: () => editorRef.current?.redo(),
    delete: () => editorRef.current?.deleteSelected(),
    copy: () => editorRef.current?.copySelected(),
    paste: () => editorRef.current?.paste(),
    selectAll: () => editorRef.current?.selectAll(),
    zoomIn: () => editorRef.current?.zoomIn(),
    zoomOut: () => editorRef.current?.zoomOut(),
    fitView: () => editorRef.current?.fitView(),
    format: () => editorRef.current?.formatGraph(),
    validate: () => editorRef.current?.validate()
  };

  return (
    <div className="workflow-canvas">
      {/* 工具栏 */}
      {!readonly && (
        <WorkflowToolbar
          actions={toolbarActions}
          onSave={onSave}
          onExecute={onExecute}
          hasUnsavedChanges={hasUnsavedChanges}
        />
      )}

      {/* 节点面板 */}
      {!readonly && <NodePalette />}

      {/* 画布编辑器 */}
      <WorkflowEditor
        ref={editorRef}
        workflow={workflow}
        readonly={readonly}
        onNodesChange={handleNodesChange}
        onEdgesChange={handleEdgesChange}
        minZoom={0.1}
        maxZoom={2}
        defaultZoom={1}
      />

      {/* 属性面板 */}
      <PropertyPanel />
    </div>
  );
};
```

```go
		WithOpeningRemark(req.OpeningRemark).
		WithModelConfig(s.toDomainModelConfig(req.ModelConfig)).
		WithMemoryConfig(s.toDomainMemoryConfig(req.MemoryConfig)).
		WithPrivacyConfig(s.toDomainPrivacyConfig(req.PrivacyConfig)).
		WithCreatedBy(agent.UserID(req.CreatedBy))

	// 3. 关联知识库
	for _, kbID := range req.KnowledgeBases {
		kb, err := s.knowledgeRepo.FindByID(ctx, knowledge.KnowledgeBaseID(kbID))
		if err != nil {
			return nil, fmt.Errorf("knowledge base not found: %s", kbID)
		}
		if kb.SpaceID() != agent.SpaceID(req.SpaceID) {
			return nil, fmt.Errorf("knowledge base does not belong to this space")
		}
		agentBuilder.WithKnowledgeBase(knowledge.KnowledgeBaseID(kbID))
	}

	// 4. 关联插件
	for _, pluginID := range req.Plugins {
		plugin, err := s.pluginRepo.FindByID(ctx, plugin.PluginID(pluginID))
		if err != nil {
			return nil, fmt.Errorf("plugin not found: %s", pluginID)
		}
		agentBuilder.WithPlugin(plugin.PluginID(pluginID))
	}

	// 5. 关联工作流
	for _, wfID := range req.Workflows {
		wf, err := s.workflowRepo.FindByID(ctx, workflow.WorkflowID(wfID))
		if err != nil {
			return nil, fmt.Errorf("workflow not found: %s", wfID)
		}
		agentBuilder.WithWorkflow(workflow.WorkflowID(wfID))
	}

	// 6. 添加变量
	for _, v := range req.Variables {
		variable := agent.NewVariable(
			agent.VariableName(v.Name),
			agent.VariableValue(v.Value),
			agent.VariableType(v.Type),
			agent.IsSecret(v.IsSecret),
		)
		agentBuilder.WithVariable(variable)
	}

	// 7. 构建 Agent
	bot, err := agentBuilder.Build()
	if err != nil {
		return nil, fmt.Errorf("build agent failed: %w", err)
	}

	// 8. 业务规则验证
	if err := s.agentRepo.ValidateAgentName(ctx, bot); err != nil {
		return nil, fmt.Errorf("agent name validation failed: %w", err)
	}

	// 9. 持久化
	if err := s.agentRepo.Save(ctx, bot); err != nil {
		return nil, fmt.Errorf("save agent failed: %w", err)
	}

	// 10. 发布领域事件
	evt := &agent.AgentCreatedEvent{
		AgentID:    bot.ID(),
		SpaceID:    bot.SpaceID(),
		Name:       bot.Name(),
		CreatedBy:  bot.CreatedBy(),
		CreatedAt:  bot.CreatedAt(),
	}
	if err := s.eventBus.Publish(ctx, evt); err != nil {
		// 记录日志但不影响主流程
		// log.Error("publish event failed", "error", err)
	}

	return bot, nil
}

// UpdateAgent 更新 Agent
func (s *Service) UpdateAgent(ctx context.Context, id string, req *UpdateAgentRequest) (*agent.Agent, error) {
	// 1. 查询现有 Agent
	bot, err := s.agentRepo.FindByID(ctx, agent.AgentID(id))
	if err != nil {
		return nil, fmt.Errorf("agent not found: %w", err)
	}

	// 2. 应用更新
	if req.Name != nil {
		if err := bot.SetName(*req.Name); err != nil {
			return nil, err
		}
	}
	if req.Description != nil {
		bot.SetDescription(*req.Description)
	}
	if req.Avatar != nil {
		bot.SetAvatar(*req.Avatar)
	}
	if req.SystemPrompt != nil {
		bot.SetSystemPrompt(*req.SystemPrompt)
	}
	if req.OpeningRemark != nil {
		bot.SetOpeningRemark(*req.OpeningRemark)
	}
	if req.ModelConfig != nil {
		if err := bot.SetModelConfig(s.toDomainModelConfig(*req.ModelConfig)); err != nil {
			return nil, err
		}
	}
	if req.MemoryConfig != nil {
		bot.SetMemoryConfig(s.toDomainMemoryConfig(*req.MemoryConfig))
	}
	if req.PrivacyConfig != nil {
		bot.SetPrivacyConfig(s.toDomainPrivacyConfig(*req.PrivacyConfig))
	}

	// 3. 验证
	if err := bot.Validate(); err != nil {
		return nil, fmt.Errorf("validation failed: %w", err)
	}

	// 4. 持久化
	if err := s.agentRepo.Update(ctx, bot); err != nil {
		return nil, fmt.Errorf("update agent failed: %w", err)
	}

	// 5. 发布事件
	evt := &agent.AgentUpdatedEvent{
		AgentID:   bot.ID(),
		UpdatedAt: bot.UpdatedAt(),
	}
	s.eventBus.Publish(ctx, evt)

	return bot, nil
}

// DeleteAgent 删除 Agent
func (s *Service) DeleteAgent(ctx context.Context, id string) error {
	bot, err := s.agentRepo.FindByID(ctx, agent.AgentID(id))
	if err != nil {
		return fmt.Errorf("agent not found: %w", err)
	}

	// 检查是否已发布
	if bot.Status() == agent.StatusPublished {
		return fmt.Errorf("cannot delete published agent")
	}

	if err := s.agentRepo.Delete(ctx, bot.ID()); err != nil {
		return fmt.Errorf("delete agent failed: %w", err)
	}

	// 发布事件
	evt := &agent.AgentDeletedEvent{
		AgentID: bot.ID(),
	}
	s.eventBus.Publish(ctx, evt)

	return nil
}

// PublishAgent 发布 Agent
func (s *Service) PublishAgent(ctx context.Context, id string, version string) error {
	bot, err := s.agentRepo.FindByID(ctx, agent.AgentID(id))
	if err != nil {
		return fmt.Errorf("agent not found: %w", err)
	}

	// 验证状态
	if err := bot.Validate(); err != nil {
		return fmt.Errorf("agent validation failed: %w", err)
	}

	// 发布
	if err := bot.Publish(version); err != nil {
		return fmt.Errorf("publish failed: %w", err)
	}

	if err := s.agentRepo.Update(ctx, bot); err != nil {
		return fmt.Errorf("update agent failed: %w", err)
	}

	// 发布事件
	evt := &agent.AgentPublishedEvent{
		AgentID:  bot.ID(),
		Version:  version,
		PublishedAt: bot.UpdatedAt(),
	}
	s.eventBus.Publish(ctx, evt)

	return nil
}
```

```go
// application/agent/validator.go
package agent

import (
	"context"
	"fmt"
	"regexp"
	"unicode/utf8"
)

type Validator struct {
	nameRegexp *regexp.Regexp
}

func NewValidator() *Validator {
	return &Validator{
		nameRegexp: regexp.MustCompile(`^[a-zA-Z0-9_\u4e00-\u9fa5]{3,50}$`),
	}
}

func (v *Validator) ValidateCreateAgentRequest(ctx context.Context, req *CreateAgentRequest) error {
	// 验证名称
	if req.Name == "" {
		return fmt.Errorf("name is required")
	}
	if !v.nameRegexp.MatchString(req.Name) {
		return fmt.Errorf("name must be 3-50 characters, only letters, numbers, underscore and Chinese")
	}

	// 验证系统提示词
	if req.SystemPrompt == "" {
		return fmt.Errorf("system prompt is required")
	}
	if utf8.RuneCountInString(req.SystemPrompt) > 10000 {
		return fmt.Errorf("system prompt too long (max 10000 characters)")
	}

	// 验证模型配置
	if req.ModelConfig.Model == "" {
		return fmt.Errorf("model is required")
	}
	if req.ModelConfig.Temperature < 0 || req.ModelConfig.Temperature > 2 {
		return fmt.Errorf("temperature must be between 0 and 2")
	}
	if req.ModelConfig.MaxTokens < 1 || req.ModelConfig.MaxTokens > 128000 {
		return fmt.Errorf("max tokens must be between 1 and 128000")
	}

	// 验证变量名
	varNames := make(map[string]bool)
	for _, variable := range req.Variables {
		if variable.Name == "" {
			return fmt.Errorf("variable name is required")
		}
		if varNames[variable.Name] {
			return fmt.Errorf("duplicate variable name: %s", variable.Name)
		}
		varNames[variable.Name] = true
	}

	return nil
}
```

## 4. 核心算法设计

### 4.1 对话记忆算法

**滑动窗口算法：**

```go
// pkg/conversation/memory/sliding_window.go
package memory

import (
	"container/list"
	"context"
)

// SlidingWindowMemory 滑动窗口记忆
type SlidingWindowMemory struct {
	maxTurns    int           // 最大轮次
	window      *list.List    // 消息窗口
	retentionDays int         // 保留天数
}

// NewSlidingWindowMemory 创建滑动窗口记忆
func NewSlidingWindowMemory(maxTurns, retentionDays int) *SlidingWindowMemory {
	return &SlidingWindowMemory{
		maxTurns:      maxTurns,
		window:        list.New(),
		retentionDays: retentionDays,
	}
}

// AddMessage 添加消息到记忆
func (m *SlidingWindowMemory) AddMessage(ctx context.Context, msg *Message) error {
	// 添加到窗口
	m.window.PushBack(msg)

	// 如果超过最大轮次，移除最旧的消息
	for m.window.Len() > m.maxTurns*2 { // 1轮 = 用户消息 + 助手消息
		m.window.Remove(m.window.Front())
	}

	return nil
}

// GetHistory 获取历史消息
func (m *SlidingWindowMemory) GetHistory(ctx context.Context) []*Message {
	messages := make([]*Message, 0, m.window.Len())

	for e := m.window.Front(); e != nil; e = e.Next() {
		msg := e.Value.(*Message)

		// 检查是否过期
		if m.isExpired(msg) {
			m.window.Remove(e)
			continue
		}

		messages = append(messages, msg)
	}

	return messages
}

// GetMessagesForLLM 获取用于 LLM 的消息格式
func (m *SlidingWindowMemory) GetMessagesForLLM(ctx context.Context) []LLMMessage {
	messages := m.GetHistory(ctx)

	llmMessages := make([]LLMMessage, 0, len(messages))
	for _, msg := range messages {
		llmMessages = append(llmMessages, LLMMessage{
			Role:    msg.Role,
			Content: msg.Content,
		})
	}

	return llmMessages
}

// Clear 清空记忆
func (m *SlidingWindowMemory) Clear(ctx context.Context) error {
	m.window.Init()
	return nil
}

// isExpired 检查消息是否过期
func (m *SlidingWindowMemory) isExpired(msg *Message) bool {
	if m.retentionDays <= 0 {
		return false
	}

	expiredTime := time.Now().AddDate(0, 0, -m.retentionDays)
	return msg.CreatedAt.Before(expiredTime)
}
```

### 4.2 知识库检索算法

**混合检索（向量 + 全文）：**

```go
// pkg/knowledge/retriever/hybrid_retriever.go
package retriever

import (
	"context"
	"math"
)

// HybridRetriever 混合检索器
type HybridRetriever struct {
	vectorRetriever  VectorRetriever  // 向量检索
	fulltextRetriever FulltextRetriever // 全文检索
	reranker         Reranker         // 重排序器
	alpha           float64          // 向量检索权重（0-1）
}

// Retrieve 检索相关文档
func (r *HybridRetriever) Retrieve(ctx context.Context, query string, topK int) ([]*DocumentScore, error) {
	// 并行检索
	type result struct {
		docs []*DocumentScore
		err  error
	}

	vectorCh := make(chan result)
	fulltextCh := make(chan result)

	// 向量检索
	go func() {
		docs, err := r.vectorRetriever.Search(ctx, query, topK*2)
		vectorCh <- result{docs, err}
	}()

	// 全文检索
	go func() {
		docs, err := r.fulltextRetriever.Search(ctx, query, topK*2)
		fulltextCh <- result{docs, err}
	}()

	// 等待结果
	vectorResult := <-vectorCh
	fulltextResult := <-fulltextCh

	if vectorResult.err != nil {
		return nil, vectorResult.err
	}
	if fulltextResult.err != nil {
		return nil, fulltextResult.err
	}

	// 合并结果
	merged := r.mergeResults(vectorResult.docs, fulltextResult.docs, r.alpha)

	// 重排序
	if r.reranker != nil {
		merged = r.reranker.Rerank(ctx, query, merged)
	}

	// 返回 Top-K
	if len(merged) > topK {
		merged = merged[:topK]
	}

	return merged, nil
}

// mergeResults 合并向量和全文检索结果
func (r *HybridRetriever) mergeResults(
	vectorDocs []*DocumentScore,
	fulltextDocs []*DocumentScore,
	alpha float64,
) []*DocumentScore {
	scores := make(map[string]*MergedScore)

	// 处理向量检索结果
	for _, doc := range vectorDocs {
		scores[doc.DocumentID] = &MergedScore{
			DocumentID: doc.DocumentID,
			Document:   doc.Document,
			VectorScore:  doc.Score,
		}
	}

	// 处理全文检索结果
	for _, doc := range fulltextDocs {
		if merged, ok := scores[doc.DocumentID]; ok {
			merged.FulltextScore = doc.Score
		} else {
			scores[doc.DocumentID] = &MergedScore{
				DocumentID:     doc.DocumentID,
				Document:       doc.Document,
				FulltextScore: doc.Score,
			}
		}
	}

	// 计算综合得分
	results := make([]*DocumentScore, 0, len(scores))
	for _, merged := range scores {
		// 归一化得分
		vectorScore := merged.VectorScore
		fulltextScore := merged.FulltextScore

		// 计算综合得分（加权融合）
		finalScore := alpha*vectorScore + (1-alpha)*fulltextScore

		results = append(results, &DocumentScore{
			DocumentID: merged.DocumentID,
			Document:   merged.Document,
			Score:      finalScore,
		})
	}

	// 按得分排序
	sort.Slice(results, func(i, j int) bool {
		return results[i].Score > results[j].Score
	})

	return results
}

type MergedScore struct {
	DocumentID      string
	Document        *Document
	VectorScore     float64
	FulltextScore   float64
}
```

### 4.3 工作流 DAG 调度算法

**拓扑排序执行：**

```go
// pkg/workflow/executor/dag_executor.go
package executor

import (
	"context"
	"sync"
)

// DAGExecutor DAG 执行器
type DAGExecutor struct {
	workflow   *Workflow
	execCtx    *ExecutionContext
	visited    map[string]bool
	executing  map[string]bool
	mu         sync.RWMutex
}

// Execute 执行工作流
func (e *DAGExecutor) Execute(ctx context.Context, input map[string]interface{}) (map[string]interface{}, error) {
	e.execCtx = NewExecutionContext(input)
	e.visited = make(map[string]bool)
	e.executing = make(map[string]bool)

	// 找到开始节点
	startNodes := e.findStartNodes()

	// 并行执行开始节点（如果有多个）
	var wg sync.WaitGroup
	errCh := make(chan error, len(startNodes))

	for _, node := range startNodes {
		wg.Add(1)
		go func(n *Node) {
			defer wg.Done()
			if err := e.executeNode(ctx, n); err != nil {
				errCh <- err
			}
		}(node)
	}

	wg.Wait()
	close(errCh)

	// 检查错误
	for err := range errCh {
		if err != nil {
			return nil, err
		}
	}

	return e.execCtx.GetOutput(), nil
}

// executeNode 执行节点
func (e *DAGExecutor) executeNode(ctx context.Context, node *Node) error {
	e.mu.Lock()
	if e.visited[node.ID] {
		e.mu.Unlock()
		return nil
	}
	if e.executing[node.ID] {
		e.mu.Unlock()
		return fmt.Errorf("circular dependency detected: %s", node.ID)
	}
	e.executing[node.ID] = true
	e.mu.Unlock()

	defer func() {
		e.mu.Lock()
		e.executing[node.ID] = false
		e.visited[node.ID] = true
		e.mu.Unlock()
	}()

	// 检查并执行依赖节点
	dependencies := e.getDependencies(node)
	for _, dep := range dependencies {
		if err := e.executeNode(ctx, dep); err != nil {
			return err
		}
	}

	// 获取输入数据
	input, err := e.getNodeInput(ctx, node)
	if err != nil {
		return err
	}

	// 执行节点逻辑
	output, err := e.executeNodeLogic(ctx, node, input)
	if err != nil {
		return err
	}

	// 保存输出
	e.execCtx.SetNodeOutput(node.ID, output)

	return nil
}

// executeNodeLogic 执行节点逻辑
func (e *DAGExecutor) executeNodeLogic(ctx context.Context, node *Node, input map[string]interface{}) (map[string]interface{}, error) {
	switch node.Type {
	case NodeTypeStart:
		return input, nil

	case NodeTypeEnd:
		return input, nil

	case NodeTypeLLM:
		return e.executeLLMNode(ctx, node, input)

	case NodeTypeAPICall:
		return e.executeAPICallNode(ctx, node, input)

	case NodeTypeCondition:
		return e.executeConditionNode(ctx, node, input)

	case NodeTypeLoop:
		return e.executeLoopNode(ctx, node, input)

	default:
		return nil, fmt.Errorf("unsupported node type: %s", node.Type)
	}
}
```

```go
	}

	// 使用 GCM 模式
	gcm, err := cipher.NewGCM(block)
	if err != nil {
		return nil, err
	}

	nonce := make([]byte, gcm.NonceSize())
	if _, err := io.ReadFull(rand.Reader, nonce); err != nil {
		return nil, err
	}

	ciphertext := gcm.Seal(nonce, nonce, plaintext, nil)
	return ciphertext, nil
}
```

```go
// pkg/rate/limiter.go
type RateLimiter struct {
	redis  *redis.Client
	limits map[string]int // resource -> requests per minute
}

func (r *RateLimiter) Allow(ctx context.Context, userID string, resource string) bool {
	key := fmt.Sprintf("rate_limit:%s:%s", userID, resource)

	limit := r.limits[resource]
	if limit == 0 {
		limit = 100 // 默认限制
	}

	// 使用 Redis INCR 实现计数器
	count, err := r.redis.Incr(ctx, key).Result()
	if err != nil {
		return false
	}

	if count == 1 {
		// 设置过期时间 1 分钟
		r.redis.Expire(ctx, key, time.Minute)
	}

	return count <= int64(limit)
}
```

## 7. 性能优化设计

### 7.1 缓存策略

**多级缓存：**
```
L1: 进程内缓存（sync.Map）
  ↓ Miss
L2: Redis 缓存
  ↓ Miss
L3: 数据库
```

**缓存实现：**
```go
// pkg/cache/multi_level_cache.go
type MultiLevelCache struct {
	l1Cache *sync.Map          // 进程内缓存
	l2Cache *redis.Client      // Redis
	ttl     time.Duration
}

func (c *MultiLevelCache) Get(ctx context.Context, key string) (interface{}, error) {
	// L1 缓存
	if val, ok := c.l1Cache.Load(key); ok {
		return val, nil
	}

	// L2 缓存
	val, err := c.l2Cache.Get(ctx, key).Result()
	if err == redis.Nil {
		return nil, ErrCacheNotFound
	}
	if err != nil {
		return nil, err
	}

	// 回写 L1
	c.l1Cache.Store(key, val)

	return val, nil
}

func (c *MultiLevelCache) Set(ctx context.Context, key string, value interface{}) error {
	// 写入 L1
	c.l1Cache.Store(key, value)

	// 写入 L2
	return c.l2Cache.Set(ctx, key, value, c.ttl).Err()
}
```
