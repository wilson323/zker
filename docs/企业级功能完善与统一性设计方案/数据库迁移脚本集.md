### 1.4 添加反范式化统计字段

**场景**: 优化查询性能

```sql
-- ============================================
-- 迁移脚本: 添加统计字段
-- 执行时间: 5-10分钟(含数据初始化)
-- ============================================

-- 步骤1: 添加统计字段
ALTER TABLE bot_conversations
    ADD COLUMN message_count INT DEFAULT 0 COMMENT '消息数量' AFTER title,
    ADD INDEX idx_message_count (message_count DESC);

ALTER TABLE bots
    ADD COLUMN conversation_count INT DEFAULT 0 COMMENT '对话数量' AFTER version,
    ADD COLUMN message_count INT DEFAULT 0 COMMENT '消息数量' AFTER conversation_count,
    ADD COLUMN follower_count INT DEFAULT 0 COMMENT '关注者数量' AFTER message_count,
    ADD INDEX idx_conversation_count (conversation_count DESC),
    ADD INDEX idx_follower_count (follower_count DESC);

ALTER TABLE bot_store_listings
    ADD COLUMN view_count INT DEFAULT 0 COMMENT '浏览次数' AFTER pricing_type,
    ADD COLUMN clone_count INT DEFAULT 0 COMMENT '复克次数' AFTER view_count,
    ADD COLUMN rating_avg DECIMAL(3, 2) COMMENT '平均评分' AFTER clone_count,
    ADD COLUMN rating_count INT DEFAULT 0 COMMENT '评价数量' AFTER rating_avg,
    ADD INDEX idx_clone_count (clone_count DESC),
    ADD INDEX idx_rating_avg (rating_avg DESC);

-- 步骤2: 初始化统计字段(从关联表计算)
-- 初始化bot_conversations.message_count
UPDATE bot_conversations c
SET message_count = (
    SELECT COUNT(*)
    FROM bot_messages m
    WHERE m.conversation_id = c.id
);

-- 初始化bots.message_count
UPDATE bots b
SET conversation_count = (
    SELECT COUNT(*)
    FROM bot_conversations c
    WHERE c.bot_id = b.id AND c.deleted_at IS NULL
),
message_count = (
    SELECT COUNT(*)
    FROM bot_messages m
    WHERE m.bot_id = b.id
);

-- 初始化bot_store_listings.rating_avg和rating_count
UPDATE bot_store_listings l
SET rating_avg = (
    SELECT AVG(rating)
    FROM bot_store_reviews r
    WHERE r.listing_id = l.id
),
rating_count = (
    SELECT COUNT(*)
    FROM bot_store_reviews r
    WHERE r.listing_id = l.id
);

-- 步骤3: 创建触发器(自动维护统计字段)
DELIMITER $$

-- 触发器: 插入消息时更新对话的消息计数
CREATE TRIGGER trg_bot_messages_insert
AFTER INSERT ON bot_messages
FOR EACH ROW
BEGIN
    UPDATE bot_conversations
    SET message_count = message_count + 1
    WHERE id = NEW.conversation_id;
END$$

-- 触发器: 删除消息时更新对话的消息计数
CREATE TRIGGER trg_bot_messages_delete
AFTER DELETE ON bot_messages
FOR EACH ROW
BEGIN
    UPDATE bot_conversations
    SET message_count = message_count - 1
    WHERE id = OLD.conversation_id;
END$$

DELIMITER ;

-- 步骤4: 验证统计字段
SELECT
    b.name AS bot_name,
    b.conversation_count,
    b.message_count,
    (SELECT COUNT(*) FROM bot_conversations c WHERE c.bot_id = b.id AND c.deleted_at IS NULL) AS actual_conv_count,
    (SELECT COUNT(*) FROM bot_messages m WHERE m.bot_id = b.id) AS actual_msg_count
FROM bots b
WHERE b.deleted_at IS NULL
LIMIT 10;
```

## 2. 数据分区迁移

### 2.1 按时间分区 (日志表)

**场景**: 大表按时间分区,提升查询性能和数据管理

```sql
-- ============================================
-- 迁移脚本: 按时间分区(月度分区)
-- 执行时间: 1-2小时(取决于数据量)
-- ============================================

-- 步骤1: 创建新分区表
CREATE TABLE bot_messages_partitioned (
    -- 字段定义与原表相同
    id BIGINT PRIMARY KEY AUTO_INCREMENT COMMENT '消息ID',
    tenant_id VARCHAR(64) NOT NULL COMMENT '租户ID',
    conversation_id BIGINT NOT NULL COMMENT '会话ID',
    bot_id BIGINT NOT NULL COMMENT 'Bot ID',
    user_id BIGINT NOT NULL COMMENT '用户ID',
    role ENUM('user', 'assistant', 'system') NOT NULL COMMENT '角色',
    content TEXT NOT NULL COMMENT '消息内容',
    content_type ENUM('text', 'image', 'audio', 'video', 'file') DEFAULT 'text' COMMENT '内容类型',
    attachments JSON COMMENT '附件列表',
    metadata JSON COMMENT '元数据',
    prompt_tokens INT DEFAULT 0 COMMENT '输入Token数',
    completion_tokens INT DEFAULT 0 COMMENT '输出Token数',
    total_tokens INT DEFAULT 0 COMMENT '总Token数',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',

    -- 索引
    INDEX idx_tenant_id (tenant_id),
    INDEX idx_conversation_id (conversation_id),
    INDEX idx_bot_id (bot_id),
    INDEX idx_user_id (user_id),
    INDEX idx_created_at (created_at DESC),

    FOREIGN KEY (conversation_id) REFERENCES bot_conversations(id) ON DELETE CASCADE,
    FOREIGN KEY (bot_id) REFERENCES bots(id) ON DELETE CASCADE,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE

) ENGINE=InnoDB
DEFAULT CHARSET=utf8mb4
COMMENT='Bot消息表(分区)'
PARTITION BY RANGE (TO_DAYS(created_at)) (
    PARTITION p202501 VALUES LESS THAN (TO_DAYS('2025-02-01')),
    PARTITION p202502 VALUES LESS THAN (TO_DAYS('2025-03-01')),
    PARTITION p202503 VALUES LESS THAN (TO_DAYS('2025-04-01')),
    PARTITION p202504 VALUES LESS THAN (TO_DAYS('2025-05-01')),
    PARTITION p202505 VALUES LESS THAN (TO_DAYS('2025-06-01')),
    PARTITION p202506 VALUES LESS THAN (TO_DAYS('2025-07-01')),
    PARTITION p202507 VALUES LESS THAN (TO_DAYS('2025-08-01')),
    PARTITION p202508 VALUES LESS THAN (TO_DAYS('2025-09-01')),
    PARTITION p202509 VALUES LESS THAN (TO_DAYS('2025-10-01')),
    PARTITION p202510 VALUES LESS THAN (TO_DAYS('2025-11-01')),
    PARTITION p202511 VALUES LESS THAN (TO_DAYS('2025-12-01')),
    PARTITION p202512 VALUES LESS THAN (TO_DAYS('2026-01-01')),
    PARTITION p_future VALUES LESS THAN MAXVALUE
);

-- 步骤2: 迁移数据(分批执行,避免锁表)
-- 批次1: 迁移2025-01数据
INSERT INTO bot_messages_partitioned
SELECT * FROM bot_messages
WHERE created_at >= '2025-01-01' AND created_at < '2025-02-01';

-- 批次2: 迁移2025-02数据
INSERT INTO bot_messages_partitioned
SELECT * FROM bot_messages
WHERE created_at >= '2025-02-01' AND created_at < '2025-03-01';

-- ... (其他批次)

-- 步骤3: 验证数据完整性
SELECT
    '原表' AS source,
    COUNT(*) AS count,
    MIN(created_at) AS min_date,
    MAX(created_at) AS max_date
FROM bot_messages
UNION ALL
SELECT
    '分区表' AS source,
    COUNT(*) AS count,
    MIN(created_at) AS min_date,
    MAX(created_at) AS max_date
FROM bot_messages_partitioned;

-- 步骤4: 重命名表(切换到新表)
-- 业务低峰期执行
RENAME TABLE

-- 步骤5: 监控一段时间后删除旧表
-- DROP TABLE bot_messages_old;

-- 步骤6: 自动管理分区(存储过程)
DELIMITER $$

CREATE PROCEDURE sp_add_monthly_partition()
BEGIN
    DECLARE next_month DATE;
    DECLARE partition_name VARCHAR(20);
    DECLARE partition_value VARCHAR(50);

    -- 计算下个月1号
    SET next_month = DATE_ADD(DATE_FORMAT(NOW(), '%Y-%m-01'), INTERVAL 2 MONTH);
    SET partition_name = CONCAT('p', DATE_FORMAT(next_month, '%Y%m'));
    SET partition_value = CONCAT('TO_DAYS(''', DATE_FORMAT(next_month, '%Y-%m-01'), ''')');

        PARTITION ', partition_name, ' VALUES LESS THAN (', partition_value, '),
        PARTITION p_future VALUES LESS THAN MAXVALUE
    )');

    PREPARE stmt FROM @sql;
    EXECUTE stmt;
    DEALLOCATE PREPARE stmt;
END$$
```

## 3. 数据归档脚本

### 3.1 归档历史数据

**场景**: 将历史数据迁移到归档表,降低主表数据量

```sql
-- ============================================
-- 迁移脚本: 归档1年前的对话数据
-- 执行时间: 1-2小时
-- ============================================

-- 步骤1: 创建归档表(结构与主表相同)
CREATE TABLE bot_conversations_archive LIKE bot_conversations;
ALTER TABLE bot_conversations_archive COMMENT 'Bot对话归档表(1年前数据)';

-- 步骤2: 创建归档任务存储过程
DELIMITER $$

CREATE PROCEDURE sp_archive_old_conversations()
BEGIN
    DECLARE archive_date DATE;
    DECLARE affected_rows INT;

    -- 归档截止日期: 1年前
    SET archive_date = DATE_SUB(CURDATE(), INTERVAL 1 YEAR);

    -- 开始事务
    START TRANSACTION;

    -- 迁移数据到归档表
    INSERT INTO bot_conversations_archive
    SELECT * FROM bot_conversations
    WHERE created_at < archive_date
      AND status IN ('archived', 'deleted');

    SET affected_rows = ROW_COUNT();

    -- 删除主表中的已归档数据
    DELETE FROM bot_conversations
    WHERE created_at < archive_date
      AND status IN ('archived', 'deleted');

    -- 提交事务
    COMMIT;

    -- 输出归档统计
    SELECT
        archive_date AS 归档截止日期,
        affected_rows AS 归档行数,
        NOW() AS 归档时间;
END$$

DELIMITER ;

-- 步骤3: 执行归档
CALL sp_archive_old_conversations();

-- 步骤4: 验证归档结果
SELECT
    COUNT(*) AS 主表数据量,
    (SELECT COUNT(*) FROM bot_conversations_archive) AS 归档表数据量
FROM bot_conversations;
```

### 3.2 归档Bot消息

**场景**: 归档3个月前的Bot消息

```sql
-- ============================================
-- 迁移脚本: 归档历史消息
-- 执行时间: 2-3小时
-- ============================================

-- 步骤1: 创建归档表
CREATE TABLE bot_messages_archive LIKE bot_messages;
ALTER TABLE bot_messages_archive COMMENT 'Bot消息归档表(3个月前数据)';

-- 步骤2: 归档存储过程
DELIMITER $$

CREATE PROCEDURE sp_archive_old_messages()
BEGIN
    DECLARE archive_date DATE;
    DECLARE affected_rows INT;
    DECLARE batch_size INT DEFAULT 10000;  -- 每批处理1万条
    DECLARE done INT DEFAULT FALSE;
    DECLARE archived_count INT DEFAULT 0;

    SET archive_date = DATE_SUB(CURDATE(), INTERVAL 3 MONTH);

    -- 循环归档(分批处理,避免锁表超时)
    batch_loop: WHILE NOT done DO
        START TRANSACTION;

        -- 迁移一批数据
        INSERT INTO bot_messages_archive
        SELECT * FROM bot_messages
        WHERE created_at < archive_date
        LIMIT batch_size;

        SET affected_rows = ROW_COUNT();
        SET archived_count = archived_count + affected_rows;

        -- 删除这批数据
        DELETE FROM bot_messages
        WHERE id IN (
            SELECT id FROM (
                SELECT id FROM bot_messages_archive
                WHERE created_at < archive_date
                LIMIT batch_size
            ) AS tmp
        );

        COMMIT;

        -- 如果没有数据了,退出循环
        IF affected_rows = 0 THEN
            SET done = TRUE;
        END IF;
    END WHILE;

    -- 输出统计
    SELECT
        archive_date AS 归档截止日期,
        archived_count AS 归档总行数,
        NOW() AS 归档时间;
END$$

DELIMITER ;

-- 步骤3: 执行归档
CALL sp_archive_old_messages();
```

```sql
-- ============================================
-- 迁移脚本: 垂直分表(冷热分离)
-- 执行时间: 1-2小时
-- ============================================

-- 步骤1: 创建热点表(频繁访问的字段)
CREATE TABLE bots_hot (
    id BIGINT PRIMARY KEY,
    tenant_id VARCHAR(64) NOT NULL,
    name VARCHAR(255) NOT NULL,
    status VARCHAR(50) NOT NULL,
    version INT DEFAULT 1,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

    INDEX idx_tenant_id (tenant_id),
    INDEX idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='Bot热点表';

-- 步骤2: 创建冷表(不常访问的字段)
CREATE TABLE bots_cold (
    bot_id BIGINT PRIMARY KEY,
    description TEXT,
    long_config JSON,
    settings JSON,
    analytics_data JSON,
    metadata JSON,

    FOREIGN KEY (bot_id) REFERENCES bots_hot(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='Bot冷表';

-- 步骤3: 迁移数据
-- 迁移热点字段
INSERT INTO bots_hot (id, tenant_id, name, status, version, created_at, updated_at)
SELECT id, tenant_id, name, status, version, created_at, updated_at
FROM bots;

-- 迁移冷字段
INSERT INTO bots_cold (bot_id, description, long_config, settings, analytics_data, metadata)
SELECT id, description, long_config, settings, analytics_data, metadata
FROM bots;

-- 步骤4: 创建视图(兼容应用层查询)
CREATE OR REPLACE VIEW v_bots AS
SELECT
    h.*,
    c.description,
    c.long_config,
    c.settings,
    c.analytics_data,
    c.metadata
FROM bots_hot h
LEFT JOIN bots_cold c ON h.id = c.bot_id;

-- 步骤5: 应用层使用视图或直接查询热点表
-- 高频查询: SELECT * FROM bots_hot WHERE status = 'published';
-- 低频查询: SELECT * FROM v_bots WHERE id = ?;
```

## 7. 迁移最佳实践

### 7.1 迁移前准备清单

```
□ 数据备份(全库+增量)
□ 迁移脚本准备
□ 测试环境验证
□ 性能测试对比
□ 回滚方案准备
□ 迁移时间窗口确定
□ 监控告警配置
□ 应急联系人确认
```

### 7.2 迁移执行流程

```
1. 制定迁移计划
   └─ 包含: 迁移步骤、时间安排、风险评估、回滚方案

2. 测试环境验证
   └─ 在测试环境完整执行迁移流程,验证脚本正确性

3. 数据备份
   └─ 全库备份 + 增量备份

4. 执行迁移
   └─ 选择业务低峰期执行(如凌晨2-6点)

5. 数据验证
   └─ 行数对比、校验和对比、业务数据验证

6. 应用验证
   └─ 功能测试、性能测试

7. 监控观察
   └─ 观察监控指标,确保无异常

8. 清理旧数据
   └─ 确认无问题后,删除旧表或备份数据
```

### 7.3 风险控制

```
风险1: 数据丢失
  ├─ 预防: 多重备份
  ├─ 监控: 实时数据对比
  └─ 应对: 立即回滚,从备份恢复

风险2: 迁移时间长,影响业务
  ├─ 预防: 分批迁移,每次处理小批量
  ├─ 监控: 监控迁移进度
  └─ 应对: 暂停迁移,恢复服务

风险3: 性能下降
  ├─ 预防: 测试环境性能测试
  ├─ 监控: 实时性能监控
  └─ 应对: 调整索引或配置,或回滚

风险4: 应用不兼容
  ├─ 预防: 提前修改应用代码
  ├─ 监控: 应用错误日志监控
  └─ 应对: 立即回滚,修复代码后重新迁移
```

## 8. 常用运维脚本

### 8.1 检查表大小

```sql
SELECT
    TABLE_NAME,
    ROUND((DATA_LENGTH + INDEX_LENGTH) / 1024 / 1024, 2) AS total_mb,
    ROUND(DATA_LENGTH / 1024 / 1024, 2) AS data_mb,
    ROUND(INDEX_LENGTH / 1024 / 1024, 2) AS index_mb,
    TABLE_ROWS
FROM information_schema.TABLES
WHERE TABLE_SCHEMA = 'coze_studio'
  AND TABLE_TYPE = 'BASE TABLE'
ORDER BY (DATA_LENGTH + INDEX_LENGTH) DESC;
```

### 8.2 检查索引使用情况

```sql
SELECT
    object_schema,
    object_name,
    index_name,
    count_star AS usage_count,
    ROUND(sum_timer_wait / 1000000000000, 2) AS total_time_sec
FROM performance_schema.table_io_waits_summary_by_index_usage
WHERE object_schema = 'coze_studio'
  AND index_name IS NOT NULL
  AND index_name != 'PRIMARY'
ORDER BY count_star DESC;
```

### 8.3 检查慢查询

```sql
SELECT
    query_time,
    lock_time,
    rows_sent,
    rows_examined,
    sql_text
FROM mysql.slow_log
WHERE start_time > DATE_SUB(NOW(), INTERVAL 1 DAY)
ORDER BY query_time DESC
LIMIT 20;
```
