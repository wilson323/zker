```go
// infra/cache/permission_cache.go
type PermissionCache struct {
    client *redis.Client
    ttl    time.Duration // 5分钟
}

func (c *PermissionCache) GetPermission(ctx context.Context, userID, resource string) (*Permission, error) {
    key := fmt.Sprintf("perm:%s:%s", userID, resource)
    val, err := c.client.Get(ctx, key).Result()
    if err == nil {
        return decodePermission(val)
    }
    return nil, err
}
```
